    95→    95→function useCopy(): [boolean, (text: string) => void] {
    96→    96→  const [copied, setCopied] = useState(false)
    97→    97→  const copy = useCallback((text: string) => {
    98→    98→    const done = () => {
    99→    99→      setCopied(true)
   100→   100→      setTimeout(() => setCopied(false), 1600)
   101→   101→    }
   102→   102→    if (navigator.clipboard?.writeText) {
   103→   103→      navigator.clipboard.writeText(text).then(done).catch(done)
   104→   104→    } else {
   105→   105→      const ta = document.createElement('textarea')
   106→   106→      ta.value = text
   107→   107→      document.body.appendChild(ta)
   108→   108→      ta.select()
   109→   109→      document.execCommand('copy')
   110→   110→      document.body.removeChild(ta)
   111→   111→      done()
   112→   112→    }
   113→   113→  }, [])
   114→   114→  return [copied, copy]
   115→   115→}

   259→   259→  async function uploadFiles(list: FileList | File[]) {
   260→   260→    const incoming = Array.from(list)
   261→   261→    if (incoming.length === 0 || uploadingCount > 0) return
   262→   262→    const room = MAX_FILES - files.length
   263→   263→    if (room <= 0) {
   264→   264→      setError(`Up to ${MAX_FILES} attachments per message.`)
   265→   265→      return
   266→   266→    }
   267→   267→    setError(null)
   268→   268→    setUploadingCount(incoming.length)
   269→   269→    const batch = incoming.slice(0, room)
   270→   270→    let failures = 0
   271→   271→    for (const f of batch) {
   272→   272→      try {
   273→   273→        await uploadOne(f)
   274→   274→      } catch (e) {
   275→   275→        failures += 1
   276→   276→        setError((e as Error).message)
   277→   277→      } finally {
   278→   278→        setUploadingCount((c) => Math.max(0, c - 1))
   279→   279→      }
   280→   280→    }
   281→   281→    if (failures > 0 && batch.length > failures) {
   282→   282→      setError('Some files were rejected — the message explains each failure.')
   283→   283→    }
   284→   284→  }

   290→   290→  /** Regenerate the response to a SPECIFIC user turn (consistent actions on
   291→   291→   *  every assistant message). For the trailing exchange this is the classic
   292→   292→   *  retry; for earlier messages the server removes that exchange and
   293→   293→   *  everything after it, then answers the same user turn again. */
   294→   294→  function regenerateFrom(asstMsg: Msg) {
   295→   295→    if (streaming) return
   296→   296→    const idx = messages.findIndex((m) => m.id === asstMsg.id)
   297→   297→    if (idx < 0) return
   298→   298→    let userIdx = -1
   299→   299→    for (let i = idx - 1; i >= 0; i -= 1) {
   300→   300→      if (messages[i].role === 'user') { userIdx = i; break }
   301→   301→    }
   302→   302→    if (userIdx < 0) return
   303→   303→    const userContent = messages[userIdx].content
   304→   304→    const isTrailing = idx === messages.length - 1
   305→   305→    // Without a persisted id a mid-conversation regenerate cannot truncate
   306→   306→    // server-side history — refuse rather than silently duplicating the turn.
   307→   307→    if (!isTrailing && !asstMsg.serverId) return
   308→   308→    // Drop everything from the user turn onward (locally; the server mirrors
   309→   309→    // this for earlier messages so history stays an honest sequence).
   310→   310→    setMessages(messages.slice(0, userIdx))
   311→   311→    setTimeout(() => void send(userContent, { retry: isTrailing && !asstMsg.serverId, retryFromId: isTrailing ? undefined : asstMsg.serverId }), 0)
   312→   312→  }

   314→   314→  async function send(overrideMessage?: string, opts?: { retry?: boolean; retryFromId?: string }) {
   315→   315→    const message = (overrideMessage ?? input).trim()
   316→   316→    if (!message || streaming) return
   317→   317→    if (overrideMessage === undefined) setInput('')
   318→   318→    setError(null)
   319→   319→    setStreaming(true)
   320→   320→    setResearching(false)
   321→   321→    setRunningTool(null)
   322→   322→    nearBottomRef.current = true // a sent message always scrolls into view
   323→   323→    streamedRef.current = ''
   324→   324→
   325→   325→    const userMsg: Msg = { id: `u-${Date.now()}`, role: 'user', content: message }
   326→   326→    const asstId = `a-${Date.now()}`
   327→   327→    setMessages((prev) => [...prev, userMsg, { id: asstId, role: 'assistant', content: '', pending: true }])
   328→   328→
   329→   329→    const controller = new AbortController()
   330→   330→    abortRef.current = controller
   331→   331→    const attachable = files.filter((f) => f.status === 'ready')
   332→   332→
   333→   333→    try {
   334→   334→      const res = await fetch(`/api/riishe/chat?stream=1${activeId ? `&conversationId=${activeId}` : ''}`, {
   335→   335→        method: 'POST',
   336→   336→        headers: { 'content-type': 'application/json' },
   337→   337→        body: JSON.stringify({
   338→   338→          message,
   339→   339→          conversationId: activeId ?? undefined,
   340→   340→          documentIds: attachable.length > 0 ? attachable.map((d) => d.id) : undefined,
   341→   341→          retry: opts?.retry ?? false,
   342→   342→          ...(opts?.retryFromId ? { retryFromMessageId: opts.retryFromId } : {}),
   343→   343→        }),
   344→   344→        signal: controller.signal,
   345→   345→      })
   346→   346→      if (!res.ok || !res.body) {
   347→   347→        // Keep the global identity in sync: a 401 here means the session is
   348→   348→        // gone server-side; drop the signed-in shell immediately.
   349→   349→        if (res.status === 401) window.dispatchEvent(new Event('bean:unauthorized'))
   350→   350→        const data = await res.json().catch(() => ({}))
   351→   351→        throw new Error(data?.error?.message ?? `Request failed (${res.status})`)
   352→   352→      }
   353→   353→      const reader = res.body.getReader()
   354→   354→      const decoder = new TextDecoder()
   355→   355→      let buffer = ''
   356→   356→      let convId: string | null = activeId
   357→   357→      let failed = false
   358→   358→
   359→   359→      while (true) {
   360→   360→        const { done, value } = await reader.read()
   361→   361→        if (done) break
   362→   362→        buffer += decoder.decode(value, { stream: true })
   363→   363→        const lines = buffer.split('\n')
   364→   364→        buffer = lines.pop() ?? ''
   365→   365→        for (const line of lines) {
   366→   366→          if (!line.startsWith('data: ')) continue
   367→   367→          let evt: {
   368→   368→            delta?: string
   369→   369→            reset?: boolean
   370→   370→            done?: boolean
   371→   371→            error?: string
   372→   372→            code?: string
   373→   373→            conversationId?: string
   374→   374→            messageId?: string
   375→   375→            partial?: boolean
   376→   376→            stage?: 'researching' | 'tools' | 'solving' | 'verifying'
   377→   377→            tool?: { name: string; status: 'started' | 'done' | 'failed' }
   378→   378→            meta?: MsgMeta
   379→   379→          }
   380→   380→          try { evt = JSON.parse(line.slice(6)) } catch { continue }
   381→   381→          if (evt.conversationId) convId = evt.conversationId
   382→   382→          if (evt.stage === 'researching') setResearching(true)
   383→   383→          if (evt.stage === 'tools') { setResearching(false); setRunningTool('working') }
   384→   384→          if (evt.stage === 'solving' || evt.stage === 'verifying') { setResearching(false); setRunningTool(null) }
   385→   385→          if (evt.tool) {
   386→   386→            if (evt.tool.status === 'started') setRunningTool(evt.tool.name)
   387→   387→            else setRunningTool(null)
   388→   388→          }
   389→   389→          if (evt.reset) {
   390→   390→            // Reset protocol: a verification rewrite replaced the streamed
   391→   391→            // draft — clear it so the final answer REPLACES, never appends.
   392→   392→            streamedRef.current = ''
   393→   393→            setMessages((prev) =>
   394→   394→              prev.map((m) => (m.id === asstId ? { ...m, content: '', pending: true } : m)),
   395→   395→            )
   396→   396→          }
   397→   397→          if (evt.error) { failed = true; setError(evt.error) }
   398→   398→          if (evt.delta) {
   399→   399→            streamedRef.current += evt.delta
   400→   400→            setMessages((prev) =>
   401→   401→              prev.map((m) => (m.id === asstId ? { ...m, content: streamedRef.current, pending: false } : m)),
   402→   402→            )
   403→   403→          }
   404→   404→          if (evt.done) {
   405→   405→            setResearching(false)
   406→   406→            setRunningTool(null)
   407→   407→            if (convId && !activeId) setActiveId(convId)
   408→   408→            if (evt.messageId) {
   409→   409→              setMessages((prev) => prev.map((m) => (m.id === asstId ? { ...m, serverId: evt.messageId } : m)))
   410→   410→            }
   411→   411→            if (evt.meta) {
   412→   412→              setMessages((prev) => prev.map((m) => (m.id === asstId ? { ...m, meta: evt.meta } : m)))
   413→   413→            }
   414→   414→          }
   415→   415→        }
   416→   416→      }
   417→   417→      // If nothing streamed (failed, or the server reported an empty partial),
   418→   418→      // drop the empty assistant bubble — the error banner already explains
   419→   419→      // what happened, and a perpetual "thinking" indicator would be a lie.
   420→   420→      if (streamedRef.current === '') {
   421→   421→        setMessages((prev) => prev.filter((m) => m.id !== asstId))
   422→   422→        // Server-side failure (e.g. upstream rate limit) reported over SSE:
   423→   423→        // the question goes back into the composer so a retry needs no
   424→   424→        // retyping. (Nothing was answered — no partial content is lost.)
   425→   425→        if (failed) restoreQuestionForRetry(message)
   426→   426→      } else if (failed) {
   427→   427→        setMessages((prev) => prev.map((m) => (m.id === asstId ? { ...m, partial: true, pending: false } : m)))
   428→   428→      }
   429→   429→      void loadConversations()
   430→   430→    } catch (e) {
   431→   431→      if ((e as Error).name === 'AbortError') {
   432→   432→        // User pressed stop: keep what arrived and label it honestly.
   433→   433→        if (streamedRef.current !== '') {
   434→   434→          setMessages((prev) => prev.map((m) => (m.id === asstId ? { ...m, partial: true, pending: false } : m)))
   435→   435→        } else {
   436→   436→          setMessages((prev) => prev.filter((m) => m.id !== asstId))
   437→   437→        }
   438→   438→      } else {
   439→   439→        setError((e as Error).message)
   440→   440→        setMessages((prev) => prev.filter((m) => m.id !== asstId || m.content.length > 0))
   441→   441→        restoreQuestionForRetry(message)
   442→   442→      }
   443→   443→    } finally {

   842→   842→                {s.date ? (
   843→   843→                  <span className="ml-1 text-white/30" title="Publication date stated by the source — distinct from the event date">
   844→   844→                    · published {s.date}
   845→   845→                  </span>
   846→   846→                ) : (
   847→   847→                  <span className="ml-1 text-white/25" title="No publication date could be verified for this source">
   848→   848→                    · no date verified
   849→   849→                  </span>
   850→   850→                )}
   851→   851→                {s.social ? <span className="ml-1 text-amber-300/50">· social-media post (unverified provenance)</span> : null}
   852→   852→                <SourceClaimLine content={claims.get(s.id)} />
   853→   853→              </li>
   854→   854→            ))}
   855→   855→          </ul>
   856→   856→        </details>
   857→   857→      ) : null}
   858→   858→    </div>
   859→   859→  )
   860→   860→}
