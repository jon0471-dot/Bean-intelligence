    92→    92→function useCopy(): [boolean, (text: string) => void] {
    93→    93→  const [copied, setCopied] = useState(false)
    94→    94→  const copy = useCallback((text: string) => {
    95→    95→    const done = () => {
    96→    96→      setCopied(true)
    97→    97→      setTimeout(() => setCopied(false), 1600)
    98→    98→    }
    99→    99→    if (navigator.clipboard?.writeText) {
   100→   100→      navigator.clipboard.writeText(text).then(done).catch(done)
   101→   101→    } else {
   102→   102→      const ta = document.createElement('textarea')
   103→   103→      ta.value = text
   104→   104→      document.body.appendChild(ta)
   105→   105→      ta.select()
   106→   106→      document.execCommand('copy')
   107→   107→      document.body.removeChild(ta)
   108→   108→      done()
   109→   109→    }
   110→   110→  }, [])
   111→   111→  return [copied, copy]
   112→   112→}

   276→   276→  async function send(overrideMessage?: string, retry = false) {
   277→   277→    const message = (overrideMessage ?? input).trim()
   278→   278→    if (!message || streaming) return
   279→   279→    if (overrideMessage === undefined) setInput('')
   280→   280→    setError(null)
   281→   281→    setStreaming(true)
   282→   282→    setResearching(false)
   283→   283→    setRunningTool(null)
   284→   284→    streamedRef.current = ''
   285→   285→
   286→   286→    const userMsg: Msg = { id: `u-${Date.now()}`, role: 'user', content: message }
   287→   287→    const asstId = `a-${Date.now()}`
   288→   288→    setMessages((prev) => [...prev, userMsg, { id: asstId, role: 'assistant', content: '', pending: true }])
   289→   289→
   290→   290→    const controller = new AbortController()
   291→   291→    abortRef.current = controller
   292→   292→    const attachable = files.filter((f) => f.status === 'ready')
   293→   293→
   294→   294→    try {
   295→   295→      const res = await fetch(`/api/riishe/chat?stream=1${activeId ? `&conversationId=${activeId}` : ''}`, {
   296→   296→        method: 'POST',
   297→   297→        headers: { 'content-type': 'application/json' },
   298→   298→        body: JSON.stringify({
   299→   299→          message,
   300→   300→          conversationId: activeId ?? undefined,
   301→   301→          documentIds: attachable.length > 0 ? attachable.map((d) => d.id) : undefined,
   302→   302→          retry,
   303→   303→        }),
   304→   304→        signal: controller.signal,
   305→   305→      })
   306→   306→      if (!res.ok || !res.body) {
   307→   307→        // Keep the global identity in sync: a 401 here means the session is
   308→   308→        // gone server-side; drop the signed-in shell immediately.
   309→   309→        if (res.status === 401) window.dispatchEvent(new Event('bean:unauthorized'))
   310→   310→        const data = await res.json().catch(() => ({}))
   311→   311→        throw new Error(data?.error?.message ?? `Request failed (${res.status})`)
   312→   312→      }
   313→   313→      const reader = res.body.getReader()
   314→   314→      const decoder = new TextDecoder()
   315→   315→      let buffer = ''
   316→   316→      let convId: string | null = activeId
   317→   317→      let failed = false
   318→   318→
   319→   319→      while (true) {
   320→   320→        const { done, value } = await reader.read()
   321→   321→        if (done) break
   322→   322→        buffer += decoder.decode(value, { stream: true })
   323→   323→        const lines = buffer.split('\n')
   324→   324→        buffer = lines.pop() ?? ''
   325→   325→        for (const line of lines) {
   326→   326→          if (!line.startsWith('data: ')) continue
   327→   327→          let evt: {
   328→   328→            delta?: string
   329→   329→            reset?: boolean
   330→   330→            done?: boolean
   331→   331→            error?: string
   332→   332→            conversationId?: string
   333→   333→            partial?: boolean
   334→   334→            stage?: 'researching' | 'tools' | 'solving' | 'verifying'
   335→   335→            tool?: { name: string; status: 'started' | 'done' | 'failed' }
   336→   336→            meta?: MsgMeta
   337→   337→          }
   338→   338→          try { evt = JSON.parse(line.slice(6)) } catch { continue }
   339→   339→          if (evt.conversationId) convId = evt.conversationId
   340→   340→          if (evt.stage === 'researching') setResearching(true)
   341→   341→          if (evt.stage === 'tools') { setResearching(false); setRunningTool('working') }
   342→   342→          if (evt.stage === 'solving' || evt.stage === 'verifying') { setResearching(false); setRunningTool(null) }
   343→   343→          if (evt.tool) {
   344→   344→            if (evt.tool.status === 'started') setRunningTool(evt.tool.name)
   345→   345→            else setRunningTool(null)
   346→   346→          }
   347→   347→          if (evt.reset) {
   348→   348→            // Reset protocol: a verification rewrite replaced the streamed
   349→   349→            // draft — clear it so the final answer REPLACES, never appends.
   350→   350→            streamedRef.current = ''
   351→   351→            setMessages((prev) =>
   352→   352→              prev.map((m) => (m.id === asstId ? { ...m, content: '', pending: true } : m)),
   353→   353→            )
   354→   354→          }
   355→   355→          if (evt.error) { failed = true; setError(evt.error) }
   356→   356→          if (evt.delta) {
   357→   357→            streamedRef.current += evt.delta
   358→   358→            setMessages((prev) =>
   359→   359→              prev.map((m) => (m.id === asstId ? { ...m, content: streamedRef.current, pending: false } : m)),
   360→   360→            )
   361→   361→          }
   362→   362→          if (evt.done) {
   363→   363→            setResearching(false)
   364→   364→            setRunningTool(null)
   365→   365→            if (convId && !activeId) setActiveId(convId)
   366→   366→            if (evt.meta) {
   367→   367→              setMessages((prev) => prev.map((m) => (m.id === asstId ? { ...m, meta: evt.meta } : m)))
   368→   368→            }
   369→   369→          }
   370→   370→        }
   371→   371→      }
   372→   372→      // If nothing streamed (failed, or the server reported an empty partial),
   373→   373→      // drop the empty assistant bubble — the error banner already explains
   374→   374→      // what happened, and a perpetual "thinking" indicator would be a lie.
   375→   375→      if (streamedRef.current === '') {
   376→   376→        setMessages((prev) => prev.filter((m) => m.id !== asstId))
   377→   377→      } else if (failed) {
   378→   378→        setMessages((prev) => prev.map((m) => (m.id === asstId ? { ...m, partial: true, pending: false } : m)))
   379→   379→      }
   380→   380→      void loadConversations()
   381→   381→    } catch (e) {
   382→   382→      if ((e as Error).name === 'AbortError') {
   383→   383→        // User pressed stop: keep what arrived and label it honestly.
   384→   384→        if (streamedRef.current !== '') {
   385→   385→          setMessages((prev) => prev.map((m) => (m.id === asstId ? { ...m, partial: true, pending: false } : m)))
   386→   386→        } else {
   387→   387→          setMessages((prev) => prev.filter((m) => m.id !== asstId))
   388→   388→        }
   389→   389→      } else {
   390→   390→        setError((e as Error).message)
   391→   391→        setMessages((prev) => prev.filter((m) => m.id !== asstId || m.content.length > 0))
   392→   392→      }
   393→   393→    } finally {
   394→   394→      setStreaming(false)
   395→   395→      setResearching(false)
   396→   396→      setRunningTool(null)
   397→   397→      abortRef.current = null
   398→   398→      setMessages((prev) => prev.map((m) => ({ ...m, pending: false })))
   399→   399→    }
   400→   400→  }
