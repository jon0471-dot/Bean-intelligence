    91→function useCopy(): [boolean, (text: string) => void] {
    92→  const [copied, setCopied] = useState(false)
    93→  const copy = useCallback((text: string) => {
    94→    const done = () => {
    95→      setCopied(true)
    96→      setTimeout(() => setCopied(false), 1600)
    97→    }
    98→    if (navigator.clipboard?.writeText) {
    99→      navigator.clipboard.writeText(text).then(done).catch(done)
   100→    } else {
   101→      const ta = document.createElement('textarea')
   102→      ta.value = text
   103→      document.body.appendChild(ta)
   104→      ta.select()
   105→      document.execCommand('copy')
   106→      document.body.removeChild(ta)
   107→      done()
   108→    }
   109→  }, [])
   110→  return [copied, copy]
   111→}

   120→        <button
   121→          onClick={() => copy(code)}
   122→          className="inline-flex items-center gap-1 rounded px-1.5 py-0.5 text-[10px] text-white/45 transition-colors hover:bg-white/10 hover:text-white/80"
   123→          aria-label="Copy code"

   125→          {copied ? <Check className="h-3 w-3 text-emerald-400" /> : <Copy className="h-3 w-3" />}
   126→          {copied ? 'Copied' : 'Copy'}

   144→  const [messages, setMessages] = useState<Msg[]>([])
   145→  const [input, setInput] = useState('')
   146→  const [streaming, setStreaming] = useState(false)
   147→  const [researching, setResearching] = useState(false)
   148→  const [runningTool, setRunningTool] = useState<string | null>(null)
   149→  const [researchQueries, setResearchQueries] = useState<string[]>([])
   150→  const [error, setError] = useState<string | null>(null)
   151→  const [documents, setDocuments] = useState<DocMeta[]>([])
   152→  const [uploadingDoc, setUploadingDoc] = useState(false)
   153→  const endRef = useRef<HTMLDivElement>(null)
   154→  const streamedRef = useRef('')
   155→  const abortRef = useRef<AbortController | null>(null)
   156→  const fileInputRef = useRef<HTMLInputElement>(null)
   157→
   158→  const loadConversations = useCallback(async () => {
   159→    try {
   160→      const d = await beanFetch<{ conversations: Conv[] }>('/api/riishe/chat')
   161→      setConversations(d.conversations)
   162→    } catch { /* non-fatal */ }
   163→  }, [])
   164→
   165→  useEffect(() => { void loadConversations() }, [loadConversations])
   166→
   167→  useEffect(() => {
   168→    endRef.current?.scrollIntoView({ behavior: 'smooth', block: 'end' })
   169→  }, [messages, streaming, researching, runningTool])

   171→  async function openConversation(id: string) {
   172→    setActiveId(id)
   173→    setError(null)
   174→    setDocuments([])
   175→    try {
   176→      const d = await beanFetch<{
   177→        conversation: { messages: Array<{ id: string; role: string; content: string; metadata?: MsgMeta | null }> }
   178→      }>(`/api/riishe/conversations/${id}`)
   179→      setMessages(
   180→        d.conversation.messages.map((m) => ({
   181→          id: m.id,
   182→          role: m.role as 'user' | 'assistant',
   183→          content: m.content,
   184→          meta: m.metadata ?? undefined,
   185→        })),
   186→      )
   187→    } catch (e) {
   188→      setError((e as Error).message)
   189→    }
   190→  }
   191→
   192→  async function deleteConversation(id: string) {
   193→    try {
   194→      await beanFetch(`/api/riishe/conversations/${id}`, { method: 'DELETE' })
   195→      if (activeId === id) { setActiveId(null); setMessages([]) }
   196→      void loadConversations()
   197→    } catch (e) {
   198→      setError((e as Error).message)
   199→    }
   200→  }
   201→
   202→  function newConversation() {
   203→    setActiveId(null)
   204→    setMessages([])
   205→    setError(null)
   206→    setDocuments([])
   207→  }
   208→
   209→  async function uploadDocument(file: File) {
   210→    if (uploadingDoc || documents.length >= 2) return
   211→    setUploadingDoc(true)
   212→    setError(null)
   213→    try {
   214→      const form = new FormData()
   215→      form.append('file', file)
   216→      const res = await fetch('/api/riishe/documents', { method: 'POST', body: form })
   217→      const data = await res.json().catch(() => ({}))
   218→      if (!res.ok) {
   219→        if (res.status === 401) window.dispatchEvent(new Event('bean:unauthorized'))
   220→        throw new Error(data?.error?.message ?? `Upload failed (${res.status})`)
   221→      }
   222→      setDocuments((prev) => [...prev, data.document].slice(0, 2))
   223→    } catch (e) {
   224→      setError((e as Error).message)
   225→    } finally {
   226→      setUploadingDoc(false)
   227→      if (fileInputRef.current) fileInputRef.current.value = ''
   228→    }
   229→  }
   230→
   231→  function stopGeneration() {
   232→    abortRef.current?.abort()
   233→  }
   234→
   235→  function regenerate() {
   236→    if (streaming) return
   237→    const lastUser = [...messages].reverse().find((m) => m.role === 'user')
   238→    if (!lastUser) return
   239→    setInput(lastUser.content)
   240→    setMessages((prev) => {
   241→      const copy = [...prev]
   242→      if (copy.length > 0 && copy[copy.length - 1].role === 'assistant') copy.pop()
   243→      return copy
   244→    })
   245→    // Send on the next tick with the restored input.
   246→    setTimeout(() => void send(lastUser.content, true), 0)
   247→  }
   248→
   249→  async function send(overrideMessage?: string, retry = false) {
   250→    const message = (overrideMessage ?? input).trim()
   251→    if (!message || streaming) return
   252→    if (overrideMessage === undefined) setInput('')
   253→    setError(null)
   254→    setStreaming(true)
   255→    setResearching(false)
   256→    setRunningTool(null)
   257→    setResearchQueries([])
   258→    streamedRef.current = ''
   259→
   260→    const userMsg: Msg = { id: `u-${Date.now()}`, role: 'user', content: message }
   261→    const asstId = `a-${Date.now()}`
   262→    setMessages((prev) => [...prev, userMsg, { id: asstId, role: 'assistant', content: '', pending: true }])
   263→
   264→    const controller = new AbortController()
   265→    abortRef.current = controller
   266→
   267→    try {
   268→      const res = await fetch(`/api/riishe/chat?stream=1${activeId ? `&conversationId=${activeId}` : ''}`, {
   269→        method: 'POST',
   270→        headers: { 'content-type': 'application/json' },
   271→        body: JSON.stringify({
   272→          message,
   273→          conversationId: activeId ?? undefined,
   274→          documentIds: documents.length > 0 ? documents.map((d) => d.id) : undefined,
   275→          retry,
   276→        }),
   277→        signal: controller.signal,
   278→      })
   279→      if (!res.ok || !res.body) {
   280→        // Keep the global identity in sync: a 401 here means the session is
   281→        // gone server-side; drop the signed-in shell immediately.
   282→        if (res.status === 401) window.dispatchEvent(new Event('bean:unauthorized'))
   283→        const data = await res.json().catch(() => ({}))
   284→        throw new Error(data?.error?.message ?? `Request failed (${res.status})`)
   285→      }
   286→      const reader = res.body.getReader()
   287→      const decoder = new TextDecoder()
   288→      let buffer = ''
   289→      let convId: string | null = activeId
   290→      let failed = false
   291→
   292→      while (true) {
   293→        const { done, value } = await reader.read()
   294→        if (done) break
   295→        buffer += decoder.decode(value, { stream: true })
   296→        const lines = buffer.split('\n')
   297→        buffer = lines.pop() ?? ''
   298→        for (const line of lines) {
   299→          if (!line.startsWith('data: ')) continue
   300→          let evt: {
   301→            delta?: string
   302→            done?: boolean
   303→            error?: string
   304→            conversationId?: string
   305→            partial?: boolean
   306→            stage?: 'researching' | 'tools' | 'solving' | 'verifying'
   307→            research?: { queries?: string[] }
   308→            tool?: { name: string; status: 'started' | 'done' | 'failed' }
   309→            meta?: MsgMeta
   310→          }
   311→          try { evt = JSON.parse(line.slice(6)) } catch { continue }
   312→          if (evt.conversationId) convId = evt.conversationId
   313→          if (evt.stage === 'researching') { setResearching(true); setResearchQueries([]) }
   314→          if (evt.stage === 'tools') { setResearching(false); setRunningTool('working') }
   315→          if (evt.stage === 'solving' || evt.stage === 'verifying') { setResearching(false); setRunningTool(null) }
   316→          if (evt.tool) {
   317→            if (evt.tool.status === 'started') setRunningTool(evt.tool.name)
   318→            else setRunningTool(null)
   319→          }
   320→          if (evt.research?.queries) setResearchQueries(evt.research.queries)
   321→          if (evt.error) { failed = true; setError(evt.error) }
   322→          if (evt.delta) {
   323→            streamedRef.current += evt.delta
   324→            setMessages((prev) =>
   325→              prev.map((m) => (m.id === asstId ? { ...m, content: streamedRef.current, pending: false } : m)),
   326→            )
   327→          }
   328→          if (evt.done) {
   329→            setResearching(false)
   330→            setRunningTool(null)
   331→            if (convId && !activeId) setActiveId(convId)
   332→            if (evt.meta) {
   333→              setMessages((prev) => prev.map((m) => (m.id === asstId ? { ...m, meta: evt.meta } : m)))
   334→            }
   335→          }
   336→        }
   337→      }
   338→      // If nothing streamed (failed, or the server reported an empty partial),
   339→      // drop the empty assistant bubble — the error banner already explains
   340→      // what happened, and a perpetual "thinking" indicator would be a lie.
   341→      if (streamedRef.current === '') {
   342→        setMessages((prev) => prev.filter((m) => m.id !== asstId))
   343→      } else if (failed) {
   344→        setMessages((prev) => prev.map((m) => (m.id === asstId ? { ...m, partial: true, pending: false } : m)))
   345→      }
   346→      void loadConversations()
   347→    } catch (e) {
   348→      if ((e as Error).name === 'AbortError') {
   349→        // User pressed stop: keep what arrived and label it honestly.
   350→        if (streamedRef.current !== '') {
   351→          setMessages((prev) => prev.map((m) => (m.id === asstId ? { ...m, partial: true, pending: false } : m)))
   352→        } else {
   353→          setMessages((prev) => prev.filter((m) => m.id !== asstId))
   354→        }
   355→      } else {
   356→        setError((e as Error).message)
   357→        setMessages((prev) => prev.filter((m) => m.id !== asstId || m.content.length > 0))
   358→      }
   359→    } finally {
   360→      setStreaming(false)
   361→      setResearching(false)
   362→      setRunningTool(null)
   363→      abortRef.current = null
   364→      setMessages((prev) => prev.map((m) => ({ ...m, pending: false })))
   365→    }
   366→  }
   367→
   368→  const citationComponents = (meta?: MsgMeta) => ({
   369→    a: ({ href, children }: { href?: string; children?: React.ReactNode }) => {
   370→      if (href?.startsWith('#bean-src-')) {

   375→            title="Jump to source"
   376→          >
   377→            {children}

   379→        )
   380→      }
   381→      return (
   382→        <a href={href} target="_blank" rel="noopener noreferrer" className="text-[#f0c75e] underline decoration-[rgba(212,175,55,0.4)] hover:decoration-[#f0c75e]">
   383→          {children}
   384→        </a>
   385→      )
   386→    },
   387→    pre: ({ children }: { children?: React.ReactNode }) => {
   388→      // React-markdown nests <code> inside <pre>; extract for highlighting.

   396→    ...(meta ? {} : {}),

   398→
   399→  const lastAssistantId = [...messages].reverse().find((m) => m.role === 'assistant' && !m.pending)?.id

   401→  return (

   406→          <Button onClick={newConversation} className="w-full justify-start bg-[rgba(212,175,55,0.12)] text-[#f0c75e] hover:bg-[rgba(212,175,55,0.2)]" variant="outline">
   407→            <Plus className="mr-1.5 h-4 w-4" /> New conversation
   408→          </Button>
   409→        </div>
   410→        <ScrollArea className="flex-1 px-2 pb-3">
   411→          {conversations.length === 0 ? (
   412→            <p className="px-2 py-6 text-center text-xs text-white/30">No conversations yet</p>
   413→          ) : (
   414→            conversations.map((c) => (
   415→              <div
   416→                key={c.id}
   417→                className={cn(
   418→                  'group mb-0.5 flex cursor-pointer items-center justify-between rounded-lg px-3 py-2 text-[13px] transition-colors',
   419→                  activeId === c.id ? 'bg-[rgba(212,175,55,0.1)] text-[#f0c75e]' : 'text-white/60 hover:bg-white/5',
   420→                )}
   421→                onClick={() => openConversation(c.id)}
   422→              >
   423→                <div className="min-w-0">
   424→                  <div className="truncate">{c.title}</div>
   425→                  <div className="text-[10px] text-white/30">{c.messages} msgs · {timeAgo(c.updatedAt)}</div>

   430→                  aria-label={`Delete ${c.title}`}
   431→                >
   432→                  <Trash2 className="h-3.5 w-3.5" />
   433→                </button>
   434→              </div>
   435→            ))
   436→          )}
   437→        </ScrollArea>
   438→      </aside>
   439→
   440→      {/* Chat column */}
   441→      <div className="flex min-w-0 flex-1 flex-col">
   442→        <div className="flex items-center gap-2 border-b border-[rgba(212,175,55,0.1)] px-5 py-3">
   443→          <MessageSquareText className="h-4 w-4 text-[#d4af37]" />
   444→          <div className="text-sm font-semibold tracking-wide">RIISHE</div>
   445→          <span className="text-xs text-white/35">Conversational AI · research · verified reasoning</span>
   446→        </div>
   447→
   448→        <div className="bean-scroll flex-1 overflow-y-auto px-5 py-4">
   449→          {messages.length === 0 ? (
   450→            <EmptyState
   451→              icon={<MessageSquareText className="h-10 w-10" />}
   452→              title="Start a conversation"
   453→              hint="Ask questions, request derivations, code help or research on current topics. RIISHE searches live sources when a question needs them, runs real tools, and verifies its arithmetic before answering. Attach a document for grounded analysis."
   454→            />
   455→          ) : (
   456→            <div className="mx-auto max-w-3xl space-y-4">
   457→              {messages.map((m) => (
   458→                <div key={m.id} className={cn('flex', m.role === 'user' ? 'justify-end' : 'justify-start')}>
   459→                  <div
   460→                    className={cn(
   461→                      'max-w-[85%] rounded-2xl px-4 py-3',
   462→                      m.role === 'user'
   463→                        ? 'bg-[rgba(212,175,55,0.14)] text-white/90'
   464→                        : 'bean-glass text-white/85',
   465→                    )}
   466→                  >
   467→                    {m.role === 'user' ? (
   468→                      <p className="whitespace-pre-wrap text-[14.5px] leading-relaxed">{m.content}</p>
   469→                    ) : m.content ? (
   470→                      <>
   471→                        <Markdown content={linkifyCitations(m.content, m.meta?.sources)} components={citationComponents(m.meta)} />
   472→                        {m.pending ? <Spinner className="ml-1 mt-1" /> : null}
   473→                        {m.partial ? (
   474→                          <div className="mt-2 inline-flex items-center gap-1.5 rounded border border-amber-500/30 bg-amber-500/10 px-2 py-0.5 text-[10px] text-amber-300">
   475→                            <AlertTriangle className="h-3 w-3" /> Incomplete — the stream was stopped or ended early
   476→                          </div>
   477→                        ) : null}
   478→                        {m.meta ? <AnswerEvidence meta={m.meta} /> : null}
   479→                        {m.id === lastAssistantId && !m.pending ? (
   480→                          <div className="mt-2 flex items-center gap-2 border-t border-[rgba(212,175,55,0.12)] pt-2">
   481→                            <CopyMessageButton content={m.content} />
   482→                            <button
   483→                              onClick={regenerate}
   484→                              className="inline-flex items-center gap-1 rounded px-1.5 py-0.5 text-[10px] text-white/45 transition-colors hover:bg-white/10 hover:text-white/80"
   485→                            >
   486→                              <RefreshCw className="h-3 w-3" /> Regenerate
   487→                            </button>
   488→                          </div>
   489→                        ) : null}
   490→                      </>
   491→                    ) : researching ? (
   492→                      <span className="inline-flex items-center gap-2 text-xs text-white/50">
   493→                        <Globe className="h-3.5 w-3.5 animate-pulse text-[#d4af37]" /> Searching the web…
   494→                        {researchQueries.length > 0 ? (
   495→                          <span className="text-white/30">({researchQueries.slice(0, 2).join(' · ')})</span>
   496→                        ) : null}
   497→                      </span>
   498→                    ) : runningTool ? (
   499→                      <span className="inline-flex items-center gap-2 text-xs text-white/50">
   500→                        <Terminal className="h-3.5 w-3.5 animate-pulse text-[#d4af37]" />
   501→                        {runningTool === 'working' ? 'Running server-side tools…' : `Running ${runningTool}…`}
   502→                      </span>
   503→                    ) : (
   504→                      <span className="inline-flex items-center gap-2 text-xs text-white/40">
   505→                        <Spinner /> RIISHE is thinking…
   506→                      </span>
   507→                    )}
   508→                  </div>
   509→                </div>
   510→              ))}
   511→              <div ref={endRef} />
   512→            </div>
   513→          )}
   514→        </div>
   515→
   516→        {error ? (
   517→          <div className="mx-5 mb-2 rounded-md border border-red-500/30 bg-red-500/10 px-3 py-2 text-xs text-red-300">{error}</div>
   518→        ) : null}

   521→        {documents.length > 0 ? (

   529→                <button
   530→                  onClick={() => setDocuments((prev) => prev.filter((x) => x.id !== d.id))}

   534→                  <X className="h-3 w-3" />

   536→              </span>
   537→            ))}
   538→          </div>
   539→        ) : null}
   540→
   541→        <div className="border-t border-[rgba(212,175,55,0.1)] p-4">
   542→          <div className="mx-auto max-w-3xl">
   543→            {/* Controls row: document attach (retrieval and code execution are
   544→                server-side policies — no user-facing toggles) */}
   545→            <div className="mb-2 flex flex-wrap items-center gap-2">
   546→              <button

   556→                ref={fileInputRef}
   557→                type="file"
   558→                accept=".txt,.md,.csv,.json,.log,.pdf"
   559→                className="hidden"
   560→                onChange={(e) => {
   561→                  const f = e.target.files?.[0]
   562→                  if (f) void uploadDocument(f)
   563→                }}
   564→              />
   565→            </div>
   566→
   567→            <div className="flex items-end gap-2">
   568→              <Textarea
   569→                value={input}
   570→                onChange={(e) => setInput(e.target.value)}
   571→                onKeyDown={(e) => {
   572→                  if (e.key === 'Enter' && !e.shiftKey) { e.preventDefault(); void send() }
   573→                }}
   574→                placeholder="Message RIISHE… (Enter to send, Shift+Enter for a new line)"
   575→                rows={Math.min(4, Math.max(1, input.split('\n').length))}
   576→                maxLength={12000}
   577→                disabled={streaming}
   578→                className="max-h-40 min-h-[46px] resize-none bg-black/30"
   579→                aria-label="Message RIISHE"
   580→              />
   581→              {streaming ? (
   582→                <Button onClick={stopGeneration} className="h-[46px] bg-[rgba(212,175,55,0.15)] px-4 text-[#f0c75e] hover:bg-[rgba(212,175,55,0.25)]" variant="outline" aria-label="Stop generating">
   583→                  <Square className="h-4 w-4" />
   584→                </Button>
   585→              ) : (
   586→                <Button onClick={() => void send()} disabled={!input.trim()} className="h-[46px] bg-[#d4af37] px-4 text-black hover:bg-[#e6c25a]" aria-label="Send message">
   587→                  <Send className="h-4 w-4" />
   588→                </Button>
   589→              )}
   590→            </div>
   591→          </div>
   592→        </div>
   593→      </div>
   594→    </div>
   595→  )
   596→}
