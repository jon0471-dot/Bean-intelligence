    91→    91→function useCopy(): [boolean, (text: string) => void] {
    92→    92→  const [copied, setCopied] = useState(false)
    93→    93→  const copy = useCallback((text: string) => {
    94→    94→    const done = () => {
    95→    95→      setCopied(true)
    96→    96→      setTimeout(() => setCopied(false), 1600)
    97→    97→    }
    98→    98→    if (navigator.clipboard?.writeText) {
    99→    99→      navigator.clipboard.writeText(text).then(done).catch(done)
   100→   100→    } else {
   101→   101→      const ta = document.createElement('textarea')
   102→   102→      ta.value = text
   103→   103→      document.body.appendChild(ta)
   104→   104→      ta.select()
   105→   105→      document.execCommand('copy')
   106→   106→      document.body.removeChild(ta)
   107→   107→      done()
   108→   108→    }
   109→   109→  }, [])
   110→   110→  return [copied, copy]
   111→   111→}

   249→   249→  async function send(overrideMessage?: string, retry = false) {
   250→   250→    const message = (overrideMessage ?? input).trim()
   251→   251→    if (!message || streaming) return
   252→   252→    if (overrideMessage === undefined) setInput('')
   253→   253→    setError(null)
   254→   254→    setStreaming(true)
   255→   255→    setResearching(false)
   256→   256→    setRunningTool(null)
   257→   257→    setResearchQueries([])
   258→   258→    streamedRef.current = ''
   259→   259→
   260→   260→    const userMsg: Msg = { id: `u-${Date.now()}`, role: 'user', content: message }
   261→   261→    const asstId = `a-${Date.now()}`
   262→   262→    setMessages((prev) => [...prev, userMsg, { id: asstId, role: 'assistant', content: '', pending: true }])
   263→   263→
   264→   264→    const controller = new AbortController()
   265→   265→    abortRef.current = controller
   266→   266→
   267→   267→    try {
   268→   268→      const res = await fetch(`/api/riishe/chat?stream=1${activeId ? `&conversationId=${activeId}` : ''}`, {
   269→   269→        method: 'POST',
   270→   270→        headers: { 'content-type': 'application/json' },
   271→   271→        body: JSON.stringify({
   272→   272→          message,
   273→   273→          conversationId: activeId ?? undefined,
   274→   274→          documentIds: documents.length > 0 ? documents.map((d) => d.id) : undefined,
   275→   275→          retry,
   276→   276→        }),
   277→   277→        signal: controller.signal,
   278→   278→      })
   279→   279→      if (!res.ok || !res.body) {
   280→   280→        // Keep the global identity in sync: a 401 here means the session is
   281→   281→        // gone server-side; drop the signed-in shell immediately.
   282→   282→        if (res.status === 401) window.dispatchEvent(new Event('bean:unauthorized'))
   283→   283→        const data = await res.json().catch(() => ({}))
   284→   284→        throw new Error(data?.error?.message ?? `Request failed (${res.status})`)
   285→   285→      }
   286→   286→      const reader = res.body.getReader()
   287→   287→      const decoder = new TextDecoder()
   288→   288→      let buffer = ''
   289→   289→      let convId: string | null = activeId
   290→   290→      let failed = false
   291→   291→
   292→   292→      while (true) {
   293→   293→        const { done, value } = await reader.read()
   294→   294→        if (done) break
   295→   295→        buffer += decoder.decode(value, { stream: true })
   296→   296→        const lines = buffer.split('\n')
   297→   297→        buffer = lines.pop() ?? ''
   298→   298→        for (const line of lines) {
   299→   299→          if (!line.startsWith('data: ')) continue
   300→   300→          let evt: {
   301→   301→            delta?: string
   302→   302→            done?: boolean
   303→   303→            error?: string
   304→   304→            conversationId?: string
   305→   305→            partial?: boolean
   306→   306→            stage?: 'researching' | 'tools' | 'solving' | 'verifying'
   307→   307→            research?: { queries?: string[] }
   308→   308→            tool?: { name: string; status: 'started' | 'done' | 'failed' }
   309→   309→            meta?: MsgMeta
   310→   310→          }
   311→   311→          try { evt = JSON.parse(line.slice(6)) } catch { continue }
   312→   312→          if (evt.conversationId) convId = evt.conversationId
   313→   313→          if (evt.stage === 'researching') { setResearching(true); setResearchQueries([]) }
   314→   314→          if (evt.stage === 'tools') { setResearching(false); setRunningTool('working') }
   315→   315→          if (evt.stage === 'solving' || evt.stage === 'verifying') { setResearching(false); setRunningTool(null) }
   316→   316→          if (evt.tool) {
   317→   317→            if (evt.tool.status === 'started') setRunningTool(evt.tool.name)
   318→   318→            else setRunningTool(null)
   319→   319→          }
   320→   320→          if (evt.research?.queries) setResearchQueries(evt.research.queries)
   321→   321→          if (evt.error) { failed = true; setError(evt.error) }
   322→   322→          if (evt.delta) {
   323→   323→            streamedRef.current += evt.delta
   324→   324→            setMessages((prev) =>
   325→   325→              prev.map((m) => (m.id === asstId ? { ...m, content: streamedRef.current, pending: false } : m)),
   326→   326→            )
   327→   327→          }
   328→   328→          if (evt.done) {
   329→   329→            setResearching(false)
   330→   330→            setRunningTool(null)
   331→   331→            if (convId && !activeId) setActiveId(convId)
   332→   332→            if (evt.meta) {
   333→   333→              setMessages((prev) => prev.map((m) => (m.id === asstId ? { ...m, meta: evt.meta } : m)))
   334→   334→            }
   335→   335→          }
   336→   336→        }
   337→   337→      }
   338→   338→      // If nothing streamed (failed, or the server reported an empty partial),
   339→   339→      // drop the empty assistant bubble — the error banner already explains
   340→   340→      // what happened, and a perpetual "thinking" indicator would be a lie.
   341→   341→      if (streamedRef.current === '') {
   342→   342→        setMessages((prev) => prev.filter((m) => m.id !== asstId))
   343→   343→      } else if (failed) {
   344→   344→        setMessages((prev) => prev.map((m) => (m.id === asstId ? { ...m, partial: true, pending: false } : m)))
   345→   345→      }
   346→   346→      void loadConversations()
   347→   347→    } catch (e) {
   348→   348→      if ((e as Error).name === 'AbortError') {
   349→   349→        // User pressed stop: keep what arrived and label it honestly.
   350→   350→        if (streamedRef.current !== '') {
   351→   351→          setMessages((prev) => prev.map((m) => (m.id === asstId ? { ...m, partial: true, pending: false } : m)))
   352→   352→        } else {
   353→   353→          setMessages((prev) => prev.filter((m) => m.id !== asstId))
   354→   354→        }
   355→   355→      } else {
   356→   356→        setError((e as Error).message)
   357→   357→        setMessages((prev) => prev.filter((m) => m.id !== asstId || m.content.length > 0))
   358→   358→      }
   359→   359→    } finally {
   360→   360→      setStreaming(false)
   361→   361→      setResearching(false)
   362→   362→      setRunningTool(null)
   363→   363→      abortRef.current = null
   364→   364→      setMessages((prev) => prev.map((m) => ({ ...m, pending: false })))
   365→   365→    }
   366→   366→  }
