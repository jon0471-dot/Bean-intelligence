    96→function useCopy(): [boolean, (text: string) => void] {
    97→  const [copied, setCopied] = useState(false)
    98→  const copy = useCallback((text: string) => {
    99→    const done = () => {
   100→      setCopied(true)
   101→      setTimeout(() => setCopied(false), 1600)
   102→    }
   103→    if (navigator.clipboard?.writeText) {
   104→      navigator.clipboard.writeText(text).then(done).catch(done)
   105→    } else {
   106→      const ta = document.createElement('textarea')
   107→      ta.value = text
   108→      document.body.appendChild(ta)
   109→      ta.select()
   110→      document.execCommand('copy')
   111→      document.body.removeChild(ta)
   112→      done()
   113→    }
   114→  }, [])
   115→  return [copied, copy]
   116→}

   146→function formatBytes(n: number): string {
   147→  if (n < 1024) return `${n} B`
   148→  if (n < 1024 * 1024) return `${(n / 1024).toFixed(1)} KB`
   149→  return `${(n / (1024 * 1024)).toFixed(1)} MB`

   223→  async function uploadOne(file: File) {
   224→    const form = new FormData()
   225→    form.append('file', file)
   226→    const res = await fetch('/api/riishe/documents', { method: 'POST', body: form })

   233→      // Unsafe or rejected outright — surface the server's actionable reason.
   234→      throw new Error(data?.error?.message ?? 'This file cannot be processed.')
   235→    }

   237→    if (data.document?.status === 'unsupported') {
   238→      setError(data.document.note ?? 'This file type cannot be analyzed yet — it stays attached but adds nothing to the answer.')

   242→  async function uploadFiles(list: FileList | File[]) {
   243→    const incoming = Array.from(list)
   244→    if (incoming.length === 0 || uploadingCount > 0) return
   245→    const room = MAX_FILES - files.length
   246→    if (room <= 0) {
   247→      setError(`Up to ${MAX_FILES} attachments per message.`)
   248→      return
   249→    }
   250→    setError(null)
   251→    setUploadingCount(incoming.length)
   252→    const batch = incoming.slice(0, room)
   253→    let failures = 0
   254→    for (const f of batch) {
   255→      try {
   256→        await uploadOne(f)
   257→      } catch (e) {
   258→        failures += 1
   259→        setError((e as Error).message)
   260→      } finally {
   261→        setUploadingCount((c) => Math.max(0, c - 1))
   262→      }
   263→    }
   264→    if (failures > 0 && batch.length > failures) {
   265→      setError('Some files were rejected — the message explains each failure.')
   266→    }
   267→  }

   273→  /** Regenerate the response to a SPECIFIC user turn (consistent actions on
   274→   *  every assistant message). For the trailing exchange this is the classic
   275→   *  retry; for earlier messages the server removes that exchange and
   276→   *  everything after it, then answers the same user turn again. */
   277→  function regenerateFrom(asstMsg: Msg) {
   278→    if (streaming) return
   279→    const idx = messages.findIndex((m) => m.id === asstMsg.id)
   280→    if (idx < 0) return
   281→    let userIdx = -1
   282→    for (let i = idx - 1; i >= 0; i -= 1) {
   283→      if (messages[i].role === 'user') { userIdx = i; break }
   284→    }
   285→    if (userIdx < 0) return
   286→    const userContent = messages[userIdx].content
   287→    const isTrailing = idx === messages.length - 1
   288→    // Without a persisted id a mid-conversation regenerate cannot truncate
   289→    // server-side history — refuse rather than silently duplicating the turn.
   290→    if (!isTrailing && !asstMsg.serverId) return
   291→    // Drop everything from the user turn onward (locally; the server mirrors
   292→    // this for earlier messages so history stays an honest sequence).
   293→    setMessages(messages.slice(0, userIdx))
   294→    setTimeout(() => void send(userContent, { retry: isTrailing && !asstMsg.serverId, retryFromId: isTrailing ? undefined : asstMsg.serverId }), 0)
   295→  }

   297→  async function send(overrideMessage?: string, opts?: { retry?: boolean; retryFromId?: string }) {
   298→    const message = (overrideMessage ?? input).trim()
   299→    if (!message || streaming) return
   300→    if (overrideMessage === undefined) setInput('')
   301→    setError(null)
   302→    setStreaming(true)
   303→    setResearching(false)
   304→    setRunningTool(null)
   305→    streamedRef.current = ''
   306→
   307→    const userMsg: Msg = { id: `u-${Date.now()}`, role: 'user', content: message }
   308→    const asstId = `a-${Date.now()}`
   309→    setMessages((prev) => [...prev, userMsg, { id: asstId, role: 'assistant', content: '', pending: true }])
   310→
   311→    const controller = new AbortController()
   312→    abortRef.current = controller
   313→    const attachable = files.filter((f) => f.status === 'ready')
   314→
   315→    try {
   316→      const res = await fetch(`/api/riishe/chat?stream=1${activeId ? `&conversationId=${activeId}` : ''}`, {
   317→        method: 'POST',
   318→        headers: { 'content-type': 'application/json' },
   319→        body: JSON.stringify({
   320→          message,
   321→          conversationId: activeId ?? undefined,
   322→          documentIds: attachable.length > 0 ? attachable.map((d) => d.id) : undefined,
   323→          retry: opts?.retry ?? false,
   324→          ...(opts?.retryFromId ? { retryFromMessageId: opts.retryFromId } : {}),
   325→        }),
   326→        signal: controller.signal,
   327→      })
   328→      if (!res.ok || !res.body) {
   329→        // Keep the global identity in sync: a 401 here means the session is
   330→        // gone server-side; drop the signed-in shell immediately.
   331→        if (res.status === 401) window.dispatchEvent(new Event('bean:unauthorized'))
   332→        const data = await res.json().catch(() => ({}))
   333→        throw new Error(data?.error?.message ?? `Request failed (${res.status})`)
   334→      }
   335→      const reader = res.body.getReader()
   336→      const decoder = new TextDecoder()
   337→      let buffer = ''
   338→      let convId: string | null = activeId
   339→      let failed = false
   340→
   341→      while (true) {
   342→        const { done, value } = await reader.read()
   343→        if (done) break
   344→        buffer += decoder.decode(value, { stream: true })
   345→        const lines = buffer.split('\n')
   346→        buffer = lines.pop() ?? ''
   347→        for (const line of lines) {
   348→          if (!line.startsWith('data: ')) continue
   349→          let evt: {
   350→            delta?: string
   351→            reset?: boolean
   352→            done?: boolean
   353→            error?: string
   354→            conversationId?: string
   355→            messageId?: string
   356→            partial?: boolean
   357→            stage?: 'researching' | 'tools' | 'solving' | 'verifying'
   358→            tool?: { name: string; status: 'started' | 'done' | 'failed' }
   359→            meta?: MsgMeta
   360→          }
   361→          try { evt = JSON.parse(line.slice(6)) } catch { continue }
   362→          if (evt.conversationId) convId = evt.conversationId
   363→          if (evt.stage === 'researching') setResearching(true)
   364→          if (evt.stage === 'tools') { setResearching(false); setRunningTool('working') }
   365→          if (evt.stage === 'solving' || evt.stage === 'verifying') { setResearching(false); setRunningTool(null) }
   366→          if (evt.tool) {
   367→            if (evt.tool.status === 'started') setRunningTool(evt.tool.name)
   368→            else setRunningTool(null)
   369→          }
   370→          if (evt.reset) {
   371→            // Reset protocol: a verification rewrite replaced the streamed
   372→            // draft — clear it so the final answer REPLACES, never appends.
   373→            streamedRef.current = ''
   374→            setMessages((prev) =>
   375→              prev.map((m) => (m.id === asstId ? { ...m, content: '', pending: true } : m)),
   376→            )
   377→          }
   378→          if (evt.error) { failed = true; setError(evt.error) }
   379→          if (evt.delta) {
   380→            streamedRef.current += evt.delta
   381→            setMessages((prev) =>
   382→              prev.map((m) => (m.id === asstId ? { ...m, content: streamedRef.current, pending: false } : m)),
   383→            )
   384→          }
   385→          if (evt.done) {
   386→            setResearching(false)
   387→            setRunningTool(null)
   388→            if (convId && !activeId) setActiveId(convId)
   389→            if (evt.messageId) {
   390→              setMessages((prev) => prev.map((m) => (m.id === asstId ? { ...m, serverId: evt.messageId } : m)))
   391→            }
   392→            if (evt.meta) {
   393→              setMessages((prev) => prev.map((m) => (m.id === asstId ? { ...m, meta: evt.meta } : m)))
   394→            }
   395→          }
   396→        }
   397→      }
   398→      // If nothing streamed (failed, or the server reported an empty partial),
   399→      // drop the empty assistant bubble — the error banner already explains
   400→      // what happened, and a perpetual "thinking" indicator would be a lie.
   401→      if (streamedRef.current === '') {
   402→        setMessages((prev) => prev.filter((m) => m.id !== asstId))
   403→      } else if (failed) {
   404→        setMessages((prev) => prev.map((m) => (m.id === asstId ? { ...m, partial: true, pending: false } : m)))
   405→      }
   406→      void loadConversations()
   407→    } catch (e) {
   408→      if ((e as Error).name === 'AbortError') {
   409→        // User pressed stop: keep what arrived and label it honestly.
   410→        if (streamedRef.current !== '') {
   411→          setMessages((prev) => prev.map((m) => (m.id === asstId ? { ...m, partial: true, pending: false } : m)))
   412→        } else {
   413→          setMessages((prev) => prev.filter((m) => m.id !== asstId))
   414→        }
   415→      } else {
   416→        setError((e as Error).message)
   417→        setMessages((prev) => prev.filter((m) => m.id !== asstId || m.content.length > 0))
   418→      }
   419→    } finally {
   420→      setStreaming(false)
   421→      setResearching(false)
   422→      setRunningTool(null)
   423→      abortRef.current = null
   424→      setMessages((prev) => prev.map((m) => ({ ...m, pending: false })))
   425→    }
   426→  }

   688→/**
   689→ * One assistant response: clean prose (citation markers render as source-name
   690→ * links; unresolved ones are stripped), the honest verification note when one
   691→ * exists, and CONSISTENT actions — Copy response, Regenerate, and View
   692→ * sources (only when sources exist) — on EVERY response, not just the last.
   693→ */
   694→function AssistantBubble({
   695→  msg,
   696→  onRegenerate,
   697→  streaming,
   698→  renderers,
   699→  researching,
   700→  runningTool,
   701→}: {
   702→  msg: Msg
   703→  onRegenerate: () => void
   704→  streaming: boolean
   705→  renderers: Parameters<typeof Markdown>[0]['components']
   706→  researching: boolean
   707→  runningTool: string | null
   708→}) {
   709→  const [sourcesOpen, setSourcesOpen] = useState(false)
   710→  const hasSources = (msg.meta?.sources?.length ?? 0) > 0
   711→  if (!msg.content) {
   712→    // Empty draft: the live status line (search / tools / thinking).
   713→    return researching ? (
   714→      <span className="inline-flex items-center gap-2 text-xs text-white/50">
   715→        <Globe className="h-3.5 w-3.5 animate-pulse text-[#d4af37]" /> Searching the web…
   716→      </span>
   717→    ) : runningTool ? (
   718→      <span className="inline-flex items-center gap-2 text-xs text-white/50">
   719→        <Terminal className="h-3.5 w-3.5 animate-pulse text-[#d4af37]" />
   720→        {runningTool === 'working' ? 'Working…' : 'Running tools…'}
   721→      </span>
   722→    ) : (
   723→      <span className="inline-flex items-center gap-2 text-xs text-white/40">
   724→        <Spinner /> Thinking…
   725→      </span>
   726→    )
   727→  }
   728→  const claims = mapSourceClaims(msg.content, msg.meta?.sources ?? [])
   729→  return (
   730→    <>
   731→      <Markdown
   732→        content={linkifyCitations(msg.content, msg.meta?.sources)}
   733→        components={renderers}
   734→      />
   735→      {msg.pending ? <Spinner className="ml-1 mt-1" /> : null}
   736→      {msg.partial ? (
   737→        <div className="mt-2 inline-flex items-center gap-1.5 rounded border border-amber-500/30 bg-amber-500/10 px-2 py-0.5 text-[10px] text-amber-300">
   738→          <AlertTriangle className="h-3 w-3" /> Incomplete — the stream was stopped or ended early
   739→        </div>
   740→      ) : null}
   741→      <AnswerMeta meta={msg.meta} forceOpen={sourcesOpen} claims={claims} />
   742→      {!msg.pending ? (
   743→        <div className="mt-2 flex flex-wrap items-center gap-2 border-t border-[rgba(212,175,55,0.12)] pt-2">
   744→          <CopyMessageButton content={msg.content} />
   745→          <button
   746→            onClick={onRegenerate}
   747→            disabled={streaming}
   748→            className="inline-flex items-center gap-1 rounded px-1.5 py-0.5 text-[10px] text-white/45 transition-colors hover:bg-white/10 hover:text-white/80 disabled:opacity-40"
   749→            aria-label="Regenerate response"
   750→          >
   751→            <RefreshCw className="h-3 w-3" /> Regenerate
   752→          </button>
   753→          {hasSources ? (
   754→            <button
   755→              onClick={() => setSourcesOpen(true)}
   756→              className="inline-flex items-center gap-1 rounded px-1.5 py-0.5 text-[10px] text-white/45 transition-colors hover:bg-white/10 hover:text-white/80"
   757→              aria-label="View sources"
   758→              aria-expanded={sourcesOpen}
   759→            >
   760→              <ExternalLink className="h-3 w-3" /> View sources
   761→            </button>
   762→          ) : null}
   763→        </div>
   764→      ) : null}
   765→    </>
   766→  )
   767→}

   783→/**
   784→ * Answer-first evidence: NO state badges, NO verification summary, NO tool
   785→ * chips, NO "last checked" telemetry in the reading flow. What remains:
   786→ *  - a short honest verification note ONLY when something material could not
   787→ *    be confirmed (uncovered claims, stale sources, wrong-case retrieval,
   788→ *    failed live verification);
   789→ *  - a subtle collapsed "Sources" section listing what was really retrieved,
   790→ *    each entry with its publisher, date, and the claim it supports.
   791→ */
   792→function AnswerMeta({
   793→  meta,
   794→  forceOpen,
   795→  claims,
   796→}: {
   797→  meta?: MsgMeta
   798→  forceOpen?: boolean
   799→  claims: Map<number, string>
   800→}) {
   801→  const detailsRef = useRef<HTMLDetailsElement>(null)
   802→  useEffect(() => {
   803→    if (forceOpen && detailsRef.current) detailsRef.current.open = true
   804→  }, [forceOpen])
   805→  const sources = meta?.sources ?? []
   806→  const note = meta?.verificationNote
   807→  const researchFailed = meta?.states?.includes('TOOL_UNAVAILABLE') ?? false
   808→  if (!researchFailed && !note && sources.length === 0) return null
   809→  return (
   810→    <div className="mt-2 border-t border-[rgba(212,175,55,0.12)] pt-2">
   811→      {researchFailed ? (
   812→        <div className="text-[11px] text-amber-300/90">
   813→          Live sources could not be checked for this answer — treat current details as unverified.
   814→        </div>
   815→      ) : note ? (
   816→        <div className="text-[11px] leading-snug text-white/40">{note}</div>
   817→      ) : null}
   818→
   819→      {sources.length > 0 ? (
   820→        <details ref={detailsRef} className="group mt-1">
   821→          <summary className="flex w-fit cursor-pointer list-none items-center gap-1 text-[11px] text-white/40 hover:text-white/65">
   822→            <ChevronDown className="h-3 w-3 transition-transform group-open:rotate-180" />
   823→            Sources ({sources.length})
   824→          </summary>
   825→          <ul className="mt-1.5 space-y-2 pl-4">
   826→            {sources.map((s) => (
   827→              <li key={s.id} id={`bean-src-${s.id}`} className="scroll-mt-6 text-[11px] leading-snug">
   828→                <a
   829→                  href={s.url}
   830→                  target="_blank"
   831→                  rel="noopener noreferrer"
   832→                  className="text-[#f0c75e] hover:underline"
   833→                >
   834→                  {s.title.slice(0, 90)}
   835→                  <ExternalLink className="ml-1 inline h-2.5 w-2.5" />
   836→                </a>
   837→                <span className="text-white/35"> — {s.hostName.replace(/^www\./, '')}</span>
   838→                {s.date ? <span className="ml-1 text-white/30">· {s.date}</span> : null}
   839→                <SourceClaimLine content={claims.get(s.id)} />
   840→              </li>
   841→            ))}
   842→          </ul>
   843→        </details>
   844→      ) : null}
   845→    </div>
   846→  )
   847→}
