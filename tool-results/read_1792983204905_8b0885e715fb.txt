   118→   118→function useCopy(): [boolean, (text: string) => void] {
   119→   119→  const [copied, setCopied] = useState(false)
   120→   120→  const copy = useCallback((text: string) => {
   121→   121→    const done = () => {
   122→   122→      setCopied(true)
   123→   123→      setTimeout(() => setCopied(false), 1600)
   124→   124→    }
   125→   125→    if (navigator.clipboard?.writeText) {
   126→   126→      navigator.clipboard.writeText(text).then(done).catch(done)
   127→   127→    } else {
   128→   128→      const ta = document.createElement('textarea')
   129→   129→      ta.value = text
   130→   130→      document.body.appendChild(ta)
   131→   131→      ta.select()
   132→   132→      document.execCommand('copy')
   133→   133→      document.body.removeChild(ta)
   134→   134→      done()
   135→   135→    }
   136→   136→  }, [])
   137→   137→  return [copied, copy]
   138→   138→}

   484→   484→  /**
   485→   485→   * Shared SSE reader for a single assistant turn (send + regenerate).
   486→   486→   * Streams into the pending bubble `asstId` and finalizes honestly.
   487→   487→   */
   488→   488→  async function streamChat(opts: {
   489→   489→    message: string
   490→   490→    asstId: string
   491→   491→    searchMode: SearchMode
   492→   492→    retry?: boolean
   493→   493→    regenerateMessageId?: string
   494→   494→    /** Local id of the just-added user bubble; replaced by the real server id. */
   495→   495→    localUserMsgId?: string
   496→   496→  }) {
   497→   497→    const convAtStart = activeId
   498→   498→    const asstId = opts.asstId
   499→   499→    const controller = new AbortController()
   500→   500→    abortRef.current = controller
   501→   501→
   502→   502→    try {
   503→   503→      const res = await fetch(`/api/riishe/chat?stream=1${convAtStart ? `&conversationId=${convAtStart}` : ''}`, {
   504→   504→        method: 'POST',
   505→   505→        headers: { 'content-type': 'application/json' },
   506→   506→        body: JSON.stringify({
   507→   507→          message: opts.message,
   508→   508→          conversationId: convAtStart ?? undefined,
   509→   509→          searchMode: opts.searchMode,
   510→   510→          allowCodeExecution,
   511→   511→          documentIds: documents.length > 0 ? documents.map((d) => d.id) : undefined,
   512→   512→          retry: opts.retry ?? false,
   513→   513→          regenerateMessageId: opts.regenerateMessageId,
   514→   514→        }),
   515→   515→        signal: controller.signal,
   516→   516→      })
   517→   517→      if (!res.ok || !res.body) {
   518→   518→        // Keep the global identity in sync: a 401 here means the session is
   519→   519→        // gone server-side; drop the signed-in shell immediately.
   520→   520→        if (res.status === 401) window.dispatchEvent(new Event('bean:unauthorized'))
   521→   521→        const data = await res.json().catch(() => ({}))
   522→   522→        throw new Error(data?.error?.message ?? `Request failed (${res.status})`)
   523→   523→      }
   524→   524→      const reader = res.body.getReader()
   525→   525→      const decoder = new TextDecoder()
   526→   526→      let buffer = ''
   527→   527→      let convId: string | null = convAtStart
   528→   528→      let failed = false
   529→   529→
   530→   530→      while (true) {
   531→   531→        const { done, value } = await reader.read()
   532→   532→        if (done) break
   533→   533→        buffer += decoder.decode(value, { stream: true })
   534→   534→        const lines = buffer.split('\n')
   535→   535→        buffer = lines.pop() ?? ''
   536→   536→        for (const line of lines) {
   537→   537→          if (!line.startsWith('data: ')) continue
   538→   538→          let evt: {
   539→   539→            delta?: string
   540→   540→            done?: boolean
   541→   541→            error?: string
   542→   542→            conversationId?: string
   543→   543→            partial?: boolean
   544→   544→            userMessageId?: string
   545→   545→            messageId?: string | null
   546→   546→            stage?: 'researching' | 'tools' | 'solving' | 'verifying'
   547→   547→            research?: { queries?: string[] }
   548→   548→            tool?: { name: string; status: 'started' | 'done' | 'failed' }
   549→   549→            meta?: MsgMeta
   550→   550→          }
   551→   551→          try { evt = JSON.parse(line.slice(6)) } catch { continue }
   552→   552→          if (evt.conversationId) convId = evt.conversationId
   553→   553→          if (evt.userMessageId && opts.localUserMsgId) {
   554→   554→            // Adopt the real persisted id for the user turn so per-message
   555→   555→            // actions address server rows.
   556→   556→            const serverId = evt.userMessageId
   557→   557→            const localId = opts.localUserMsgId
   558→   558→            setMessages((prev) => prev.map((m) => (m.id === localId ? { ...m, id: serverId } : m)))
   559→   559→          }
   560→   560→          if (evt.stage === 'researching') { setResearching(true); setResearchQueries([]) }
   561→   561→          if (evt.stage === 'tools') { setResearching(false); setRunningTool('working') }
   562→   562→          if (evt.stage === 'solving' || evt.stage === 'verifying') { setResearching(false); setRunningTool(null) }
   563→   563→          if (evt.tool) {
   564→   564→            if (evt.tool.status === 'started') setRunningTool(evt.tool.name)
   565→   565→            else setRunningTool(null)
   566→   566→          }
   567→   567→          if (evt.research?.queries) setResearchQueries(evt.research.queries)
   568→   568→          if (evt.error) { failed = true; setError(evt.error) }
   569→   569→          if (evt.delta) {
   570→   570→            streamedRef.current += evt.delta
   571→   571→            setMessages((prev) =>
   572→   572→              prev.map((m) => (m.id === asstId ? { ...m, content: streamedRef.current, pending: false } : m)),
   573→   573→            )
   574→   574→          }
   575→   575→          if (evt.done) {
   576→   576→            setResearching(false)
   577→   577→            setRunningTool(null)
   578→   578→            if (convId && !convAtStart) setActiveId(convId)
   579→   579→            const serverAsstId = evt.messageId ?? null
   580→   580→            setMessages((prev) =>
   581→   581→              prev.map((m) =>
   582→   582→                m.id === asstId ? { ...m, id: serverAsstId ?? m.id, meta: evt.meta ?? m.meta } : m,
   583→   583→              ),
   584→   584→            )
   585→   585→          }
   586→   586→        }
   587→   587→      }
   588→   588→      // If nothing streamed (failed, or the server reported an empty partial),
   589→   589→      // drop the empty assistant bubble — the error banner already explains
   590→   590→      // what happened, and a perpetual "thinking" indicator would be a lie.
   591→   591→      if (streamedRef.current === '') {
   592→   592→        setMessages((prev) => prev.filter((m) => m.id !== asstId))
   593→   593→      } else if (failed) {
   594→   594→        setMessages((prev) => prev.map((m) => (m.id === asstId ? { ...m, partial: true, pending: false } : m)))
   595→   595→      }
   596→   596→      void loadConversations()
   597→   597→    } catch (e) {
   598→   598→      if ((e as Error).name === 'AbortError') {
   599→   599→        // User pressed stop: keep what arrived and label it honestly.
   600→   600→        if (streamedRef.current !== '') {
   601→   601→          setMessages((prev) => prev.map((m) => (m.id === asstId ? { ...m, partial: true, pending: false } : m)))
   602→   602→        } else {
   603→   603→          setMessages((prev) => prev.filter((m) => m.id !== asstId))
   604→   604→        }
   605→   605→      } else {
   606→   606→        setError((e as Error).message)
   607→   607→        setMessages((prev) => prev.filter((m) => m.id !== asstId || m.content.length > 0))
   608→   608→      }
   609→   609→    } finally {
   610→   610→      setStreaming(false)
   611→   611→      setResearching(false)
   612→   612→      setRunningTool(null)
   613→   613→      abortRef.current = null
   614→   614→      setMessages((prev) => prev.map((m) => ({ ...m, pending: false })))
   615→   615→    }
   616→   616→  }

   618→   618→  async function send(overrideMessage?: string, retry = false) {
   619→   619→    const message = (overrideMessage ?? input).trim()
   620→   620→    if (!message || streaming) return
   621→   621→    if (overrideMessage === undefined) setInput('')

   624→   624→    setResearching(false)
   625→   625→    setRunningTool(null)
   626→   626→    setResearchQueries([])
   627→   627→    streamedRef.current = ''
   628→   628→
   629→   629→    const userMsg: Msg = { id: `u-${Date.now()}`, role: 'user', content: message }
   630→   630→    const asstId = `a-${Date.now()}`
   631→   631→    setMessages((prev) => [...prev, userMsg, { id: asstId, role: 'assistant', content: '', pending: true }])
   632→   632→    await streamChat({ message, asstId, searchMode, retry, localUserMsgId: userMsg.id })
   633→   633→  }

   635→   635→  /**
   636→   636→   * Regenerate ANY assistant response in place: the old answer (and every
   637→   637→   * message after it) is discarded locally and server-side; the original
   638→   638→   * question is re-answered with the mode recorded for THAT answer, so the
   639→   639→   * regeneration reflects the original context and retrieval choice. While
   640→   640→   * a stream is running the action is disabled — no duplicate requests.
   641→   641→   */
   642→   642→  async function regenerateFrom(messageId: string) {
   643→   643→    if (streaming) return

   651→   651→
   652→   652→    setError(null)

   661→   661→      { id: asstId, role: 'assistant' as const, content: '', pending: true },
   662→   662→    ])

   665→   665→

   678→   678→          {conversations.length === 0 ? (
   679→   679→            <p className="px-2 py-6 text-center text-xs text-white/30">No conversations yet</p>
   680→   680→          ) : (
   681→   681→            conversations.map((c) => (
   682→   682→              <div
   683→   683→                key={c.id}
   684→   684→                className={cn(
   685→   685→                  'group mb-0.5 flex cursor-pointer items-center justify-between rounded-lg px-3 py-2 text-[13px] transition-colors',
   686→   686→                  activeId === c.id ? 'bg-[rgba(212,175,55,0.1)] text-[#f0c75e]' : 'text-white/60 hover:bg-white/5',
   687→   687→                )}

   862→   862→/**
   863→   863→ * One assistant message: markdown with citation chips, evidence panel,
   864→   864→ * compact sources and the Copy/Regenerate/View-sources action row. The
   865→   865→ * action row renders for EVERY assistant response (short or long), in every
   866→   866→ * mode — never only for the latest one.
   867→   867→ */
   868→   868→function AssistantMessage({
   869→   869→  m,
   870→   870→  researching,
   871→   871→  runningTool,
   872→   872→  researchQueries,
   873→   873→  streaming,
   874→   874→  isLatest,
   875→   875→  onRegenerate,
   876→   876→}: {
   877→   877→  m: Msg
   878→   878→  researching: boolean
   879→   879→  runningTool: string | null
   880→   880→  researchQueries: string[]
   881→   881→  streaming: boolean
   882→   882→  isLatest: boolean
   883→   883→  onRegenerate: () => void
   884→   884→}) {
   885→   885→  const meta = m.meta
   886→   886→  const [previewId, setPreviewId] = useState<number | null>(null)
   887→   887→  const [sourcesOpen, setSourcesOpen] = useState(false)
   888→   888→  const [highlightId, setHighlightId] = useState<number | null>(null)
   889→   889→  const entryRefs = useRef(new Map<number, HTMLLIElement | null>())
   890→   890→
   891→   891→  const registerRef = useCallback((id: number, el: HTMLLIElement | null) => {
   892→   892→    entryRefs.current.set(id, el)
   893→   893→  }, [])
   894→   894→
   895→   895→  // Citation chip click: open the Sources section and spotlight the entry.
   896→   896→  const jumpToSource = useCallback((id: number) => {
   897→   897→    setSourcesOpen(true)
   898→   898→    setHighlightId(id)
   899→   899→    window.setTimeout(() => {
   900→   900→      entryRefs.current.get(id)?.scrollIntoView({ behavior: 'smooth', block: 'center' })
   901→   901→    }, 80)
   902→   902→    window.setTimeout(() => setHighlightId((cur) => (cur === id ? null : cur)), 2400)
   903→   903→  }, [])
   904→   904→
   905→   905→  const sources = meta?.sources ?? []
   906→   906→  const hasSources = sources.length > 0
   907→   907→
   908→   908→  const mdComponents = {
   909→   909→    a: ({ href, children }: { href?: string; children?: React.ReactNode }) => {
   910→   910→      if (href?.startsWith('#bean-src-')) {
   911→   911→        const id = Number(href.replace('#bean-src-', ''))
   912→   912→        const source = sources.find((s) => s.id === id)
   913→   913→        if (!source) return <span className="text-white/60">{children}</span>
   914→   914→        return (
   915→   915→          <CitationChip
   916→   916→            id={id}
   917→   917→            source={source}
   918→   918→            previewed={previewId === id}
   919→   919→            onPreview={setPreviewId}
   920→   920→            onLeave={() => setPreviewId(null)}
   921→   921→            onJump={jumpToSource}
   922→   922→          />
   923→   923→        )
   924→   924→      }
   925→   925→      return (
   926→   926→        <a href={href} target="_blank" rel="noopener noreferrer" className="text-[#f0c75e] underline decoration-[rgba(212,175,55,0.4)] hover:decoration-[#f0c75e]">
   927→   927→          {children}
   928→   928→        </a>
   929→   929→      )
   930→   930→    },
   931→   931→    pre: ({ children }: { children?: React.ReactNode }) => {
   932→   932→      // React-markdown nests <code> inside <pre>; extract for highlighting.
   933→   933→      const child = Array.isArray(children) ? children[0] : children
   934→   934→      const props = (child as { props?: { className?: string; children?: ReactNode } })?.props
   935→   935→      const lang = props?.className?.replace(/language-/, '') || undefined
   936→   936→      const code = String(props?.children ?? '').replace(/\n$/, '')
   937→   937→      if (code === '') return <pre>{children}</pre>
   938→   938→      return <CodeBlock code={code} lang={lang} />
   939→   939→    },
   940→   940→  }
   941→   941→
   942→   942→  return (
   943→   943→    <div className="flex justify-start">
   944→   944→      <div className="bean-glass max-w-[92%] rounded-2xl px-4 py-3 text-white/85 sm:max-w-[85%]">
   945→   945→        {m.content ? (
   946→   946→          <>
   947→   947→            <Markdown content={linkifyCitations(m.content, sources)} components={mdComponents} />
   948→   948→            {m.pending ? <Spinner className="ml-1 mt-1" /> : null}
   949→   949→            {m.partial ? (
   950→   950→              <div className="mt-2 inline-flex items-center gap-1.5 rounded border border-amber-500/30 bg-amber-500/10 px-2 py-0.5 text-[10px] text-amber-300">
   951→   951→                <AlertTriangle className="h-3 w-3" /> Incomplete — the stream was stopped or ended early
   952→   952→              </div>
   953→   953→            ) : null}
   954→   954→            {meta ? <AnswerEvidence meta={meta} /> : null}
   955→   955→            {hasSources && meta ? (
   956→   956→              <SourcesSection
   957→   957→                sources={sources}
   958→   958→                open={sourcesOpen}
   959→   959→                onOpenChange={setSourcesOpen}
   960→   960→                highlightId={highlightId}
   961→   961→                registerRef={registerRef}
   962→   962→              />
   963→   963→            ) : null}
   964→   964→            {!m.pending ? (
   965→   965→              <MessageActions
   966→   966→                copyText={stripCitationMarkers(m.content)}
   967→   967→                disabled={streaming}
   968→   968→                hasSources={hasSources}
   969→   969→                sourcesOpen={sourcesOpen}
   970→   970→                onToggleSources={() => setSourcesOpen((v) => !v)}
   971→   971→                onRegenerate={onRegenerate}
   972→   972→                modeLabel={meta?.searchMode}
   973→   973→              />
   974→   974→            ) : null}
   975→   975→          </>
   976→   976→        ) : researching && isLatest ? (
   977→   977→          <span className="inline-flex items-center gap-2 text-xs text-white/50">
   978→   978→            <Globe className="h-3.5 w-3.5 animate-pulse text-[#d4af37]" /> Searching the web…
   979→   979→            {researchQueries.length > 0 ? (
   980→   980→              <span className="text-white/30">({researchQueries.slice(0, 2).join(' · ')})</span>
   981→   981→            ) : null}
   982→   982→          </span>
   983→   983→        ) : runningTool && isLatest ? (
   984→   984→          <span className="inline-flex items-center gap-2 text-xs text-white/50">
   985→   985→            <Terminal className="h-3.5 w-3.5 animate-pulse text-[#d4af37]" />
   986→   986→            {runningTool === 'working' ? 'Running server-side tools…' : `Running ${runningTool}…`}
   987→   987→          </span>
   988→   988→        ) : (
   989→   989→          <span className="inline-flex items-center gap-2 text-xs text-white/40">
   990→   990→            <Spinner /> RIISHE is thinking…
   991→   991→          </span>
   992→   992→        )}
   993→   993→      </div>
   994→   994→    </div>
   995→   995→  )
   996→   996→}

   998→   998→/** Evidence panel: state badges, verification summary and honest degradation. */

  1007→  1007→
