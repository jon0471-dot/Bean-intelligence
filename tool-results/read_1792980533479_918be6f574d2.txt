    97→function useCopy(): [boolean, (text: string) => void] {
    98→  const [copied, setCopied] = useState(false)
    99→  const copy = useCallback((text: string) => {
   100→    const done = () => {
   101→      setCopied(true)
   102→      setTimeout(() => setCopied(false), 1600)
   103→    }
   104→    if (navigator.clipboard?.writeText) {
   105→      navigator.clipboard.writeText(text).then(done).catch(done)
   106→    } else {
   107→      const ta = document.createElement('textarea')
   108→      ta.value = text
   109→      document.body.appendChild(ta)
   110→      ta.select()
   111→      document.execCommand('copy')
   112→      document.body.removeChild(ta)
   113→      done()
   114→    }
   115→  }, [])
   116→  return [copied, copy]
   117→}

   155→  const [researchQueries, setResearchQueries] = useState<string[]>([])

   204→      void loadConversations()
   205→    } catch (e) {
   206→      setError((e as Error).message)

   210→  function newConversation() {

   213→    setError(null)
   214→    setDocuments([])

   217→  async function uploadDocument(file: File) {
   218→    if (uploadingDoc || documents.length >= 2) return
   219→    setUploadingDoc(true)
   220→    setError(null)
   221→    try {
   222→      const form = new FormData()
   223→      form.append('file', file)
   224→      const res = await fetch('/api/riishe/documents', { method: 'POST', body: form })
   225→      const data = await res.json().catch(() => ({}))
   226→      if (!res.ok) {
   227→        if (res.status === 401) window.dispatchEvent(new Event('bean:unauthorized'))
   228→        throw new Error(data?.error?.message ?? `Upload failed (${res.status})`)
   229→      }
   230→      setDocuments((prev) => [...prev, data.document].slice(0, 2))
   231→    } catch (e) {
   232→      setError((e as Error).message)
   233→    } finally {
   234→      setUploadingDoc(false)
   235→      if (fileInputRef.current) fileInputRef.current.value = ''
   236→    }
   237→  }

   239→  function stopGeneration() {
   240→    abortRef.current?.abort()

   243→  function regenerate() {
   244→    if (streaming) return
   245→    const lastUser = [...messages].reverse().find((m) => m.role === 'user')
   246→    if (!lastUser) return
   247→    setInput(lastUser.content)
   248→    setMessages((prev) => {
   249→      const copy = [...prev]
   250→      if (copy.length > 0 && copy[copy.length - 1].role === 'assistant') copy.pop()
   251→      return copy
   252→    })
   253→    // Send on the next tick with the restored input.
   254→    setTimeout(() => void send(lastUser.content, true), 0)
   255→  }

   257→  async function send(overrideMessage?: string, retry = false) {
   258→    const message = (overrideMessage ?? input).trim()
   259→    if (!message || streaming) return
   260→    if (overrideMessage === undefined) setInput('')
   261→    setError(null)
   262→    setStreaming(true)
   263→    setResearching(false)
   264→    setRunningTool(null)
   265→    setResearchQueries([])
   266→    streamedRef.current = ''
   267→
   268→    const userMsg: Msg = { id: `u-${Date.now()}`, role: 'user', content: message }
   269→    const asstId = `a-${Date.now()}`
   270→    setMessages((prev) => [...prev, userMsg, { id: asstId, role: 'assistant', content: '', pending: true }])
   271→
   272→    const controller = new AbortController()
   273→    abortRef.current = controller
   274→
   275→    try {
   276→      const res = await fetch(`/api/riishe/chat?stream=1${activeId ? `&conversationId=${activeId}` : ''}`, {
   277→        method: 'POST',
   278→        headers: { 'content-type': 'application/json' },
   279→        body: JSON.stringify({
   280→          message,
   281→          conversationId: activeId ?? undefined,
   282→          searchMode,
   283→          allowCodeExecution,
   284→          documentIds: documents.length > 0 ? documents.map((d) => d.id) : undefined,
   285→          retry,
   286→        }),
   287→        signal: controller.signal,
   288→      })
   289→      if (!res.ok || !res.body) {
   290→        // Keep the global identity in sync: a 401 here means the session is
   291→        // gone server-side; drop the signed-in shell immediately.
   292→        if (res.status === 401) window.dispatchEvent(new Event('bean:unauthorized'))
   293→        const data = await res.json().catch(() => ({}))
   294→        throw new Error(data?.error?.message ?? `Request failed (${res.status})`)
   295→      }
   296→      const reader = res.body.getReader()
   297→      const decoder = new TextDecoder()
   298→      let buffer = ''
   299→      let convId: string | null = activeId
   300→      let failed = false
   301→
   302→      while (true) {
   303→        const { done, value } = await reader.read()
   304→        if (done) break
   305→        buffer += decoder.decode(value, { stream: true })
   306→        const lines = buffer.split('\n')
   307→        buffer = lines.pop() ?? ''
   308→        for (const line of lines) {
   309→          if (!line.startsWith('data: ')) continue
   310→          let evt: {
   311→            delta?: string
   312→            done?: boolean
   313→            error?: string
   314→            conversationId?: string
   315→            partial?: boolean
   316→            stage?: 'researching' | 'tools' | 'solving' | 'verifying'
   317→            research?: { queries?: string[] }
   318→            tool?: { name: string; status: 'started' | 'done' | 'failed' }
   319→            meta?: MsgMeta
   320→          }
   321→          try { evt = JSON.parse(line.slice(6)) } catch { continue }
   322→          if (evt.conversationId) convId = evt.conversationId
   323→          if (evt.stage === 'researching') { setResearching(true); setResearchQueries([]) }
   324→          if (evt.stage === 'tools') { setResearching(false); setRunningTool('working') }
   325→          if (evt.stage === 'solving' || evt.stage === 'verifying') { setResearching(false); setRunningTool(null) }
   326→          if (evt.tool) {
   327→            if (evt.tool.status === 'started') setRunningTool(evt.tool.name)
   328→            else setRunningTool(null)
   329→          }
   330→          if (evt.research?.queries) setResearchQueries(evt.research.queries)
   331→          if (evt.error) { failed = true; setError(evt.error) }
   332→          if (evt.delta) {
   333→            streamedRef.current += evt.delta
   334→            setMessages((prev) =>
   335→              prev.map((m) => (m.id === asstId ? { ...m, content: streamedRef.current, pending: false } : m)),
   336→            )
   337→          }
   338→          if (evt.done) {
   339→            setResearching(false)
   340→            setRunningTool(null)
   341→            if (convId && !activeId) setActiveId(convId)
   342→            if (evt.meta) {
   343→              setMessages((prev) => prev.map((m) => (m.id === asstId ? { ...m, meta: evt.meta } : m)))
   344→            }
   345→          }
   346→        }
   347→      }
   348→      // If nothing streamed (failed, or the server reported an empty partial),
   349→      // drop the empty assistant bubble — the error banner already explains
   350→      // what happened, and a perpetual "thinking" indicator would be a lie.
   351→      if (streamedRef.current === '') {
   352→        setMessages((prev) => prev.filter((m) => m.id !== asstId))
   353→      } else if (failed) {
   354→        setMessages((prev) => prev.map((m) => (m.id === asstId ? { ...m, partial: true, pending: false } : m)))
   355→      }
   356→      void loadConversations()
   357→    } catch (e) {
   358→      if ((e as Error).name === 'AbortError') {
   359→        // User pressed stop: keep what arrived and label it honestly.
   360→        if (streamedRef.current !== '') {
   361→          setMessages((prev) => prev.map((m) => (m.id === asstId ? { ...m, partial: true, pending: false } : m)))
   362→        } else {
   363→          setMessages((prev) => prev.filter((m) => m.id !== asstId))
   364→        }
   365→      } else {
   366→        setError((e as Error).message)
   367→        setMessages((prev) => prev.filter((m) => m.id !== asstId || m.content.length > 0))
   368→      }
   369→    } finally {
   370→      setStreaming(false)
   371→      setResearching(false)
   372→      setRunningTool(null)
   373→      abortRef.current = null
   374→      setMessages((prev) => prev.map((m) => ({ ...m, pending: false })))
   375→    }
   376→  }

   378→  const citationComponents = (meta?: MsgMeta) => ({

   461→              icon={<MessageSquareText className="h-10 w-10" />}

   463→              hint="Ask questions, request derivations, code help or research on current topics. RIISHE searches live sources when a question needs them, runs real tools, and verifies its arithmetic before answering. Attach a document for grounded analysis."
   464→            />
   465→          ) : (
   466→            <div className="mx-auto max-w-3xl space-y-4">
   467→              {messages.map((m) => (
   468→                <div key={m.id} className={cn('flex', m.role === 'user' ? 'justify-end' : 'justify-start')}>
   469→                  <div
   470→                    className={cn(
   471→                      'max-w-[85%] rounded-2xl px-4 py-3',
   472→                      m.role === 'user'
   473→                        ? 'bg-[rgba(212,175,55,0.14)] text-white/90'
   474→                        : 'bean-glass text-white/85',
   475→                    )}
   476→                  >
   477→                    {m.role === 'user' ? (
   478→                      <p className="whitespace-pre-wrap text-[14.5px] leading-relaxed">{m.content}</p>
   479→                    ) : m.content ? (
   480→                      <>

   607→
   608→            <div className="flex items-end gap-2">
   609→              <Textarea
   610→                value={input}

   612→                onKeyDown={(e) => {

   614→                }}
   615→                placeholder="Message RIISHE… (Enter to send, Shift+Enter for a new line)"

   624→                  <Square className="h-4 w-4" />

   628→                  <Send className="h-4 w-4" />
   629→                </Button>

   631→            </div>
   632→          </div>
   633→        </div>
   634→      </div>
   635→    </div>
   636→  )
