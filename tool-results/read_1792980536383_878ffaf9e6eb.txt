    97→    97→function useCopy(): [boolean, (text: string) => void] {
    98→    98→  const [copied, setCopied] = useState(false)
    99→    99→  const copy = useCallback((text: string) => {
   100→   100→    const done = () => {
   101→   101→      setCopied(true)
   102→   102→      setTimeout(() => setCopied(false), 1600)
   103→   103→    }
   104→   104→    if (navigator.clipboard?.writeText) {
   105→   105→      navigator.clipboard.writeText(text).then(done).catch(done)
   106→   106→    } else {
   107→   107→      const ta = document.createElement('textarea')
   108→   108→      ta.value = text
   109→   109→      document.body.appendChild(ta)
   110→   110→      ta.select()
   111→   111→      document.execCommand('copy')
   112→   112→      document.body.removeChild(ta)
   113→   113→      done()
   114→   114→    }
   115→   115→  }, [])
   116→   116→  return [copied, copy]
   117→   117→}

   257→   257→  async function send(overrideMessage?: string, retry = false) {
   258→   258→    const message = (overrideMessage ?? input).trim()
   259→   259→    if (!message || streaming) return
   260→   260→    if (overrideMessage === undefined) setInput('')
   261→   261→    setError(null)
   262→   262→    setStreaming(true)
   263→   263→    setResearching(false)
   264→   264→    setRunningTool(null)
   265→   265→    setResearchQueries([])
   266→   266→    streamedRef.current = ''
   267→   267→
   268→   268→    const userMsg: Msg = { id: `u-${Date.now()}`, role: 'user', content: message }
   269→   269→    const asstId = `a-${Date.now()}`
   270→   270→    setMessages((prev) => [...prev, userMsg, { id: asstId, role: 'assistant', content: '', pending: true }])
   271→   271→
   272→   272→    const controller = new AbortController()
   273→   273→    abortRef.current = controller
   274→   274→
   275→   275→    try {
   276→   276→      const res = await fetch(`/api/riishe/chat?stream=1${activeId ? `&conversationId=${activeId}` : ''}`, {
   277→   277→        method: 'POST',
   278→   278→        headers: { 'content-type': 'application/json' },
   279→   279→        body: JSON.stringify({
   280→   280→          message,
   281→   281→          conversationId: activeId ?? undefined,
   282→   282→          searchMode,
   283→   283→          allowCodeExecution,
   284→   284→          documentIds: documents.length > 0 ? documents.map((d) => d.id) : undefined,
   285→   285→          retry,
   286→   286→        }),
   287→   287→        signal: controller.signal,
   288→   288→      })
   289→   289→      if (!res.ok || !res.body) {
   290→   290→        // Keep the global identity in sync: a 401 here means the session is
   291→   291→        // gone server-side; drop the signed-in shell immediately.
   292→   292→        if (res.status === 401) window.dispatchEvent(new Event('bean:unauthorized'))
   293→   293→        const data = await res.json().catch(() => ({}))
   294→   294→        throw new Error(data?.error?.message ?? `Request failed (${res.status})`)
   295→   295→      }
   296→   296→      const reader = res.body.getReader()
   297→   297→      const decoder = new TextDecoder()
   298→   298→      let buffer = ''
   299→   299→      let convId: string | null = activeId
   300→   300→      let failed = false
   301→   301→
   302→   302→      while (true) {
   303→   303→        const { done, value } = await reader.read()
   304→   304→        if (done) break
   305→   305→        buffer += decoder.decode(value, { stream: true })
   306→   306→        const lines = buffer.split('\n')
   307→   307→        buffer = lines.pop() ?? ''
   308→   308→        for (const line of lines) {
   309→   309→          if (!line.startsWith('data: ')) continue
   310→   310→          let evt: {
   311→   311→            delta?: string
   312→   312→            done?: boolean
   313→   313→            error?: string
   314→   314→            conversationId?: string
   315→   315→            partial?: boolean
   316→   316→            stage?: 'researching' | 'tools' | 'solving' | 'verifying'
   317→   317→            research?: { queries?: string[] }
   318→   318→            tool?: { name: string; status: 'started' | 'done' | 'failed' }
   319→   319→            meta?: MsgMeta
   320→   320→          }
   321→   321→          try { evt = JSON.parse(line.slice(6)) } catch { continue }
   322→   322→          if (evt.conversationId) convId = evt.conversationId
   323→   323→          if (evt.stage === 'researching') { setResearching(true); setResearchQueries([]) }
   324→   324→          if (evt.stage === 'tools') { setResearching(false); setRunningTool('working') }
   325→   325→          if (evt.stage === 'solving' || evt.stage === 'verifying') { setResearching(false); setRunningTool(null) }
   326→   326→          if (evt.tool) {
   327→   327→            if (evt.tool.status === 'started') setRunningTool(evt.tool.name)
   328→   328→            else setRunningTool(null)
   329→   329→          }
   330→   330→          if (evt.research?.queries) setResearchQueries(evt.research.queries)
   331→   331→          if (evt.error) { failed = true; setError(evt.error) }
   332→   332→          if (evt.delta) {
   333→   333→            streamedRef.current += evt.delta
   334→   334→            setMessages((prev) =>
   335→   335→              prev.map((m) => (m.id === asstId ? { ...m, content: streamedRef.current, pending: false } : m)),
   336→   336→            )
   337→   337→          }
   338→   338→          if (evt.done) {
   339→   339→            setResearching(false)
   340→   340→            setRunningTool(null)
   341→   341→            if (convId && !activeId) setActiveId(convId)
   342→   342→            if (evt.meta) {
   343→   343→              setMessages((prev) => prev.map((m) => (m.id === asstId ? { ...m, meta: evt.meta } : m)))
   344→   344→            }
   345→   345→          }
   346→   346→        }
   347→   347→      }
   348→   348→      // If nothing streamed (failed, or the server reported an empty partial),
   349→   349→      // drop the empty assistant bubble — the error banner already explains
   350→   350→      // what happened, and a perpetual "thinking" indicator would be a lie.
   351→   351→      if (streamedRef.current === '') {
   352→   352→        setMessages((prev) => prev.filter((m) => m.id !== asstId))
   353→   353→      } else if (failed) {
   354→   354→        setMessages((prev) => prev.map((m) => (m.id === asstId ? { ...m, partial: true, pending: false } : m)))
   355→   355→      }
   356→   356→      void loadConversations()
   357→   357→    } catch (e) {
   358→   358→      if ((e as Error).name === 'AbortError') {
   359→   359→        // User pressed stop: keep what arrived and label it honestly.
   360→   360→        if (streamedRef.current !== '') {
   361→   361→          setMessages((prev) => prev.map((m) => (m.id === asstId ? { ...m, partial: true, pending: false } : m)))
   362→   362→        } else {
   363→   363→          setMessages((prev) => prev.filter((m) => m.id !== asstId))
   364→   364→        }
   365→   365→      } else {
   366→   366→        setError((e as Error).message)
   367→   367→        setMessages((prev) => prev.filter((m) => m.id !== asstId || m.content.length > 0))
   368→   368→      }
   369→   369→    } finally {
   370→   370→      setStreaming(false)
   371→   371→      setResearching(false)
   372→   372→      setRunningTool(null)
   373→   373→      abortRef.current = null
   374→   374→      setMessages((prev) => prev.map((m) => ({ ...m, pending: false })))
   375→   375→    }
   376→   376→  }
