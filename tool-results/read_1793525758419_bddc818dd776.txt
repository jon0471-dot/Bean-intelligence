    95→function useCopy(): [boolean, (text: string) => void] {
    96→  const [copied, setCopied] = useState(false)
    97→  const copy = useCallback((text: string) => {
    98→    const done = () => {
    99→      setCopied(true)
   100→      setTimeout(() => setCopied(false), 1600)
   101→    }
   102→    if (navigator.clipboard?.writeText) {
   103→      navigator.clipboard.writeText(text).then(done).catch(done)
   104→    } else {
   105→      const ta = document.createElement('textarea')
   106→      ta.value = text
   107→      document.body.appendChild(ta)
   108→      ta.select()
   109→      document.execCommand('copy')
   110→      document.body.removeChild(ta)
   111→      done()
   112→    }
   113→  }, [])
   114→  return [copied, copy]
   115→}

   145→function formatBytes(n: number): string {
   146→  if (n < 1024) return `${n} B`
   147→  if (n < 1024 * 1024) return `${(n / 1024).toFixed(1)} KB`
   148→  return `${(n / (1024 * 1024)).toFixed(1)} MB`
   149→}

   240→  async function uploadOne(file: File) {
   241→    const form = new FormData()
   242→    form.append('file', file)
   243→    const res = await fetch('/api/riishe/documents', { method: 'POST', body: form })
   244→    const data = await res.json().catch(() => ({}))
   245→    if (res.status === 401) window.dispatchEvent(new Event('bean:unauthorized'))
   246→    if (!res.ok && res.status !== 415) {
   247→      throw new Error(data?.error?.message ?? `Upload failed (${res.status})`)
   248→    }
   249→    if (res.status === 415 && !data?.document) {
   250→      // Unsafe or rejected outright — surface the server's actionable reason.
   251→      throw new Error(data?.error?.message ?? 'This file cannot be processed.')
   252→    }
   253→    setFiles((prev) => [...prev, data.document as FileMeta].slice(0, MAX_FILES))
   254→    if (data.document?.status === 'unsupported') {
   255→      setError(data.document.note ?? 'This file type cannot be analyzed yet — it stays attached but adds nothing to the answer.')
   256→    }
   257→  }

   259→  async function uploadFiles(list: FileList | File[]) {
   260→    const incoming = Array.from(list)
   261→    if (incoming.length === 0 || uploadingCount > 0) return
   262→    const room = MAX_FILES - files.length
   263→    if (room <= 0) {
   264→      setError(`Up to ${MAX_FILES} attachments per message.`)
   265→      return
   266→    }
   267→    setError(null)
   268→    setUploadingCount(incoming.length)
   269→    const batch = incoming.slice(0, room)
   270→    let failures = 0
   271→    for (const f of batch) {
   272→      try {
   273→        await uploadOne(f)
   274→      } catch (e) {
   275→        failures += 1
   276→        setError((e as Error).message)
   277→      } finally {
   278→        setUploadingCount((c) => Math.max(0, c - 1))
   279→      }
   280→    }
   281→    if (failures > 0 && batch.length > failures) {
   282→      setError('Some files were rejected — the message explains each failure.')
   283→    }
   284→  }

   290→  /** Regenerate the response to a SPECIFIC user turn (consistent actions on
   291→   *  every assistant message). For the trailing exchange this is the classic
   292→   *  retry; for earlier messages the server removes that exchange and
   293→   *  everything after it, then answers the same user turn again. */
   294→  function regenerateFrom(asstMsg: Msg) {
   295→    if (streaming) return
   296→    const idx = messages.findIndex((m) => m.id === asstMsg.id)
   297→    if (idx < 0) return
   298→    let userIdx = -1
   299→    for (let i = idx - 1; i >= 0; i -= 1) {
   300→      if (messages[i].role === 'user') { userIdx = i; break }
   301→    }
   302→    if (userIdx < 0) return
   303→    const userContent = messages[userIdx].content
   304→    const isTrailing = idx === messages.length - 1
   305→    // Without a persisted id a mid-conversation regenerate cannot truncate
   306→    // server-side history — refuse rather than silently duplicating the turn.
   307→    if (!isTrailing && !asstMsg.serverId) return
   308→    // Drop everything from the user turn onward (locally; the server mirrors
   309→    // this for earlier messages so history stays an honest sequence).
   310→    setMessages(messages.slice(0, userIdx))
   311→    setTimeout(() => void send(userContent, { retry: isTrailing && !asstMsg.serverId, retryFromId: isTrailing ? undefined : asstMsg.serverId }), 0)
   312→  }

   314→  async function send(overrideMessage?: string, opts?: { retry?: boolean; retryFromId?: string }) {
   315→    const message = (overrideMessage ?? input).trim()
   316→    if (!message || streaming) return
   317→    if (overrideMessage === undefined) setInput('')
   318→    setError(null)
   319→    setStreaming(true)
   320→    setResearching(false)
   321→    setRunningTool(null)
   322→    nearBottomRef.current = true // a sent message always scrolls into view
   323→    streamedRef.current = ''
   324→
   325→    const userMsg: Msg = { id: `u-${Date.now()}`, role: 'user', content: message }
   326→    const asstId = `a-${Date.now()}`
   327→    setMessages((prev) => [...prev, userMsg, { id: asstId, role: 'assistant', content: '', pending: true }])
   328→
   329→    const controller = new AbortController()
   330→    abortRef.current = controller
   331→    const attachable = files.filter((f) => f.status === 'ready')
   332→
   333→    try {
   334→      const res = await fetch(`/api/riishe/chat?stream=1${activeId ? `&conversationId=${activeId}` : ''}`, {
   335→        method: 'POST',
   336→        headers: { 'content-type': 'application/json' },
   337→        body: JSON.stringify({
   338→          message,
   339→          conversationId: activeId ?? undefined,
   340→          documentIds: attachable.length > 0 ? attachable.map((d) => d.id) : undefined,
   341→          retry: opts?.retry ?? false,
   342→          ...(opts?.retryFromId ? { retryFromMessageId: opts.retryFromId } : {}),
   343→        }),
   344→        signal: controller.signal,
   345→      })
   346→      if (!res.ok || !res.body) {
   347→        // Keep the global identity in sync: a 401 here means the session is
   348→        // gone server-side; drop the signed-in shell immediately.
   349→        if (res.status === 401) window.dispatchEvent(new Event('bean:unauthorized'))
   350→        const data = await res.json().catch(() => ({}))
   351→        throw new Error(data?.error?.message ?? `Request failed (${res.status})`)
   352→      }
   353→      const reader = res.body.getReader()
   354→      const decoder = new TextDecoder()
   355→      let buffer = ''
   356→      let convId: string | null = activeId
   357→      let failed = false
   358→
   359→      while (true) {
   360→        const { done, value } = await reader.read()
   361→        if (done) break
   362→        buffer += decoder.decode(value, { stream: true })
   363→        const lines = buffer.split('\n')
   364→        buffer = lines.pop() ?? ''
   365→        for (const line of lines) {
   366→          if (!line.startsWith('data: ')) continue
   367→          let evt: {
   368→            delta?: string
   369→            reset?: boolean
   370→            done?: boolean
   371→            error?: string
   372→            code?: string
   373→            conversationId?: string
   374→            messageId?: string
   375→            partial?: boolean
   376→            stage?: 'researching' | 'tools' | 'solving' | 'verifying'
   377→            tool?: { name: string; status: 'started' | 'done' | 'failed' }
   378→            meta?: MsgMeta
   379→          }
   380→          try { evt = JSON.parse(line.slice(6)) } catch { continue }
   381→          if (evt.conversationId) convId = evt.conversationId
   382→          if (evt.stage === 'researching') setResearching(true)
   383→          if (evt.stage === 'tools') { setResearching(false); setRunningTool('working') }
   384→          if (evt.stage === 'solving' || evt.stage === 'verifying') { setResearching(false); setRunningTool(null) }
   385→          if (evt.tool) {
   386→            if (evt.tool.status === 'started') setRunningTool(evt.tool.name)
   387→            else setRunningTool(null)
   388→          }
   389→          if (evt.reset) {
   390→            // Reset protocol: a verification rewrite replaced the streamed
   391→            // draft — clear it so the final answer REPLACES, never appends.
   392→            streamedRef.current = ''
   393→            setMessages((prev) =>
   394→              prev.map((m) => (m.id === asstId ? { ...m, content: '', pending: true } : m)),
   395→            )
   396→          }
   397→          if (evt.error) { failed = true; setError(evt.error) }
   398→          if (evt.delta) {
   399→            streamedRef.current += evt.delta
   400→            setMessages((prev) =>
   401→              prev.map((m) => (m.id === asstId ? { ...m, content: streamedRef.current, pending: false } : m)),
   402→            )
   403→          }
   404→          if (evt.done) {
   405→            setResearching(false)
   406→            setRunningTool(null)
   407→            if (convId && !activeId) setActiveId(convId)
   408→            if (evt.messageId) {
   409→              setMessages((prev) => prev.map((m) => (m.id === asstId ? { ...m, serverId: evt.messageId } : m)))
   410→            }
   411→            if (evt.meta) {
   412→              setMessages((prev) => prev.map((m) => (m.id === asstId ? { ...m, meta: evt.meta } : m)))
   413→            }
   414→          }
   415→        }
   416→      }
   417→      // If nothing streamed (failed, or the server reported an empty partial),
   418→      // drop the empty assistant bubble — the error banner already explains
   419→      // what happened, and a perpetual "thinking" indicator would be a lie.
   420→      if (streamedRef.current === '') {
   421→        setMessages((prev) => prev.filter((m) => m.id !== asstId))
   422→        // Server-side failure (e.g. upstream rate limit) reported over SSE:
   423→        // the question goes back into the composer so a retry needs no
   424→        // retyping. (Nothing was answered — no partial content is lost.)
   425→        if (failed) restoreQuestionForRetry(message)
   426→      } else if (failed) {
   427→        setMessages((prev) => prev.map((m) => (m.id === asstId ? { ...m, partial: true, pending: false } : m)))
   428→      }
   429→      void loadConversations()
   430→    } catch (e) {
   431→      if ((e as Error).name === 'AbortError') {
   432→        // User pressed stop: keep what arrived and label it honestly.
   433→        if (streamedRef.current !== '') {
   434→          setMessages((prev) => prev.map((m) => (m.id === asstId ? { ...m, partial: true, pending: false } : m)))
   435→        } else {
   436→          setMessages((prev) => prev.filter((m) => m.id !== asstId))
   437→        }
   438→      } else {
   439→        setError((e as Error).message)
   440→        setMessages((prev) => prev.filter((m) => m.id !== asstId || m.content.length > 0))
   441→        restoreQuestionForRetry(message)
   442→      }
   443→    } finally {
   444→      setStreaming(false)
   445→      setResearching(false)
   446→      setRunningTool(null)
   447→      abortRef.current = null
   448→      setMessages((prev) => prev.map((m) => ({ ...m, pending: false })))
   449→    }
   450→  }

   692→/**
   693→ * One assistant response: clean prose (citation markers render as source-name
   694→ * links; unresolved ones are stripped), the honest verification note when one
   695→ * exists, and CONSISTENT actions — Copy response, Regenerate, and View
   696→ * sources (only when sources exist) — on EVERY response, not just the last.
   697→ */
   698→function AssistantBubble({
   699→  msg,
   700→  onRegenerate,
   701→  streaming,
   702→  renderers,
   703→  researching,
   704→  runningTool,
   705→}: {
   706→  msg: Msg
   707→  onRegenerate: () => void
   708→  streaming: boolean
   709→  renderers: Parameters<typeof Markdown>[0]['components']
   710→  researching: boolean
   711→  runningTool: string | null
   712→}) {
   713→  const [sourcesOpen, setSourcesOpen] = useState(false)
   714→  const hasSources = (msg.meta?.sources?.length ?? 0) > 0
   715→  if (!msg.content) {
   716→    // Empty draft: the live status line (search / tools / thinking).
   717→    return researching ? (
   718→      <span className="inline-flex items-center gap-2 text-xs text-white/50">
   719→        <Globe className="h-3.5 w-3.5 animate-pulse text-[#d4af37]" /> Searching the web…
   720→      </span>
   721→    ) : runningTool ? (
   722→      <span className="inline-flex items-center gap-2 text-xs text-white/50">
   723→        <Terminal className="h-3.5 w-3.5 animate-pulse text-[#d4af37]" />
   724→        {runningTool === 'working' ? 'Working…' : 'Running tools…'}
   725→      </span>
   726→    ) : (
   727→      <span className="inline-flex items-center gap-2 text-xs text-white/40">
   728→        <Spinner /> Thinking…
   729→      </span>
   730→    )
   731→  }
   732→  const claims = mapSourceClaims(msg.content, msg.meta?.sources ?? [])
   733→  return (
   734→    <>
   735→      <Markdown
   736→        content={linkifyCitations(msg.content, msg.meta?.sources)}
   737→        components={renderers}
   738→      />
   739→      {msg.pending ? <Spinner className="ml-1 mt-1" /> : null}
   740→      {msg.partial ? (
   741→        <div className="mt-2 inline-flex items-center gap-1.5 rounded border border-amber-500/30 bg-amber-500/10 px-2 py-0.5 text-[10px] text-amber-300">
   742→          <AlertTriangle className="h-3 w-3" /> Incomplete — the stream was stopped or ended early
   743→        </div>
   744→      ) : null}
   745→      <AnswerMeta meta={msg.meta} forceOpen={sourcesOpen} claims={claims} />
   746→      {!msg.pending ? (
   747→        <div className="mt-2 flex flex-wrap items-center gap-2 border-t border-[rgba(212,175,55,0.12)] pt-2">
   748→          <CopyMessageButton content={msg.content} />
   749→          <button
   750→            onClick={onRegenerate}
   751→            disabled={streaming}
   752→            className="inline-flex items-center gap-1 rounded px-1.5 py-0.5 text-[10px] text-white/45 transition-colors hover:bg-white/10 hover:text-white/80 disabled:opacity-40"
   753→            aria-label="Regenerate response"
   754→          >
   755→            <RefreshCw className="h-3 w-3" /> Regenerate
   756→          </button>
   757→          {hasSources ? (
   758→            <button
   759→              onClick={() => setSourcesOpen(true)}
   760→              className="inline-flex items-center gap-1 rounded px-1.5 py-0.5 text-[10px] text-white/45 transition-colors hover:bg-white/10 hover:text-white/80"
   761→              aria-label="View sources"
   762→              aria-expanded={sourcesOpen}
   763→            >
   764→              <ExternalLink className="h-3 w-3" /> View sources
   765→            </button>
   766→          ) : null}
   767→        </div>
   768→      ) : null}
   769→    </>
   770→  )
   771→}

   787→/**
   788→ * Answer-first evidence: NO state badges, NO verification summary, NO tool
   789→ * chips, NO "last checked" telemetry in the reading flow. What remains:
   790→ *  - a short honest verification note ONLY when something material could not
   791→ *    be confirmed (uncovered claims, stale sources, wrong-case retrieval,
   792→ *    failed live verification);
   793→ *  - a subtle collapsed "Sources" section listing what was really retrieved,
   794→ *    each entry with its publisher, date, and the claim it supports.
   795→ */
   796→function AnswerMeta({
   797→  meta,
   798→  forceOpen,
   799→  claims,
   800→}: {
   801→  meta?: MsgMeta
   802→  forceOpen?: boolean
   803→  claims: Map<number, string>
   804→}) {
   805→  const detailsRef = useRef<HTMLDetailsElement>(null)
   806→  useEffect(() => {
   807→    if (forceOpen && detailsRef.current) detailsRef.current.open = true
   808→  }, [forceOpen])
   809→  const sources = meta?.sources ?? []
   810→  const note = meta?.verificationNote
   811→  const researchFailed = meta?.states?.includes('TOOL_UNAVAILABLE') ?? false
   812→  if (!researchFailed && !note && sources.length === 0) return null
   813→  return (
   814→    <div className="mt-2 border-t border-[rgba(212,175,55,0.12)] pt-2">
   815→      {researchFailed ? (
   816→        <div className="text-[11px] text-amber-300/90">
   817→          Live sources could not be checked for this answer — treat current details as unverified.
   818→        </div>
   819→      ) : note ? (
   820→        <div className="text-[11px] leading-snug text-white/40">{note}</div>
   821→      ) : null}
   822→
   823→      {sources.length > 0 ? (
   824→        <details ref={detailsRef} className="group mt-1">
   825→          <summary className="flex w-fit cursor-pointer list-none items-center gap-1 text-[11px] text-white/40 hover:text-white/65">
   826→            <ChevronDown className="h-3 w-3 transition-transform group-open:rotate-180" />
   827→            Sources ({sources.length})
   828→          </summary>
   829→          <ul className="mt-1.5 space-y-2 pl-4">
   830→            {sources.map((s) => (
   831→              <li key={s.id} id={`bean-src-${s.id}`} className="scroll-mt-6 break-words text-[11px] leading-snug">
   832→                <a
   833→                  href={s.url}
   834→                  target="_blank"
   835→                  rel="noopener noreferrer"
   836→                  className="break-words text-[#f0c75e] hover:underline"
   837→                >
   838→                  {s.title.slice(0, 90)}
   839→                  <ExternalLink className="ml-1 inline h-2.5 w-2.5" />
   840→                </a>
   841→                <span className="text-white/35"> — {s.hostName.replace(/^www\./, '')}</span>
   842→                {s.date ? (
   843→                  <span className="ml-1 text-white/30" title="Publication date stated by the source — distinct from the event date">
   844→                    · published {s.date}
   845→                  </span>
   846→                ) : (
   847→                  <span className="ml-1 text-white/25" title="No publication date could be verified for this source">
   848→                    · no date verified
   849→                  </span>
   850→                )}
   851→                {s.social ? <span className="ml-1 text-amber-300/50">· social-media post (unverified provenance)</span> : null}
   852→                <SourceClaimLine content={claims.get(s.id)} />
   853→              </li>
   854→            ))}
   855→          </ul>
   856→        </details>
   857→      ) : null}
   858→    </div>
   859→  )
   860→}
