   121→   121→function useCopy(): [boolean, (text: string) => void] {
   122→   122→  const [copied, setCopied] = useState(false)
   123→   123→  const copy = useCallback((text: string) => {
   124→   124→    const done = () => {
   125→   125→      setCopied(true)
   126→   126→      setTimeout(() => setCopied(false), 1600)
   127→   127→    }
   128→   128→    if (navigator.clipboard?.writeText) {
   129→   129→      navigator.clipboard.writeText(text).then(done).catch(done)
   130→   130→    } else {
   131→   131→      const ta = document.createElement('textarea')
   132→   132→      ta.value = text
   133→   133→      document.body.appendChild(ta)
   134→   134→      ta.select()
   135→   135→      document.execCommand('copy')
   136→   136→      document.body.removeChild(ta)
   137→   137→      done()
   138→   138→    }
   139→   139→  }, [])
   140→   140→  return [copied, copy]
   141→   141→}

   491→   491→  /**
   492→   492→   * Shared SSE reader for a single assistant turn (send + regenerate).
   493→   493→   * Streams into the pending bubble `asstId` and finalizes honestly.
   494→   494→   */
   495→   495→  async function streamChat(opts: {
   496→   496→    message: string
   497→   497→    asstId: string
   498→   498→    searchMode: SearchMode
   499→   499→    retry?: boolean
   500→   500→    regenerateMessageId?: string
   501→   501→    /** Local id of the just-added user bubble; replaced by the real server id. */
   502→   502→    localUserMsgId?: string
   503→   503→  }) {
   504→   504→    const convAtStart = activeId
   505→   505→    const asstId = opts.asstId
   506→   506→    const controller = new AbortController()
   507→   507→    abortRef.current = controller
   508→   508→
   509→   509→    try {
   510→   510→      const res = await fetch(`/api/riishe/chat?stream=1${convAtStart ? `&conversationId=${convAtStart}` : ''}`, {
   511→   511→        method: 'POST',
   512→   512→        headers: { 'content-type': 'application/json' },
   513→   513→        body: JSON.stringify({
   514→   514→          message: opts.message,
   515→   515→          conversationId: convAtStart ?? undefined,
   516→   516→          searchMode: opts.searchMode,
   517→   517→          allowCodeExecution,
   518→   518→          documentIds: documents.length > 0 ? documents.map((d) => d.id) : undefined,
   519→   519→          retry: opts.retry ?? false,
   520→   520→          regenerateMessageId: opts.regenerateMessageId,
   521→   521→        }),
   522→   522→        signal: controller.signal,
   523→   523→      })
   524→   524→      if (!res.ok || !res.body) {
   525→   525→        // Keep the global identity in sync: a 401 here means the session is
   526→   526→        // gone server-side; drop the signed-in shell immediately.
   527→   527→        if (res.status === 401) window.dispatchEvent(new Event('bean:unauthorized'))
   528→   528→        const data = await res.json().catch(() => ({}))
   529→   529→        if (data?.error?.message) throw new Error(data.error.message)
   530→   530→        // A 502/504 with no API error body means an intermediary (serving
   531→   531→        // gateway) ended the request before the app could answer — typically
   532→   532→        // when a researched turn exceeds the gateway's time limit. Say that
   533→   533→        // honestly instead of a bare status code; the server may still have
   534→   534→        // finished, so the saved conversation holds the answer if it exists.
   535→   535→        if (res.status === 502 || res.status === 504) {
   536→   536→          throw new Error(
   537→   537→            'The connection was cut off before the answer completed (gateway ' +
   538→   538→              `${res.status}). This can happen when a researched reply takes too long. ` +
   539→   539→              'Please retry — or send again with search off for a faster response.',
   540→   540→          )
   541→   541→        }
   542→   542→        throw new Error(`Request failed (${res.status})`)
   543→   543→      }
   544→   544→      const reader = res.body.getReader()
   545→   545→      const decoder = new TextDecoder()
   546→   546→      let buffer = ''
   547→   547→      let convId: string | null = convAtStart
   548→   548→      let failed = false
   549→   549→
   550→   550→      while (true) {
   551→   551→        const { done, value } = await reader.read()
   552→   552→        if (done) break
   553→   553→        buffer += decoder.decode(value, { stream: true })
   554→   554→        const lines = buffer.split('\n')
   555→   555→        buffer = lines.pop() ?? ''
   556→   556→        for (const line of lines) {
   557→   557→          if (!line.startsWith('data: ')) continue
   558→   558→          let evt: {
   559→   559→            delta?: string
   560→   560→            done?: boolean
   561→   561→            error?: string
   562→   562→            reset?: boolean
   563→   563→            content?: string
   564→   564→            conversationId?: string
   565→   565→            partial?: boolean
   566→   566→            userMessageId?: string
   567→   567→            messageId?: string | null
   568→   568→            stage?: 'researching' | 'tools' | 'solving' | 'verifying'
   569→   569→            research?: { queries?: string[] }
   570→   570→            tool?: { name: string; status: 'started' | 'done' | 'failed' }
   571→   571→            meta?: MsgMeta
   572→   572→          }
   573→   573→          try { evt = JSON.parse(line.slice(6)) } catch { continue }
   574→   574→          if (evt.conversationId) convId = evt.conversationId
   575→   575→          if (evt.userMessageId && opts.localUserMsgId) {
   576→   576→            // Adopt the real persisted id for the user turn so per-message
   577→   577→            // actions address server rows.
   578→   578→            const serverId = evt.userMessageId
   579→   579→            const localId = opts.localUserMsgId
   580→   580→            setMessages((prev) => prev.map((m) => (m.id === localId ? { ...m, id: serverId } : m)))
   581→   581→          }
   582→   582→          if (evt.stage === 'researching') { setResearching(true); setResearchQueries([]) }
   583→   583→          if (evt.stage === 'tools') { setResearching(false); setRunningTool('working') }
   584→   584→          if (evt.stage === 'solving' || evt.stage === 'verifying') { setResearching(false); setRunningTool(null) }
   585→   585→          if (evt.tool) {
   586→   586→            if (evt.tool.status === 'started') setRunningTool(evt.tool.name)
   587→   587→            else setRunningTool(null)
   588→   588→          }
   589→   589→          if (evt.research?.queries) setResearchQueries(evt.research.queries)
   590→   590→          if (evt.error) { failed = true; setError(evt.error) }
   591→   591→          if (evt.reset) {
   592→   592→            // A verification rewrite REPLACES the buffered draft: clear the
   593→   593→            // accumulated text before the corrected delta arrives.
   594→   594→            streamedRef.current = ''
   595→   595→            setMessages((prev) => prev.map((m) => (m.id === asstId ? { ...m, content: '' } : m)))
   596→   596→          }
   597→   597→          if (evt.delta) {
   598→   598→            streamedRef.current += evt.delta
   599→   599→            setMessages((prev) =>
   600→   600→              prev.map((m) => (m.id === asstId ? { ...m, content: streamedRef.current, pending: false } : m)),
   601→   601→            )
   602→   602→          }
   603→   603→          if (evt.done) {
   604→   604→            setResearching(false)
   605→   605→            setRunningTool(null)
   606→   606→            if (convId && !convAtStart) setActiveId(convId)
   607→   607→            const serverAsstId = evt.messageId ?? null
   608→   608→            // The persisted content is authoritative (verification rewrites,
   609→   609→            // SELF_CHECK lifting): adopt it as the visible answer.
   610→   610→            const finalText = typeof evt.content === 'string' && evt.content ? evt.content : streamedRef.current
   611→   611→            streamedRef.current = finalText
   612→   612→            setMessages((prev) =>
   613→   613→              prev.map((m) =>
   614→   614→                m.id === asstId ? { ...m, id: serverAsstId ?? m.id, content: finalText, meta: evt.meta ?? m.meta } : m,
   615→   615→              ),
   616→   616→            )
   617→   617→          }
   618→   618→        }
   619→   619→      }
   620→   620→      // If nothing streamed (failed, or the server reported an empty partial),
   621→   621→      // drop the empty assistant bubble — the error banner already explains
   622→   622→      // what happened, and a perpetual "thinking" indicator would be a lie.
   623→   623→      if (streamedRef.current === '') {
   624→   624→        setMessages((prev) => prev.filter((m) => m.id !== asstId))
   625→   625→      } else if (failed) {
   626→   626→        setMessages((prev) => prev.map((m) => (m.id === asstId ? { ...m, partial: true, pending: false } : m)))
   627→   627→      }
   628→   628→      void loadConversations()
   629→   629→    } catch (e) {
   630→   630→      if ((e as Error).name === 'AbortError') {
   631→   631→        // User pressed stop: keep what arrived and label it honestly.
   632→   632→        if (streamedRef.current !== '') {
   633→   633→          setMessages((prev) => prev.map((m) => (m.id === asstId ? { ...m, partial: true, pending: false } : m)))
   634→   634→        } else {
   635→   635→          setMessages((prev) => prev.filter((m) => m.id !== asstId))
   636→   636→        }
   637→   637→      } else {
   638→   638→        setError((e as Error).message)
   639→   639→        setMessages((prev) => prev.filter((m) => m.id !== asstId || m.content.length > 0))
   640→   640→      }
   641→   641→    } finally {
   642→   642→      setStreaming(false)
   643→   643→      setResearching(false)
   644→   644→      setRunningTool(null)
   645→   645→      abortRef.current = null
   646→   646→      setMessages((prev) => prev.map((m) => ({ ...m, pending: false })))
   647→   647→    }
   648→   648→  }

   650→   650→  async function send(overrideMessage?: string, retry = false) {
   651→   651→    const message = (overrideMessage ?? input).trim()
   652→   652→    if (!message || streaming) return
   653→   653→    if (overrideMessage === undefined) setInput('')

   660→   660→

   662→   662→    const asstId = `a-${Date.now()}`
   663→   663→    setMessages((prev) => [...prev, userMsg, { id: asstId, role: 'assistant', content: '', pending: true }])
   664→   664→    await streamChat({ message, asstId, searchMode, retry, localUserMsgId: userMsg.id })
   665→   665→  }

   667→   667→  /**
   668→   668→   * Regenerate ANY assistant response in place: the old answer (and every
   669→   669→   * message after it) is discarded locally and server-side; the original
   670→   670→   * question is re-answered with the mode recorded for THAT answer, so the
   671→   671→   * regeneration reflects the original context and retrieval choice. While
   672→   672→   * a stream is running the action is disabled — no duplicate requests.
   673→   673→   */
   674→   674→  async function regenerateFrom(messageId: string) {

   680→   680→    const userBefore = idx > 0 ? messages[idx - 1] : null

   682→   682→    const mode: SearchMode = target.meta?.searchMode ?? searchMode
   683→   683→
   684→   684→    setError(null)

   692→   692→      ...prev.slice(0, idx),
   693→   693→      { id: asstId, role: 'assistant' as const, content: '', pending: true },
   694→   694→    ])
   695→   695→    await streamChat({ message: userBefore.content, asstId, searchMode: mode, regenerateMessageId: messageId })
   696→   696→  }

   711→   711→            <p className="px-2 py-6 text-center text-xs text-white/30">No conversations yet</p>
   712→   712→          ) : (
   713→   713→            conversations.map((c) => (
   714→   714→              <div
   715→   715→                key={c.id}
   716→   716→                className={cn(
   717→   717→                  'group mb-0.5 flex cursor-pointer items-center justify-between rounded-lg px-3 py-2 text-[13px] transition-colors',
   718→   718→                  activeId === c.id ? 'bg-[rgba(212,175,55,0.1)] text-[#f0c75e]' : 'text-white/60 hover:bg-white/5',
   719→   719→                )}
   720→   720→                onClick={() => openConversation(c.id)}
   721→   721→              >
   722→   722→                <div className="min-w-0">

   725→   725→                </div>

   894→   894→/**
   895→   895→ * One assistant message: markdown with citation chips, evidence panel,
   896→   896→ * compact sources and the Copy/Regenerate/View-sources action row. The
   897→   897→ * action row renders for EVERY assistant response (short or long), in every
   898→   898→ * mode — never only for the latest one.
   899→   899→ */
   900→   900→function AssistantMessage({
   901→   901→  m,
   902→   902→  researching,
   903→   903→  runningTool,
   904→   904→  researchQueries,
   905→   905→  streaming,
   906→   906→  isLatest,
   907→   907→  onRegenerate,
   908→   908→}: {
   909→   909→  m: Msg
   910→   910→  researching: boolean
   911→   911→  runningTool: string | null
   912→   912→  researchQueries: string[]
   913→   913→  streaming: boolean
   914→   914→  isLatest: boolean
   915→   915→  onRegenerate: () => void
   916→   916→}) {
   917→   917→  const meta = m.meta
   918→   918→  const [previewId, setPreviewId] = useState<number | null>(null)
   919→   919→  const [sourcesOpen, setSourcesOpen] = useState(false)
   920→   920→  const [highlightId, setHighlightId] = useState<number | null>(null)
   921→   921→  const entryRefs = useRef(new Map<number, HTMLLIElement | null>())
   922→   922→
   923→   923→  const registerRef = useCallback((id: number, el: HTMLLIElement | null) => {
   924→   924→    entryRefs.current.set(id, el)
   925→   925→  }, [])
   926→   926→
   927→   927→  // Citation chip click: open the Sources section and spotlight the entry.
   928→   928→  const jumpToSource = useCallback((id: number) => {
   929→   929→    setSourcesOpen(true)
   930→   930→    setHighlightId(id)
   931→   931→    window.setTimeout(() => {
   932→   932→      entryRefs.current.get(id)?.scrollIntoView({ behavior: 'smooth', block: 'center' })
   933→   933→    }, 80)
   934→   934→    window.setTimeout(() => setHighlightId((cur) => (cur === id ? null : cur)), 2400)
   935→   935→  }, [])
   936→   936→
   937→   937→  const sources = meta?.sources ?? []
   938→   938→  const hasSources = sources.length > 0
   939→   939→
   940→   940→  const mdComponents = {
   941→   941→    a: ({ href, children }: { href?: string; children?: React.ReactNode }) => {
   942→   942→      if (href?.startsWith('#bean-src-')) {
   943→   943→        const id = Number(href.replace('#bean-src-', ''))
   944→   944→        const source = sources.find((s) => s.id === id)
   945→   945→        if (!source) return <span className="text-white/60">{children}</span>
   946→   946→        const label = String(children ?? '').replace(/[()]/g, '')
   947→   947→        return (
   948→   948→          <CitationLink
   949→   949→            id={id}
   950→   950→            label={label}
   951→   951→            source={source}
   952→   952→            previewed={previewId === id}
   953→   953→            onPreview={setPreviewId}
   954→   954→            onLeave={() => setPreviewId(null)}
   955→   955→            onJump={jumpToSource}
   956→   956→          />
   957→   957→        )
   958→   958→      }
   959→   959→      return (
   960→   960→        <a href={href} target="_blank" rel="noopener noreferrer" className="text-[#f0c75e] underline decoration-[rgba(212,175,55,0.4)] hover:decoration-[#f0c75e]">
   961→   961→          {children}
   962→   962→        </a>
   963→   963→      )
   964→   964→    },
   965→   965→    pre: ({ children }: { children?: React.ReactNode }) => {
   966→   966→      // React-markdown nests <code> inside <pre>; extract for highlighting.
   967→   967→      const child = Array.isArray(children) ? children[0] : children
   968→   968→      const props = (child as { props?: { className?: string; children?: ReactNode } })?.props
   969→   969→      const lang = props?.className?.replace(/language-/, '') || undefined
   970→   970→      const code = String(props?.children ?? '').replace(/\n$/, '')
   971→   971→      if (code === '') return <pre>{children}</pre>
   972→   972→      return <CodeBlock code={code} lang={lang} />
   973→   973→    },
   974→   974→  }
   975→   975→
   976→   976→  return (
   977→   977→    <div className="flex justify-start">
   978→   978→      <div className="bean-glass max-w-[92%] rounded-2xl px-4 py-3 text-white/85 sm:max-w-[85%]">
   979→   979→        {m.content ? (
   980→   980→          <>
   981→   981→            <Markdown content={renderNamedCitations(m.content, sources)} components={mdComponents} />
   982→   982→            {m.pending ? <Spinner className="ml-1 mt-1" /> : null}
   983→   983→            {m.partial ? (
   984→   984→              <div className="mt-2 inline-flex items-center gap-1.5 rounded border border-amber-500/30 bg-amber-500/10 px-2 py-0.5 text-[10px] text-amber-300">
   985→   985→                <AlertTriangle className="h-3 w-3" /> Incomplete — the stream was stopped or ended early
   986→   986→              </div>
   987→   987→            ) : null}
   988→   988→            {meta ? <AnswerEvidence meta={meta} /> : null}
   989→   989→            {hasSources && meta ? (
   990→   990→              <SourcesSection
   991→   991→                sources={sources}
   992→   992→                open={sourcesOpen}
   993→   993→                onOpenChange={setSourcesOpen}
   994→   994→                highlightId={highlightId}
   995→   995→                registerRef={registerRef}
   996→   996→              />
   997→   997→            ) : null}
   998→   998→            {!m.pending ? (
   999→   999→              <MessageActions
  1000→  1000→                copyText={stripCitationMarkers(m.content)}
  1001→  1001→                disabled={streaming}
  1002→  1002→                hasSources={hasSources}
  1003→  1003→                sourcesOpen={sourcesOpen}
  1004→  1004→                onToggleSources={() => setSourcesOpen((v) => !v)}
  1005→  1005→                onRegenerate={onRegenerate}
  1006→  1006→                modeLabel={meta?.searchMode}
  1007→  1007→              />
  1008→  1008→            ) : null}
  1009→  1009→          </>
  1010→  1010→        ) : researching && isLatest ? (
  1011→  1011→          <span className="inline-flex items-center gap-2 text-xs text-white/50">
  1012→  1012→            <Globe className="h-3.5 w-3.5 animate-pulse text-[#d4af37]" /> Searching the web…
  1013→  1013→            {researchQueries.length > 0 ? (
  1014→  1014→              <span className="text-white/30">({researchQueries.slice(0, 2).join(' · ')})</span>
  1015→  1015→            ) : null}
  1016→  1016→          </span>
  1017→  1017→        ) : runningTool && isLatest ? (
  1018→  1018→          <span className="inline-flex items-center gap-2 text-xs text-white/50">
  1019→  1019→            <Terminal className="h-3.5 w-3.5 animate-pulse text-[#d4af37]" />
  1020→  1020→            {runningTool === 'working' ? 'Running server-side tools…' : `Running ${runningTool}…`}
  1021→  1021→          </span>
  1022→  1022→        ) : (
  1023→  1023→          <span className="inline-flex items-center gap-2 text-xs text-white/40">
  1024→  1024→            <Spinner /> RIISHE is thinking…
  1025→  1025→          </span>
  1026→  1026→        )}
  1027→  1027→      </div>
  1028→  1028→    </div>
  1029→  1029→  )
  1030→  1030→}

  1032→  1032→/** Evidence panel: state badges, verification summary and honest degradation. */

  1035→  1035→    .map((s) => STATE_BADGES[s])

  1037→  1037→  const checks = meta.verification ?? []
  1038→  1038→  const toolRuns = meta.toolRuns ?? []
  1039→  1039→
  1040→  1040→  if (badges.length === 0 && checks.length === 0) return null
  1041→  1041→
