   111→async function pollJob<T extends { status: string; id: string }>(
   112→  id: string,
   113→  urlFor: (jobId: string) => string,
   114→  maxTries = 80,
   115→): Promise<T | null> {
   116→  for (let i = 0; i < maxTries; i++) {
   117→    await new Promise((r) => setTimeout(r, 1500))
   118→    try {
   119→      const d = await beanFetch<{ job: T }>(urlFor(id))
   120→      if (d.job.status === 'completed' || d.job.status === 'failed') return d.job
   121→    } catch {
   122→      return null
   123→    }
   124→  }
   125→  return null
   126→}

   357→  async function startRecording() {
   358→    setActionError(null)
   359→    // Interrupt any playback the moment the user grabs the mic.
   360→    liveAudioRef.current?.pause()
   361→    if (!availability || !cap('liveVoice')?.available) {
   362→      setActionError('Live voice is not available right now — see the service status above.')
   363→      return
   364→    }
   365→    if (typeof navigator === 'undefined' || !navigator.mediaDevices?.getUserMedia || typeof MediaRecorder === 'undefined') {
   366→      setActionError('This browser does not support microphone recording (MediaRecorder). Try a current version of Chrome, Edge, Firefox or Safari.')
   367→      return
   368→    }
   369→    try {
   370→      const stream = await navigator.mediaDevices.getUserMedia({ audio: true })
   371→      const mimeType = MediaRecorder.isTypeSupported('audio/webm;codecs=opus') ? 'audio/webm;codecs=opus' : ''
   372→      const recorder = new MediaRecorder(stream, mimeType ? { mimeType } : undefined)
   373→      chunksRef.current = []
   374→      recorder.ondataavailable = (ev) => {
   375→        if (ev.data.size > 0) chunksRef.current.push(ev.data)
   376→      }
   377→      recorder.onstop = () => {
   378→        stream.getTracks().forEach((tr) => tr.stop())
   379→        void processTurn(new Blob(chunksRef.current, { type: recorder.mimeType || 'audio/webm' }))
   380→      }
   381→      recorderRef.current = recorder
   382→      recorder.start()
   383→      setLivePhase('recording')
   384→      setRecordSecs(0)
   385→      if (timerRef.current) clearInterval(timerRef.current)
   386→      timerRef.current = setInterval(() => setRecordSecs((s) => s + 1), 1000)
   387→    } catch (e) {
   388→      const err = e as DOMException
   389→      if (err.name === 'NotAllowedError' || err.name === 'SecurityError') {
   390→        setActionError('Microphone access was denied. Allow microphone permission for this site in your browser settings, then try again.')
   391→      } else if (err.name === 'NotFoundError' || err.name === 'DevicesNotFoundError') {
   392→        setActionError('No microphone was found on this device.')
   393→      } else {
   394→        setActionError(`Could not start recording: ${err.message || err.name}`)
   395→      }
   396→    }
   397→  }

   406→  async function processTurn(blob: Blob) {
   407→    if (blob.size === 0) {
   408→      setLivePhase('idle')
   409→      setActionError('The recording was empty — try speaking a little longer.')
   410→      return
   411→    }
   412→    setLivePhase('transcribing')
   413→    setActionError(null)
   414→    try {
   415→      // 1. Transcribe the recorded turn through the REAL ASR pipeline.
   416→      const fd = new FormData()
   417→      fd.append('file', new File([blob], 'turn.webm', { type: blob.type || 'audio/webm' }))
   418→      const res = await fetch('/api/gangzii/transcribe', { method: 'POST', body: fd })
   419→      const data = (await res.json().catch(() => ({}))) as { ok?: boolean; job?: { id: string }; error?: { message?: string } }
   420→      if (!res.ok || data.ok === false || !data.job) {
   421→        throw new Error(data.error?.message ?? 'The recorded turn could not be transcribed.')
   422→      }
   423→      const tJob = await pollJob<LabJob>(data.job.id, (id) => `/api/gangzii/jobs/${id}`)
   424→      if (!tJob || tJob.status !== 'completed' || !tJob.transcript) {
   425→        throw new Error(tJob?.errorMessage ?? 'Transcription failed for this turn.')
   426→      }
   427→      const transcript = tJob.transcript
   428→
   429→      // 2. Ask the BEAN conversation model for a spoken-style reply.
   430→      setLivePhase('thinking')
   431→      const live = await beanFetch<{ reply: string }>('/api/gangzii/live', {
   432→        method: 'POST',
   433→        json: { transcript, history: historyRef.current },
   434→      })
   435→
   436→      // 3. Speak the reply through the REAL TTS job pipeline.
   437→      setLivePhase('speaking')
   438→      await beanFetch('/api/gangzii/tts', {
   439→        method: 'POST',
   440→        json: { text: live.reply, voice: liveVoice, speed: 1.0 },
   441→      })
   442→      const lists = await loadLists()
   443→      const spoken = lists.tts.find((j) => j.status !== 'failed' && j.voice === liveVoice && j.text === live.reply)
   444→      let audioUrl: string | null = null
   445→      if (spoken) {
   446→        const done = spoken.status === 'completed' ? spoken : await pollJob<TtsJob>(spoken.id, (id) => `/api/gangzii/jobs/${id}`)
   447→        if (done && done.status === 'completed' && done.audioUrl) audioUrl = done.audioUrl
   448→      }
   449→
   450→      const turn: Turn = { id: tJob.id, transcript, reply: live.reply, audioUrl, error: null }
   451→      historyRef.current = [...historyRef.current, { role: 'user' as const, content: transcript }, { role: 'assistant' as const, content: live.reply }].slice(-12)
   452→      setTurns((prev) => [...prev, turn])
   453→
   454→      if (audioUrl) {
   455→        // Autoplay the spoken reply; browsers may block — fall back to manual play.
   456→        try {
   457→          const el = liveAudioRef.current
   458→          if (el) {
   459→            el.src = audioUrl
   460→            await el.play()
   461→          }
   462→        } catch {
   463→          /* autoplay blocked — the turn card exposes a play button */
   464→        }
   465→      }
   466→      setLivePhase('idle')
   467→    } catch (e) {
   468→      setTurns((prev) => [
   469→        ...prev,
   470→        { id: `err-${Date.now()}`, transcript: '', reply: '', audioUrl: null, error: (e as Error).message },
   471→      ])
   472→      setLivePhase('idle')
   473→    }
   474→  }
