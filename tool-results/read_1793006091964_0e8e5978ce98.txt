   121→function useCopy(): [boolean, (text: string) => void] {
   122→  const [copied, setCopied] = useState(false)
   123→  const copy = useCallback((text: string) => {
   124→    const done = () => {
   125→      setCopied(true)
   126→      setTimeout(() => setCopied(false), 1600)
   127→    }
   128→    if (navigator.clipboard?.writeText) {
   129→      navigator.clipboard.writeText(text).then(done).catch(done)
   130→    } else {
   131→      const ta = document.createElement('textarea')
   132→      ta.value = text
   133→      document.body.appendChild(ta)
   134→      ta.select()
   135→      document.execCommand('copy')
   136→      document.body.removeChild(ta)
   137→      done()
   138→    }
   139→  }, [])
   140→  return [copied, copy]
   141→}

   491→  /**
   492→   * Shared SSE reader for a single assistant turn (send + regenerate).
   493→   * Streams into the pending bubble `asstId` and finalizes honestly.
   494→   */
   495→  async function streamChat(opts: {
   496→    message: string
   497→    asstId: string
   498→    searchMode: SearchMode
   499→    retry?: boolean
   500→    regenerateMessageId?: string
   501→    /** Local id of the just-added user bubble; replaced by the real server id. */
   502→    localUserMsgId?: string
   503→  }) {
   504→    const convAtStart = activeId
   505→    const asstId = opts.asstId
   506→    const controller = new AbortController()
   507→    abortRef.current = controller
   508→
   509→    try {
   510→      const res = await fetch(`/api/riishe/chat?stream=1${convAtStart ? `&conversationId=${convAtStart}` : ''}`, {
   511→        method: 'POST',
   512→        headers: { 'content-type': 'application/json' },
   513→        body: JSON.stringify({
   514→          message: opts.message,
   515→          conversationId: convAtStart ?? undefined,
   516→          searchMode: opts.searchMode,
   517→          allowCodeExecution,
   518→          documentIds: documents.length > 0 ? documents.map((d) => d.id) : undefined,
   519→          retry: opts.retry ?? false,
   520→          regenerateMessageId: opts.regenerateMessageId,
   521→        }),
   522→        signal: controller.signal,
   523→      })
   524→      if (!res.ok || !res.body) {
   525→        // Keep the global identity in sync: a 401 here means the session is
   526→        // gone server-side; drop the signed-in shell immediately.
   527→        if (res.status === 401) window.dispatchEvent(new Event('bean:unauthorized'))
   528→        const data = await res.json().catch(() => ({}))
   529→        if (data?.error?.message) throw new Error(data.error.message)
   530→        // A 502/504 with no API error body means an intermediary (serving
   531→        // gateway) ended the request before the app could answer — typically
   532→        // when a researched turn exceeds the gateway's time limit. Say that
   533→        // honestly instead of a bare status code; the server may still have
   534→        // finished, so the saved conversation holds the answer if it exists.
   535→        if (res.status === 502 || res.status === 504) {
   536→          throw new Error(
   537→            'The connection was cut off before the answer completed (gateway ' +
   538→              `${res.status}). This can happen when a researched reply takes too long. ` +
   539→              'Please retry — or send again with search off for a faster response.',
   540→          )
   541→        }
   542→        throw new Error(`Request failed (${res.status})`)
   543→      }
   544→      const reader = res.body.getReader()
   545→      const decoder = new TextDecoder()
   546→      let buffer = ''
   547→      let convId: string | null = convAtStart
   548→      let failed = false
   549→
   550→      while (true) {
   551→        const { done, value } = await reader.read()
   552→        if (done) break
   553→        buffer += decoder.decode(value, { stream: true })
   554→        const lines = buffer.split('\n')
   555→        buffer = lines.pop() ?? ''
   556→        for (const line of lines) {
   557→          if (!line.startsWith('data: ')) continue
   558→          let evt: {
   559→            delta?: string
   560→            done?: boolean
   561→            error?: string
   562→            reset?: boolean
   563→            content?: string
   564→            conversationId?: string
   565→            partial?: boolean
   566→            userMessageId?: string
   567→            messageId?: string | null
   568→            stage?: 'researching' | 'tools' | 'solving' | 'verifying'
   569→            research?: { queries?: string[] }
   570→            tool?: { name: string; status: 'started' | 'done' | 'failed' }
   571→            meta?: MsgMeta
   572→          }
   573→          try { evt = JSON.parse(line.slice(6)) } catch { continue }
   574→          if (evt.conversationId) convId = evt.conversationId
   575→          if (evt.userMessageId && opts.localUserMsgId) {
   576→            // Adopt the real persisted id for the user turn so per-message
   577→            // actions address server rows.
   578→            const serverId = evt.userMessageId
   579→            const localId = opts.localUserMsgId
   580→            setMessages((prev) => prev.map((m) => (m.id === localId ? { ...m, id: serverId } : m)))
   581→          }
   582→          if (evt.stage === 'researching') { setResearching(true); setResearchQueries([]) }
   583→          if (evt.stage === 'tools') { setResearching(false); setRunningTool('working') }
   584→          if (evt.stage === 'solving' || evt.stage === 'verifying') { setResearching(false); setRunningTool(null) }
   585→          if (evt.tool) {
   586→            if (evt.tool.status === 'started') setRunningTool(evt.tool.name)
   587→            else setRunningTool(null)
   588→          }
   589→          if (evt.research?.queries) setResearchQueries(evt.research.queries)
   590→          if (evt.error) { failed = true; setError(evt.error) }
   591→          if (evt.reset) {
   592→            // A verification rewrite REPLACES the buffered draft: clear the
   593→            // accumulated text before the corrected delta arrives.
   594→            streamedRef.current = ''
   595→            setMessages((prev) => prev.map((m) => (m.id === asstId ? { ...m, content: '' } : m)))
   596→          }
   597→          if (evt.delta) {
   598→            streamedRef.current += evt.delta
   599→            setMessages((prev) =>
   600→              prev.map((m) => (m.id === asstId ? { ...m, content: streamedRef.current, pending: false } : m)),
   601→            )
   602→          }
   603→          if (evt.done) {
   604→            setResearching(false)
   605→            setRunningTool(null)
   606→            if (convId && !convAtStart) setActiveId(convId)
   607→            const serverAsstId = evt.messageId ?? null
   608→            // The persisted content is authoritative (verification rewrites,
   609→            // SELF_CHECK lifting): adopt it as the visible answer.
   610→            const finalText = typeof evt.content === 'string' && evt.content ? evt.content : streamedRef.current
   611→            streamedRef.current = finalText
   612→            setMessages((prev) =>
   613→              prev.map((m) =>
   614→                m.id === asstId ? { ...m, id: serverAsstId ?? m.id, content: finalText, meta: evt.meta ?? m.meta } : m,
   615→              ),
   616→            )
   617→          }
   618→        }
   619→      }
   620→      // If nothing streamed (failed, or the server reported an empty partial),
   621→      // drop the empty assistant bubble — the error banner already explains
   622→      // what happened, and a perpetual "thinking" indicator would be a lie.
   623→      if (streamedRef.current === '') {
   624→        setMessages((prev) => prev.filter((m) => m.id !== asstId))
   625→      } else if (failed) {
   626→        setMessages((prev) => prev.map((m) => (m.id === asstId ? { ...m, partial: true, pending: false } : m)))
   627→      }
   628→      void loadConversations()
   629→    } catch (e) {
   630→      if ((e as Error).name === 'AbortError') {
   631→        // User pressed stop: keep what arrived and label it honestly.
   632→        if (streamedRef.current !== '') {
   633→          setMessages((prev) => prev.map((m) => (m.id === asstId ? { ...m, partial: true, pending: false } : m)))
   634→        } else {
   635→          setMessages((prev) => prev.filter((m) => m.id !== asstId))
   636→        }
   637→      } else {
   638→        setError((e as Error).message)
   639→        setMessages((prev) => prev.filter((m) => m.id !== asstId || m.content.length > 0))
   640→      }
   641→    } finally {
   642→      setStreaming(false)
   643→      setResearching(false)
   644→      setRunningTool(null)
   645→      abortRef.current = null
   646→      setMessages((prev) => prev.map((m) => ({ ...m, pending: false })))
   647→    }
   648→  }

   650→  async function send(overrideMessage?: string, retry = false) {
   651→    const message = (overrideMessage ?? input).trim()
   652→    if (!message || streaming) return
   653→    if (overrideMessage === undefined) setInput('')
   654→    setError(null)
   655→    setStreaming(true)
   656→    setResearching(false)
   657→    setRunningTool(null)
   658→    setResearchQueries([])
   659→    streamedRef.current = ''
   660→
   661→    const userMsg: Msg = { id: `u-${Date.now()}`, role: 'user', content: message }
   662→    const asstId = `a-${Date.now()}`
   663→    setMessages((prev) => [...prev, userMsg, { id: asstId, role: 'assistant', content: '', pending: true }])
   664→    await streamChat({ message, asstId, searchMode, retry, localUserMsgId: userMsg.id })
   665→  }

   667→  /**
   668→   * Regenerate ANY assistant response in place: the old answer (and every
   669→   * message after it) is discarded locally and server-side; the original
   670→   * question is re-answered with the mode recorded for THAT answer, so the
   671→   * regeneration reflects the original context and retrieval choice. While
   672→   * a stream is running the action is disabled — no duplicate requests.
   673→   */
   674→  async function regenerateFrom(messageId: string) {
   675→    if (streaming) return
   676→    const idx = messages.findIndex((m) => m.id === messageId)
   677→    if (idx < 0) return
   678→    const target = messages[idx]
   679→    if (target.role !== 'assistant' || target.pending) return
   680→    const userBefore = idx > 0 ? messages[idx - 1] : null
   681→    if (!userBefore || userBefore.role !== 'user') return
   682→    const mode: SearchMode = target.meta?.searchMode ?? searchMode
   683→
   684→    setError(null)
   685→    setStreaming(true)
   686→    setResearching(false)
   687→    setRunningTool(null)
   688→    setResearchQueries([])
   689→    streamedRef.current = ''
   690→    const asstId = `a-${Date.now()}`
   691→    setMessages((prev) => [
   692→      ...prev.slice(0, idx),
   693→      { id: asstId, role: 'assistant' as const, content: '', pending: true },
   694→    ])
   695→    await streamChat({ message: userBefore.content, asstId, searchMode: mode, regenerateMessageId: messageId })
   696→  }
   697→
   698→  const lastAssistantId = [...messages].reverse().find((m) => m.role === 'assistant' && !m.pending)?.id
   699→
   700→  return (
   701→    <div className="flex h-[calc(100vh-3.5rem-2rem)] min-h-[540px] lg:h-[calc(100vh-3.5rem-1.5rem)]">
   702→      {/* Conversation rail */}
   703→      <aside className="hidden w-60 shrink-0 flex-col border-r border-[rgba(212,175,55,0.1)] md:flex">
   704→        <div className="p-3">
   705→          <Button onClick={newConversation} className="w-full justify-start bg-[rgba(212,175,55,0.12)] text-[#f0c75e] hover:bg-[rgba(212,175,55,0.2)]" variant="outline">
   706→            <Plus className="mr-1.5 h-4 w-4" /> New conversation
   707→          </Button>
   708→        </div>
   709→        <ScrollArea className="flex-1 px-2 pb-3">
   710→          {conversations.length === 0 ? (
   711→            <p className="px-2 py-6 text-center text-xs text-white/30">No conversations yet</p>
   712→          ) : (
   713→            conversations.map((c) => (
   714→              <div
   715→                key={c.id}
   716→                className={cn(
   717→                  'group mb-0.5 flex cursor-pointer items-center justify-between rounded-lg px-3 py-2 text-[13px] transition-colors',
   718→                  activeId === c.id ? 'bg-[rgba(212,175,55,0.1)] text-[#f0c75e]' : 'text-white/60 hover:bg-white/5',
   719→                )}
   720→                onClick={() => openConversation(c.id)}
   721→              >
   722→                <div className="min-w-0">
   723→                  <div className="truncate">{c.title}</div>
   724→                  <div className="text-[10px] text-white/30">{c.messages} msgs · {timeAgo(c.updatedAt)}</div>
   725→                </div>
   726→                <button
   727→                  className="ml-2 hidden text-white/30 hover:text-red-400 group-hover:block"
   728→                  onClick={(e) => { e.stopPropagation(); void deleteConversation(c.id) }}
   729→                  aria-label={`Delete ${c.title}`}
   730→                >
   731→                  <Trash2 className="h-3.5 w-3.5" />
   732→                </button>
   733→              </div>
   734→            ))
   735→          )}
   736→        </ScrollArea>
   737→      </aside>
   738→
   739→      {/* Chat column */}
   740→      <div className="flex min-w-0 flex-1 flex-col">
   741→        <div className="flex items-center gap-2 border-b border-[rgba(212,175,55,0.1)] px-5 py-3">
   742→          <MessageSquareText className="h-4 w-4 text-[#d4af37]" />
   743→          <div className="text-sm font-semibold tracking-wide">RIISHE</div>
   744→          <span className="text-xs text-white/35">Conversational AI · research · verified reasoning</span>
   745→        </div>
   746→
   747→        <div className="bean-scroll flex-1 overflow-y-auto px-5 py-4">
   748→          {messages.length === 0 ? (
   749→            <EmptyState
   750→              icon={<MessageSquareText className="h-10 w-10" />}
   751→              title="Start a conversation"
   752→              hint="Ask questions, request derivations, code help or research on current topics. RIISHE searches live sources when a question needs them, runs real tools, and verifies its arithmetic before answering. Attach a document for grounded analysis."
   753→            />
   754→          ) : (
   755→            <div className="mx-auto max-w-3xl space-y-4">
   756→              {messages.map((m) =>
   757→                m.role === 'user' ? (
   758→                  <div key={m.id} className="flex justify-end">
   759→                    <div className="max-w-[85%] rounded-2xl bg-[rgba(212,175,55,0.14)] px-4 py-3 text-white/90">
   760→                      <p className="whitespace-pre-wrap text-[14.5px] leading-relaxed">{m.content}</p>
   761→                    </div>
   762→                  </div>
   763→                ) : (
   764→                  <AssistantMessage
   765→                    key={m.id}
   766→                    m={m}
   767→                    researching={researching}
   768→                    runningTool={runningTool}
   769→                    researchQueries={researchQueries}
   770→                    streaming={streaming}
   771→                    isLatest={m.id === lastAssistantId}
   772→                    onRegenerate={() => void regenerateFrom(m.id)}
   773→                  />
   774→                ),
   775→              )}
   776→              <div ref={endRef} />
   777→            </div>
   778→          )}
   779→        </div>
   780→
   781→        {error ? (
   782→          <div className="mx-5 mb-2 rounded-md border border-red-500/30 bg-red-500/10 px-3 py-2 text-xs text-red-300">{error}</div>
   783→        ) : null}
   784→
   785→        {/* Attachments */}
   786→        {documents.length > 0 ? (
   787→          <div className="mx-auto flex w-full max-w-3xl flex-wrap gap-2 px-4 pb-1">

   789→              <span key={d.id} className="inline-flex items-center gap-1.5 rounded-full border border-[rgba(212,175,55,0.3)] bg-[rgba(212,175,55,0.08)] px-2.5 py-1 text-[11px] text-[#f0c75e]">
   790→                <FileText className="h-3 w-3" />
   791→                {d.name}
   792→                {typeof d.characters === 'number' ? <span className="text-white/35">· {d.characters.toLocaleString()} chars</span> : null}
   793→                {d.truncated ? <span className="text-amber-300">· truncated</span> : null}
   794→                <button
   795→                  onClick={() => setDocuments((prev) => prev.filter((x) => x.id !== d.id))}
   796→                  className="ml-0.5 text-white/40 hover:text-white/80"
   797→                  aria-label={`Remove ${d.name}`}
   798→                >
   799→                  <X className="h-3 w-3" />
   800→                </button>
   801→              </span>
   802→            ))}
   803→          </div>
   804→        ) : null}
   805→
   806→        <div className="border-t border-[rgba(212,175,55,0.1)] p-4">
   807→          <div className="mx-auto max-w-3xl">
   808→            {/* Controls row: search mode, attach, sandbox opt-in */}
   809→            <div className="mb-2 flex flex-wrap items-center gap-2">
   810→              <div className="inline-flex overflow-hidden rounded-lg border border-[rgba(212,175,55,0.25)]" role="group" aria-label="Web search mode">
   811→                {SEARCH_MODES.map((m) => (
   812→                  <button
   813→                    key={m.id}
   814→                    onClick={() => setSearchMode(m.id)}
   815→                    title={m.title}
   816→                    className={cn(
   817→                      'px-2.5 py-1 text-[11px] transition-colors',
   818→                      searchMode === m.id
   819→                        ? 'bg-[rgba(212,175,55,0.2)] text-[#f0c75e]'
   820→                        : 'bg-transparent text-white/45 hover:bg-white/5 hover:text-white/70',
   821→                    )}
   822→                  >
   823→                    <Globe className="mr-1 inline h-3 w-3" />

   830→                disabled={uploadingDoc || documents.length >= 2}
   831→                className="inline-flex items-center gap-1.5 rounded-lg border border-[rgba(212,175,55,0.25)] px-2.5 py-1 text-[11px] text-white/55 transition-colors hover:bg-white/5 hover:text-white/80 disabled:opacity-50"
   832→                title="Attach a document (txt, md, csv, json, pdf — max 10 MB, up to 2)"

   835→                Attach document
   836→              </button>
   837→              <input
   838→                ref={fileInputRef}
   839→                type="file"
   840→                accept=".txt,.md,.csv,.json,.log,.pdf"
   841→                className="hidden"
   842→                onChange={(e) => {
   843→                  const f = e.target.files?.[0]
   844→                  if (f) void uploadDocument(f)
   845→                }}
   846→              />
   847→              <button
   848→                onClick={() => setAllowCodeExecution((v) => !v)}
   849→                title="Allow RIISHE to run short JavaScript/TypeScript snippets in the restricted sandbox for this conversation"

   852→                  allowCodeExecution

   854→                    : 'border-[rgba(212,175,55,0.25)] text-white/55 hover:bg-white/5 hover:text-white/80',
   855→                )}
   856→                aria-pressed={allowCodeExecution}

   858→                <Terminal className="h-3 w-3" />
   859→                Code sandbox {allowCodeExecution ? 'on' : 'off'}
   860→              </button>
   861→            </div>
   862→
   863→            <div className="flex items-end gap-2">
   864→              <Textarea
   865→                value={input}
   866→                onChange={(e) => setInput(e.target.value)}
   867→                onKeyDown={(e) => {
   868→                  if (e.key === 'Enter' && !e.shiftKey) { e.preventDefault(); void send() }
   869→                }}
   870→                placeholder="Message RIISHE… (Enter to send, Shift+Enter for a new line)"
   871→                rows={Math.min(4, Math.max(1, input.split('\n').length))}
   872→                maxLength={12000}
   873→                disabled={streaming}
   874→                className="max-h-40 min-h-[46px] resize-none bg-black/30"
   875→                aria-label="Message RIISHE"
   876→              />
   877→              {streaming ? (
   878→                <Button onClick={stopGeneration} className="h-[46px] bg-[rgba(212,175,55,0.15)] px-4 text-[#f0c75e] hover:bg-[rgba(212,175,55,0.25)]" variant="outline" aria-label="Stop generating">
   879→                  <Square className="h-4 w-4" />
   880→                </Button>
   881→              ) : (
   882→                <Button onClick={() => void send()} disabled={!input.trim()} className="h-[46px] bg-[#d4af37] px-4 text-black hover:bg-[#e6c25a]" aria-label="Send message">
   883→                  <Send className="h-4 w-4" />
   884→                </Button>
   885→              )}
   886→            </div>
   887→          </div>
   888→        </div>
   889→      </div>
   890→    </div>
   891→  )

   894→/**
   895→ * One assistant message: markdown with citation chips, evidence panel,
   896→ * compact sources and the Copy/Regenerate/View-sources action row. The
   897→ * action row renders for EVERY assistant response (short or long), in every
   898→ * mode — never only for the latest one.
   899→ */
   900→function AssistantMessage({
   901→  m,
   902→  researching,
   903→  runningTool,
   904→  researchQueries,
   905→  streaming,
   906→  isLatest,
   907→  onRegenerate,
   908→}: {
   909→  m: Msg
   910→  researching: boolean
   911→  runningTool: string | null
   912→  researchQueries: string[]
   913→  streaming: boolean
   914→  isLatest: boolean
   915→  onRegenerate: () => void
   916→}) {
   917→  const meta = m.meta
   918→  const [previewId, setPreviewId] = useState<number | null>(null)
   919→  const [sourcesOpen, setSourcesOpen] = useState(false)
   920→  const [highlightId, setHighlightId] = useState<number | null>(null)
   921→  const entryRefs = useRef(new Map<number, HTMLLIElement | null>())
   922→
   923→  const registerRef = useCallback((id: number, el: HTMLLIElement | null) => {
   924→    entryRefs.current.set(id, el)
   925→  }, [])
   926→
   927→  // Citation chip click: open the Sources section and spotlight the entry.
   928→  const jumpToSource = useCallback((id: number) => {
   929→    setSourcesOpen(true)
   930→    setHighlightId(id)
   931→    window.setTimeout(() => {
   932→      entryRefs.current.get(id)?.scrollIntoView({ behavior: 'smooth', block: 'center' })
   933→    }, 80)
   934→    window.setTimeout(() => setHighlightId((cur) => (cur === id ? null : cur)), 2400)
   935→  }, [])
   936→
   937→  const sources = meta?.sources ?? []
   938→  const hasSources = sources.length > 0
   939→
   940→  const mdComponents = {
   941→    a: ({ href, children }: { href?: string; children?: React.ReactNode }) => {
   942→      if (href?.startsWith('#bean-src-')) {
   943→        const id = Number(href.replace('#bean-src-', ''))
   944→        const source = sources.find((s) => s.id === id)
   945→        if (!source) return <span className="text-white/60">{children}</span>
   946→        const label = String(children ?? '').replace(/[()]/g, '')
   947→        return (
   948→          <CitationLink
   949→            id={id}
   950→            label={label}
   951→            source={source}
   952→            previewed={previewId === id}
   953→            onPreview={setPreviewId}
   954→            onLeave={() => setPreviewId(null)}
   955→            onJump={jumpToSource}
   956→          />
   957→        )
   958→      }
   959→      return (
   960→        <a href={href} target="_blank" rel="noopener noreferrer" className="text-[#f0c75e] underline decoration-[rgba(212,175,55,0.4)] hover:decoration-[#f0c75e]">
   961→          {children}
   962→        </a>
   963→      )
   964→    },
   965→    pre: ({ children }: { children?: React.ReactNode }) => {
   966→      // React-markdown nests <code> inside <pre>; extract for highlighting.
   967→      const child = Array.isArray(children) ? children[0] : children
   968→      const props = (child as { props?: { className?: string; children?: ReactNode } })?.props
   969→      const lang = props?.className?.replace(/language-/, '') || undefined
   970→      const code = String(props?.children ?? '').replace(/\n$/, '')
   971→      if (code === '') return <pre>{children}</pre>
   972→      return <CodeBlock code={code} lang={lang} />
   973→    },
   974→  }
   975→
   976→  return (
   977→    <div className="flex justify-start">
   978→      <div className="bean-glass max-w-[92%] rounded-2xl px-4 py-3 text-white/85 sm:max-w-[85%]">
   979→        {m.content ? (
   980→          <>
   981→            <Markdown content={renderNamedCitations(m.content, sources)} components={mdComponents} />
   982→            {m.pending ? <Spinner className="ml-1 mt-1" /> : null}
   983→            {m.partial ? (
   984→              <div className="mt-2 inline-flex items-center gap-1.5 rounded border border-amber-500/30 bg-amber-500/10 px-2 py-0.5 text-[10px] text-amber-300">
   985→                <AlertTriangle className="h-3 w-3" /> Incomplete — the stream was stopped or ended early
   986→              </div>
   987→            ) : null}
   988→            {meta ? <AnswerEvidence meta={meta} /> : null}
   989→            {hasSources && meta ? (
   990→              <SourcesSection
   991→                sources={sources}
   992→                open={sourcesOpen}
   993→                onOpenChange={setSourcesOpen}
   994→                highlightId={highlightId}
   995→                registerRef={registerRef}
   996→              />
   997→            ) : null}
   998→            {!m.pending ? (
   999→              <MessageActions
  1000→                copyText={stripCitationMarkers(m.content)}
  1001→                disabled={streaming}
  1002→                hasSources={hasSources}
  1003→                sourcesOpen={sourcesOpen}
