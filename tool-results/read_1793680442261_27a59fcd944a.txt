    74→function langOf(path: string): string {
    75→  const ext = path.split('.').pop()?.toLowerCase() ?? ''
    76→  const map: Record<string, string> = {
    77→    js: 'javascript', jsx: 'javascript', mjs: 'javascript',
    78→    ts: 'typescript', tsx: 'typescript',
    79→    py: 'python', html: 'html', css: 'css', json: 'json', md: 'markdown', sql: 'sql',
    80→  }
    81→  return map[ext] ?? 'text'
    82→}

    84→function formatBytes(n: number): string {
    85→  if (n < 1024) return `${n} B`
    86→  if (n < 1024 * 1024) return `${(n / 1024).toFixed(1)} KB`
    87→  return `${(n / (1024 * 1024)).toFixed(1)} MB`
    88→}

   256→  async function send(overrideMessage?: string) {
   257→    const message = (overrideMessage ?? input).trim()
   258→    if (!message || streaming) return
   259→    if (overrideMessage === undefined) setInput('')
   260→    setError(null)
   261→    setStreaming(true)
   262→    setThinking(true)
   263→    streamedRef.current = ''
   264→
   265→    const tempId = `u-${Date.now()}`
   266→    const asstId = `a-${Date.now()}`
   267→    setMessages((prev) => [...prev, { id: tempId, role: 'user', content: message, changes: [] }, { id: asstId, role: 'assistant', content: '', changes: [] }])
   268→
   269→    const controller = new AbortController()
   270→    abortRef.current = controller
   271→    const readyFiles = attachments.filter((f) => f.status === 'ready')
   272→
   273→    try {
   274→      const res = await fetch(`/api/code-giftz/chat?stream=1${projectId ? `&projectId=${projectId}` : ''}`, {
   275→        method: 'POST',
   276→        headers: { 'content-type': 'application/json' },
   277→        body: JSON.stringify({
   278→          message,
   279→          projectId: projectId ?? undefined,
   280→          fileIds: readyFiles.length > 0 ? readyFiles.map((f) => f.id) : undefined,
   281→        }),
   282→        signal: controller.signal,
   283→      })
   284→      if (!res.ok || !res.body) {
   285→        if (res.status === 401) window.dispatchEvent(new Event('bean:unauthorized'))
   286→        const data = await res.json().catch(() => ({}))
   287→        throw new Error(data?.error?.message ?? `Request failed (${res.status})`)
   288→      }
   289→      const reader = res.body.getReader()
   290→      const decoder = new TextDecoder()
   291→      let buffer = ''
   292→      let newProjectId: string | null = projectId
   293→      let changes: FileChange[] = []
   294→
   295→      while (true) {
   296→        const { done, value } = await reader.read()
   297→        if (done) break
   298→        buffer += decoder.decode(value, { stream: true })
   299→        const lines = buffer.split('\n')
   300→        buffer = lines.pop() ?? ''
   301→        for (const line of lines) {
   302→          if (!line.startsWith('data: ')) continue
   303→          let evt: { delta?: string; stage?: string; done?: boolean; error?: string; projectId?: string; changes?: FileChange[]; files?: Array<{ path: string; origin: string }> }
   304→          try { evt = JSON.parse(line.slice(6)) } catch { continue }
   305→          if (evt.projectId) newProjectId = evt.projectId
   306→          if (evt.stage === 'thinking') setThinking(true)
   307→          if (evt.error) setError(evt.error)
   308→          if (evt.delta) {
   309→            streamedRef.current += evt.delta
   310→            setThinking(false)
   311→            setMessages((prev) => prev.map((m) => (m.id === asstId ? { ...m, content: streamedRef.current } : m)))
   312→          }
   313→          if (evt.done) {
   314→            changes = evt.changes ?? []
   315→            setThinking(false)
   316→            setLastChanges(changes)
   317→            if (evt.files) {
   318→              // Refresh file contents from the server (source of truth).
   319→              setFiles((prev) => {
   320→                const next = evt.files!
   321→                return next.map((n) => {
   322→                  const known = prev.find((p) => p.path === n.path)
   323→                  return { path: n.path, content: known?.content ?? '', origin: n.origin as ProjectFile['origin'] }
   324→                })
   325→              })
   326→              // Fetch fresh contents for accuracy.
   327→              void (async () => {
   328→                try {
   329→                  const d = await beanFetch<{ project: { files: ProjectFile[] } }>(`/api/code-giftz/chat/${newProjectId}`)
   330→                  setFiles(d.project.files)
   331→                  setOpenTabs((prev) => {
   332→                    const merged = [...prev]
   333→                    for (const f of d.project.files) if (!merged.includes(f.path)) merged.push(f.path)
   334→                    return merged
   335→                  })
   336→                  if (!activeFile && d.project.files.length > 0) setActiveFile(d.project.files[0].path)
   337→                } catch { /* non-fatal */ }
   338→              })()
   339→            }
   340→            setMessages((prev) => prev.map((m) => (m.id === asstId ? { ...m, changes: changes.map((c) => ({ path: c.path, change: c.change })) } : m)))
   341→            // Auto-open the workspace on real changes.
   342→            if (changes.length > 0) {
   343→              setShowPreview((v) => v)
   344→              const firstChanged = changes[0].path
   345→              setOpenTabs((prev) => (prev.includes(firstChanged) ? prev : [...prev, firstChanged]))
   346→              setActiveFile(firstChanged)
   347→              setFileView('diff')
   348→              setEditing(false)
   349→            }
   350→          }
   351→        }
   352→      }
   353→      if (streamedRef.current === '') {
   354→        setMessages((prev) => prev.filter((m) => m.id !== asstId))
   355→      }
   356→      if (newProjectId && !projectId) setProjectId(newProjectId)
   357→      void loadProjects()
   358→    } catch (e) {
   359→      if ((e as Error).name === 'AbortError') {
   360→        if (streamedRef.current !== '') {
   361→          setMessages((prev) => prev.map((m) => (m.id === asstId ? { ...m, content: streamedRef.current } : m)))
   362→        } else {
   363→          setMessages((prev) => prev.filter((m) => m.id !== asstId))
   364→        }
   365→      } else {
   366→        setError((e as Error).message)
   367→        setMessages((prev) => prev.filter((m) => m.id !== asstId || m.content.length > 0))
   368→      }
   369→    } finally {
   370→      setStreaming(false)
   371→      setThinking(false)
   372→      abortRef.current = null
   373→    }
   374→  }
