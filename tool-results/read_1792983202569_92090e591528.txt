   118→function useCopy(): [boolean, (text: string) => void] {
   119→  const [copied, setCopied] = useState(false)
   120→  const copy = useCallback((text: string) => {
   121→    const done = () => {
   122→      setCopied(true)
   123→      setTimeout(() => setCopied(false), 1600)
   124→    }
   125→    if (navigator.clipboard?.writeText) {
   126→      navigator.clipboard.writeText(text).then(done).catch(done)
   127→    } else {
   128→      const ta = document.createElement('textarea')
   129→      ta.value = text
   130→      document.body.appendChild(ta)
   131→      ta.select()
   132→      document.execCommand('copy')
   133→      document.body.removeChild(ta)
   134→      done()
   135→    }
   136→  }, [])
   137→  return [copied, copy]
   138→}

   484→  /**
   485→   * Shared SSE reader for a single assistant turn (send + regenerate).
   486→   * Streams into the pending bubble `asstId` and finalizes honestly.

   488→  async function streamChat(opts: {
   489→    message: string
   490→    asstId: string
   491→    searchMode: SearchMode
   492→    retry?: boolean
   493→    regenerateMessageId?: string
   494→    /** Local id of the just-added user bubble; replaced by the real server id. */
   495→    localUserMsgId?: string
   496→  }) {
   497→    const convAtStart = activeId
   498→    const asstId = opts.asstId
   499→    const controller = new AbortController()
   500→    abortRef.current = controller
   501→
   502→    try {
   503→      const res = await fetch(`/api/riishe/chat?stream=1${convAtStart ? `&conversationId=${convAtStart}` : ''}`, {
   504→        method: 'POST',
   505→        headers: { 'content-type': 'application/json' },
   506→        body: JSON.stringify({
   507→          message: opts.message,
   508→          conversationId: convAtStart ?? undefined,
   509→          searchMode: opts.searchMode,
   510→          allowCodeExecution,
   511→          documentIds: documents.length > 0 ? documents.map((d) => d.id) : undefined,
   512→          retry: opts.retry ?? false,
   513→          regenerateMessageId: opts.regenerateMessageId,
   514→        }),
   515→        signal: controller.signal,
   516→      })
   517→      if (!res.ok || !res.body) {
   518→        // Keep the global identity in sync: a 401 here means the session is
   519→        // gone server-side; drop the signed-in shell immediately.
   520→        if (res.status === 401) window.dispatchEvent(new Event('bean:unauthorized'))
   521→        const data = await res.json().catch(() => ({}))
   522→        throw new Error(data?.error?.message ?? `Request failed (${res.status})`)
   523→      }
   524→      const reader = res.body.getReader()
   525→      const decoder = new TextDecoder()
   526→      let buffer = ''
   527→      let convId: string | null = convAtStart
   528→      let failed = false
   529→
   530→      while (true) {
   531→        const { done, value } = await reader.read()
   532→        if (done) break
   533→        buffer += decoder.decode(value, { stream: true })
   534→        const lines = buffer.split('\n')
   535→        buffer = lines.pop() ?? ''
   536→        for (const line of lines) {
   537→          if (!line.startsWith('data: ')) continue
   538→          let evt: {
   539→            delta?: string
   540→            done?: boolean
   541→            error?: string
   542→            conversationId?: string
   543→            partial?: boolean
   544→            userMessageId?: string
   545→            messageId?: string | null
   546→            stage?: 'researching' | 'tools' | 'solving' | 'verifying'
   547→            research?: { queries?: string[] }
   548→            tool?: { name: string; status: 'started' | 'done' | 'failed' }
   549→            meta?: MsgMeta
   550→          }
   551→          try { evt = JSON.parse(line.slice(6)) } catch { continue }
   552→          if (evt.conversationId) convId = evt.conversationId
   553→          if (evt.userMessageId && opts.localUserMsgId) {
   554→            // Adopt the real persisted id for the user turn so per-message
   555→            // actions address server rows.
   556→            const serverId = evt.userMessageId
   557→            const localId = opts.localUserMsgId
   558→            setMessages((prev) => prev.map((m) => (m.id === localId ? { ...m, id: serverId } : m)))
   559→          }
   560→          if (evt.stage === 'researching') { setResearching(true); setResearchQueries([]) }
   561→          if (evt.stage === 'tools') { setResearching(false); setRunningTool('working') }
   562→          if (evt.stage === 'solving' || evt.stage === 'verifying') { setResearching(false); setRunningTool(null) }
   563→          if (evt.tool) {
   564→            if (evt.tool.status === 'started') setRunningTool(evt.tool.name)
   565→            else setRunningTool(null)
   566→          }
   567→          if (evt.research?.queries) setResearchQueries(evt.research.queries)
   568→          if (evt.error) { failed = true; setError(evt.error) }
   569→          if (evt.delta) {
   570→            streamedRef.current += evt.delta
   571→            setMessages((prev) =>
   572→              prev.map((m) => (m.id === asstId ? { ...m, content: streamedRef.current, pending: false } : m)),
   573→            )
   574→          }
   575→          if (evt.done) {
   576→            setResearching(false)
   577→            setRunningTool(null)
   578→            if (convId && !convAtStart) setActiveId(convId)
   579→            const serverAsstId = evt.messageId ?? null
   580→            setMessages((prev) =>
   581→              prev.map((m) =>
   582→                m.id === asstId ? { ...m, id: serverAsstId ?? m.id, meta: evt.meta ?? m.meta } : m,
   583→              ),
   584→            )
   585→          }
   586→        }
   587→      }
   588→      // If nothing streamed (failed, or the server reported an empty partial),
   589→      // drop the empty assistant bubble — the error banner already explains
   590→      // what happened, and a perpetual "thinking" indicator would be a lie.
   591→      if (streamedRef.current === '') {
   592→        setMessages((prev) => prev.filter((m) => m.id !== asstId))
   593→      } else if (failed) {
   594→        setMessages((prev) => prev.map((m) => (m.id === asstId ? { ...m, partial: true, pending: false } : m)))
   595→      }
   596→      void loadConversations()
   597→    } catch (e) {
   598→      if ((e as Error).name === 'AbortError') {
   599→        // User pressed stop: keep what arrived and label it honestly.
   600→        if (streamedRef.current !== '') {
   601→          setMessages((prev) => prev.map((m) => (m.id === asstId ? { ...m, partial: true, pending: false } : m)))
   602→        } else {
   603→          setMessages((prev) => prev.filter((m) => m.id !== asstId))
   604→        }
   605→      } else {
   606→        setError((e as Error).message)
   607→        setMessages((prev) => prev.filter((m) => m.id !== asstId || m.content.length > 0))
   608→      }
   609→    } finally {
   610→      setStreaming(false)
   611→      setResearching(false)
   612→      setRunningTool(null)
   613→      abortRef.current = null
   614→      setMessages((prev) => prev.map((m) => ({ ...m, pending: false })))
   615→    }
   616→  }

   618→  async function send(overrideMessage?: string, retry = false) {
   619→    const message = (overrideMessage ?? input).trim()
   620→    if (!message || streaming) return
   621→    if (overrideMessage === undefined) setInput('')

   623→    setStreaming(true)
   624→    setResearching(false)
   625→    setRunningTool(null)
   626→    setResearchQueries([])
   627→    streamedRef.current = ''
   628→
   629→    const userMsg: Msg = { id: `u-${Date.now()}`, role: 'user', content: message }
   630→    const asstId = `a-${Date.now()}`
   631→    setMessages((prev) => [...prev, userMsg, { id: asstId, role: 'assistant', content: '', pending: true }])
   632→    await streamChat({ message, asstId, searchMode, retry, localUserMsgId: userMsg.id })
   633→  }

   635→  /**
   636→   * Regenerate ANY assistant response in place: the old answer (and every
   637→   * message after it) is discarded locally and server-side; the original
   638→   * question is re-answered with the mode recorded for THAT answer, so the
   639→   * regeneration reflects the original context and retrieval choice. While
   640→   * a stream is running the action is disabled — no duplicate requests.
   641→   */
   642→  async function regenerateFrom(messageId: string) {

   644→    const idx = messages.findIndex((m) => m.id === messageId)

   646→    const target = messages[idx]
   647→    if (target.role !== 'assistant' || target.pending) return
   648→    const userBefore = idx > 0 ? messages[idx - 1] : null
   649→    if (!userBefore || userBefore.role !== 'user') return
   650→    const mode: SearchMode = target.meta?.searchMode ?? searchMode
   651→
   652→    setError(null)
   653→    setStreaming(true)
   654→    setResearching(false)
   655→    setRunningTool(null)
   656→    setResearchQueries([])
   657→    streamedRef.current = ''
   658→    const asstId = `a-${Date.now()}`
   659→    setMessages((prev) => [
   660→      ...prev.slice(0, idx),
   661→      { id: asstId, role: 'assistant' as const, content: '', pending: true },
   662→    ])
   663→    await streamChat({ message: userBefore.content, asstId, searchMode: mode, regenerateMessageId: messageId })
   664→  }
   665→
   666→  const lastAssistantId = [...messages].reverse().find((m) => m.role === 'assistant' && !m.pending)?.id
   667→
   668→  return (
   669→    <div className="flex h-[calc(100vh-3.5rem-2rem)] min-h-[540px] lg:h-[calc(100vh-3.5rem-1.5rem)]">
   670→      {/* Conversation rail */}
   671→      <aside className="hidden w-60 shrink-0 flex-col border-r border-[rgba(212,175,55,0.1)] md:flex">
   672→        <div className="p-3">
   673→          <Button onClick={newConversation} className="w-full justify-start bg-[rgba(212,175,55,0.12)] text-[#f0c75e] hover:bg-[rgba(212,175,55,0.2)]" variant="outline">
   674→            <Plus className="mr-1.5 h-4 w-4" /> New conversation
   675→          </Button>
   676→        </div>
   677→        <ScrollArea className="flex-1 px-2 pb-3">
   678→          {conversations.length === 0 ? (
   679→            <p className="px-2 py-6 text-center text-xs text-white/30">No conversations yet</p>
   680→          ) : (
   681→            conversations.map((c) => (
   682→              <div
   683→                key={c.id}
   684→                className={cn(
   685→                  'group mb-0.5 flex cursor-pointer items-center justify-between rounded-lg px-3 py-2 text-[13px] transition-colors',
   686→                  activeId === c.id ? 'bg-[rgba(212,175,55,0.1)] text-[#f0c75e]' : 'text-white/60 hover:bg-white/5',
   687→                )}
   688→                onClick={() => openConversation(c.id)}
   689→              >
   690→                <div className="min-w-0">
   691→                  <div className="truncate">{c.title}</div>
   692→                  <div className="text-[10px] text-white/30">{c.messages} msgs · {timeAgo(c.updatedAt)}</div>
   693→                </div>
   694→                <button
   695→                  className="ml-2 hidden text-white/30 hover:text-red-400 group-hover:block"
   696→                  onClick={(e) => { e.stopPropagation(); void deleteConversation(c.id) }}
   697→                  aria-label={`Delete ${c.title}`}
   698→                >
   699→                  <Trash2 className="h-3.5 w-3.5" />
   700→                </button>
   701→              </div>
   702→            ))
   703→          )}
   704→        </ScrollArea>
   705→      </aside>
   706→
   707→      {/* Chat column */}
   708→      <div className="flex min-w-0 flex-1 flex-col">
   709→        <div className="flex items-center gap-2 border-b border-[rgba(212,175,55,0.1)] px-5 py-3">
   710→          <MessageSquareText className="h-4 w-4 text-[#d4af37]" />
   711→          <div className="text-sm font-semibold tracking-wide">RIISHE</div>
   712→          <span className="text-xs text-white/35">Conversational AI · research · verified reasoning</span>
   713→        </div>
   714→
   715→        <div className="bean-scroll flex-1 overflow-y-auto px-5 py-4">
   716→          {messages.length === 0 ? (
   717→            <EmptyState
   718→              icon={<MessageSquareText className="h-10 w-10" />}
   719→              title="Start a conversation"
   720→              hint="Ask questions, request derivations, code help or research on current topics. RIISHE searches live sources when a question needs them, runs real tools, and verifies its arithmetic before answering. Attach a document for grounded analysis."
   721→            />
   722→          ) : (
   723→            <div className="mx-auto max-w-3xl space-y-4">
   724→              {messages.map((m) =>
   725→                m.role === 'user' ? (
   726→                  <div key={m.id} className="flex justify-end">
   727→                    <div className="max-w-[85%] rounded-2xl bg-[rgba(212,175,55,0.14)] px-4 py-3 text-white/90">
   728→                      <p className="whitespace-pre-wrap text-[14.5px] leading-relaxed">{m.content}</p>
   729→                    </div>
   730→                  </div>
   731→                ) : (
   732→                  <AssistantMessage
   733→                    key={m.id}
   734→                    m={m}
   735→                    researching={researching}
   736→                    runningTool={runningTool}
   737→                    researchQueries={researchQueries}
   738→                    streaming={streaming}
   739→                    isLatest={m.id === lastAssistantId}
   740→                    onRegenerate={() => void regenerateFrom(m.id)}
   741→                  />
   742→                ),
   743→              )}
   744→              <div ref={endRef} />
   745→            </div>
   746→          )}
   747→        </div>
   748→
   749→        {error ? (
   750→          <div className="mx-5 mb-2 rounded-md border border-red-500/30 bg-red-500/10 px-3 py-2 text-xs text-red-300">{error}</div>
   751→        ) : null}

   758→                <FileText className="h-3 w-3" />
   759→                {d.name}
   760→                {typeof d.characters === 'number' ? <span className="text-white/35">· {d.characters.toLocaleString()} chars</span> : null}
   761→                {d.truncated ? <span className="text-amber-300">· truncated</span> : null}
   762→                <button
   763→                  onClick={() => setDocuments((prev) => prev.filter((x) => x.id !== d.id))}
   764→                  className="ml-0.5 text-white/40 hover:text-white/80"
   765→                  aria-label={`Remove ${d.name}`}
   766→                >
   767→                  <X className="h-3 w-3" />

   772→        ) : null}
   773→
   774→        <div className="border-t border-[rgba(212,175,55,0.1)] p-4">
   775→          <div className="mx-auto max-w-3xl">
   776→            {/* Controls row: search mode, attach, sandbox opt-in */}
   777→            <div className="mb-2 flex flex-wrap items-center gap-2">

   779→                {SEARCH_MODES.map((m) => (

   847→                  <Square className="h-4 w-4" />

   862→/**
   863→ * One assistant message: markdown with citation chips, evidence panel,
   864→ * compact sources and the Copy/Regenerate/View-sources action row. The
   865→ * action row renders for EVERY assistant response (short or long), in every
   866→ * mode — never only for the latest one.
   867→ */
   868→function AssistantMessage({
   869→  m,
   870→  researching,
   871→  runningTool,
   872→  researchQueries,
   873→  streaming,
   874→  isLatest,
   875→  onRegenerate,
   876→}: {
   877→  m: Msg
   878→  researching: boolean
   879→  runningTool: string | null
   880→  researchQueries: string[]
   881→  streaming: boolean
   882→  isLatest: boolean
   883→  onRegenerate: () => void
   884→}) {
   885→  const meta = m.meta
   886→  const [previewId, setPreviewId] = useState<number | null>(null)
   887→  const [sourcesOpen, setSourcesOpen] = useState(false)
   888→  const [highlightId, setHighlightId] = useState<number | null>(null)
   889→  const entryRefs = useRef(new Map<number, HTMLLIElement | null>())
   890→
   891→  const registerRef = useCallback((id: number, el: HTMLLIElement | null) => {
   892→    entryRefs.current.set(id, el)
   893→  }, [])
   894→
   895→  // Citation chip click: open the Sources section and spotlight the entry.
   896→  const jumpToSource = useCallback((id: number) => {
   897→    setSourcesOpen(true)
   898→    setHighlightId(id)
   899→    window.setTimeout(() => {
   900→      entryRefs.current.get(id)?.scrollIntoView({ behavior: 'smooth', block: 'center' })
   901→    }, 80)
   902→    window.setTimeout(() => setHighlightId((cur) => (cur === id ? null : cur)), 2400)
   903→  }, [])
   904→
   905→  const sources = meta?.sources ?? []
   906→  const hasSources = sources.length > 0
   907→
   908→  const mdComponents = {
   909→    a: ({ href, children }: { href?: string; children?: React.ReactNode }) => {
   910→      if (href?.startsWith('#bean-src-')) {
   911→        const id = Number(href.replace('#bean-src-', ''))
   912→        const source = sources.find((s) => s.id === id)
   913→        if (!source) return <span className="text-white/60">{children}</span>
   914→        return (
   915→          <CitationChip
   916→            id={id}
   917→            source={source}
   918→            previewed={previewId === id}
   919→            onPreview={setPreviewId}
   920→            onLeave={() => setPreviewId(null)}
   921→            onJump={jumpToSource}
   922→          />
   923→        )
   924→      }
   925→      return (
   926→        <a href={href} target="_blank" rel="noopener noreferrer" className="text-[#f0c75e] underline decoration-[rgba(212,175,55,0.4)] hover:decoration-[#f0c75e]">
   927→          {children}
   928→        </a>
   929→      )
   930→    },
   931→    pre: ({ children }: { children?: React.ReactNode }) => {
   932→      // React-markdown nests <code> inside <pre>; extract for highlighting.
   933→      const child = Array.isArray(children) ? children[0] : children
   934→      const props = (child as { props?: { className?: string; children?: ReactNode } })?.props
   935→      const lang = props?.className?.replace(/language-/, '') || undefined
   936→      const code = String(props?.children ?? '').replace(/\n$/, '')
   937→      if (code === '') return <pre>{children}</pre>
   938→      return <CodeBlock code={code} lang={lang} />
   939→    },
   940→  }
   941→
   942→  return (
   943→    <div className="flex justify-start">
   944→      <div className="bean-glass max-w-[92%] rounded-2xl px-4 py-3 text-white/85 sm:max-w-[85%]">
   945→        {m.content ? (
   946→          <>
   947→            <Markdown content={linkifyCitations(m.content, sources)} components={mdComponents} />
   948→            {m.pending ? <Spinner className="ml-1 mt-1" /> : null}
   949→            {m.partial ? (
   950→              <div className="mt-2 inline-flex items-center gap-1.5 rounded border border-amber-500/30 bg-amber-500/10 px-2 py-0.5 text-[10px] text-amber-300">
   951→                <AlertTriangle className="h-3 w-3" /> Incomplete — the stream was stopped or ended early
   952→              </div>
   953→            ) : null}
   954→            {meta ? <AnswerEvidence meta={meta} /> : null}
   955→            {hasSources && meta ? (
   956→              <SourcesSection
   957→                sources={sources}
   958→                open={sourcesOpen}
   959→                onOpenChange={setSourcesOpen}
   960→                highlightId={highlightId}
   961→                registerRef={registerRef}
   962→              />
   963→            ) : null}
   964→            {!m.pending ? (
   965→              <MessageActions
   966→                copyText={stripCitationMarkers(m.content)}
   967→                disabled={streaming}
   968→                hasSources={hasSources}
   969→                sourcesOpen={sourcesOpen}
   970→                onToggleSources={() => setSourcesOpen((v) => !v)}
   971→                onRegenerate={onRegenerate}
   972→                modeLabel={meta?.searchMode}
   973→              />
   974→            ) : null}
   975→          </>
   976→        ) : researching && isLatest ? (
   977→          <span className="inline-flex items-center gap-2 text-xs text-white/50">
   978→            <Globe className="h-3.5 w-3.5 animate-pulse text-[#d4af37]" /> Searching the web…
   979→            {researchQueries.length > 0 ? (
   980→              <span className="text-white/30">({researchQueries.slice(0, 2).join(' · ')})</span>
   981→            ) : null}
   982→          </span>
   983→        ) : runningTool && isLatest ? (
   984→          <span className="inline-flex items-center gap-2 text-xs text-white/50">
   985→            <Terminal className="h-3.5 w-3.5 animate-pulse text-[#d4af37]" />
   986→            {runningTool === 'working' ? 'Running server-side tools…' : `Running ${runningTool}…`}
   987→          </span>
   988→        ) : (
   989→          <span className="inline-flex items-center gap-2 text-xs text-white/40">
   990→            <Spinner /> RIISHE is thinking…
