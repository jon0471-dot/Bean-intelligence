    80→    80→  'You generate concise web search queries for a fact-checking pipeline. ' +
    81→    81→  'Given a user question, output 1-3 search queries that would surface authoritative, current sources. ' +

    84→    84→/**
    85→    85→ * Straightforward-question detector: when the message is a short,
    86→    86→ * single-question lookup, the user's own wording IS the strong query and the
    87→    87→ * query-refinement model call is pure added latency. Complex, compound or
    88→    88→ * long questions still get refinement.
    89→    89→ */
    90→    90→export function isStraightforwardQuestion(message: string): boolean {
    91→    91→  const text = message.trim()
    92→    92→  if (text.length === 0 || text.length > 140) return false
    93→    93→  // Multiple sentences (excluding a single trailing '?') suggest compound asks.
    94→    94→  const sentences = text.replace(/[?!.]+\s*$/, '').split(/[?!.]\s+/).filter((s) => s.trim().length > 0)
    95→    95→  if (sentences.length > 1) return false
    96→    96→  // Comparison openers are compound by nature ("Compare X and Y", "X vs Y").
    97→    97→  if (/^(compare|difference|differences|which\s+is\s+better)\b/i.test(text)) return false
    98→    98→  // Multi-part coordination: multiple "and"/"or", or a conjunction followed
    99→    99→  // by another question word ("X and what changed in Y").
   100→   100→  if (/\b(and|or)\b/i.test(text) && (text.match(/\b(and|or)\b/gi) ?? []).length > 1) return false
   101→   101→  if (/\b(and|or)\s+(what|how|why|when|who|where|which)\b/i.test(text)) return false
   102→   102→  return true
   103→   103→}

   107→   107→  const fallback = { queries: [truncateQuery(message)], usedFallback: true }

   119→   119→              'Respond with JSON only: {"queries": ["...", "..."]}',

   137→   137→function truncateQuery(message: string): string {
   138→   138→  const q = message.replace(/\s+/g, ' ').trim()
   139→   139→  return q.length <= 280 ? q : `${q.slice(0, 277)}...`
   140→   140→}
   141→   141→
   142→   142→/**
   143→   143→ * Strip directive scaffolding the user wrapped around a lookup so the search
   144→   144→ * provider receives the SUBJECT, not the imperative. Measured failure class
   145→   145→ * (worklog): "Research a current topic: Gingee girl missing" searched (and
   146→   146→ * would have searched) for the literal sentence including the prefix.
   147→   147→ * Applied ONLY on the fast path; complex questions get LLM refinement.
   148→   148→ */
   149→   149→export function toSearchQuery(message: string): string {
   150→   150→  let q = message.replace(/\s+/g, ' ').trim()
   151→   151→  for (let i = 0; i < 3; i += 1) {
   152→   152→    const next = q
   153→   153→      .replace(/^(?:please\s+)?(?:kindly\s+)?(?:research|search(?:\s+(?:the\s+web|the\s+internet|online))?(?:\s+for)?|google|look\s+up|lookup|find(?:\s+out)?|check(?:\s+online)?|browse)\b[\s:,-]*/i, '')
   154→   154→      .replace(/^(?:a\s+|an\s+|the\s+)?(?:current|recent|latest|breaking)\s+(?:topic|topics|news|story|stories|event|events|affairs?|developments?)\s*[:\-—]\s*/i, '')
   155→   155→      .replace(/\s*(?:right\s+now|as\s+of\s+(?:now|today)|these\s+days)\s*[?!.]*\s*$/i, '')
   156→   156→      .trim()
   157→   157→    if (next === q || next.length === 0) break
   158→   158→    q = next
   159→   159→  }
   160→   160→  return q.length > 0 ? (q.length <= 280 ? q : `${q.slice(0, 277)}...`) : truncateQuery(message)
   161→   161→}

   163→   163→/**
   164→   164→ * True when a freshness-sensitive (news) query deserves ONE bounded follow-up
   165→   165→ * search for newer reporting: every selected source is either undated or the
   166→   166→ * newest dated source is older than 45 days. Older reporting still ranks —
   167→   167→ * as background, never presented as the latest development.

   172→   172→  if (ages.length === 0) return true

   174→   174→}

   192→   192→/** Collapse near-duplicate queries (normalized word-set equality) before dispatch. */
   193→   193→export function dedupeQueries(queries: string[]): string[] {
   194→   194→  const seen = new Set<string>()
   195→   195→  const out: string[] = []
   196→   196→  for (const q of queries) {
   197→   197→    const key = q.toLowerCase().replace(/[^\p{L}\p{N}]+/gu, ' ').trim().split(' ').sort().join(' ')
   198→   198→    if (key.length === 0 || seen.has(key)) continue
   199→   199→    seen.add(key)
   200→   200→    out.push(q)
   201→   201→  }
   202→   202→  return out
   203→   203→}

   205→   205→/**
   206→   206→ * Sensitive-case deterministic query set (work order: prioritize official
   207→   207→ * police/government statements AND search for newer reporting before
   208→   208→ * summarizing an ongoing case). Added on top of the policy/LLM queries and
   209→   209→ * BEFORE dedupe so near-duplicates collapse; bounded so a sensitive turn
   210→   210→ * never fans out unbounded searches.
   211→   211→ *
   212→   212→ * Returns the (possibly re-anchored) query list plus the freshness query
   213→   213→ * actually appended (for the honest degraded note), or null.
   214→   214→ */
   215→   215→export function buildSensitiveQuerySet(
   216→   216→  message: string,
   217→   217→  queries: string[],
   218→   218→): { queries: string[]; addedFreshnessQuery: string | null; addedOfficialQuery: string | null } {
   219→   219→  const sensitive = detectSensitiveCase(message)
   220→   220→  if (!sensitive.sensitive) return { queries, addedFreshnessQuery: null, addedOfficialQuery: null }
   221→   221→  const core = coreTopic(message) ?? queries[0] ?? null
   222→   222→  if (!core) return { queries, addedFreshnessQuery: null, addedOfficialQuery: null }
   223→   223→  const next = [...queries]
   224→   224→  // Anchor the set to the core topic when the generated queries drifted.
   225→   225→  const coreKey = core.toLowerCase().slice(0, 24)
   226→   226→  if (!next.some((q) => q.toLowerCase().includes(coreKey))) {
   227→   227→    next.unshift(truncateQuery(core))
   228→   228→  }
   229→   229→  const lowerJoined = next.join(' \n ').toLowerCase()
   230→   230→  const freshnessQuery = truncateQuery(`${core} latest update`)
   231→   231→  const officialQuery = truncateQuery(`${core} police official statement`)
   232→   232→  let addedFreshnessQuery: string | null = null
   233→   233→  let addedOfficialQuery: string | null = null
   234→   234→  // Skip an extra when an existing query ALREADY targets that intent —
   235→   235→  // near-duplicate searches burn the (rate-limited) gateway for nothing.
   236→   236→  const targetsFreshness = /\b(latest|newest|update|updates|today|now)\b/i.test(lowerJoined)
   237→   237→  const targetsOfficial = /\b(police|official|government|statement|cb-cid|cbcid)\b/i.test(lowerJoined)
   238→   238→  if (!targetsFreshness) {
   239→   239→    next.push(freshnessQuery)
   240→   240→    addedFreshnessQuery = freshnessQuery
   241→   241→  }
   242→   242→  if (!targetsOfficial) {
   243→   243→    next.push(officialQuery)
   244→   244→    addedOfficialQuery = officialQuery
   245→   245→  }
   246→   246→  return { queries: next.slice(0, 5), addedFreshnessQuery, addedOfficialQuery }
   247→   247→}

   255→   255→/**
   256→   256→ * Decide how many top-ranked pages to read in full. Snippets that already
   257→   257→ * carry substantial grounded text make page retrieval pure added latency;
   258→   258→ * thin snippets (or high-risk questions where snippet over-interpretation is
   259→   259→ * likely) justify reading the source. Ongoing-case questions get one extra
   260→   260→ * read: identifying details (names, dates, statuses) in snippets are
   261→   261→ * frequently wrong or compressed — the work order requires verifying the
   262→   262→ * article's actual content, and for an evolving case the cost is bounded.
   263→   263→ */
   264→   264→export function decidePageReads(
   265→   265→  ranked: RankedSource[],
   266→   266→  decision: SearchDecision,
   267→   267→  opts: RunResearchOptions,
   268→   268→  ongoingCase = false,
   269→   269→): number {
   270→   270→  const configured =
   271→   271→    opts.readTopN ??
   272→   272→    (decision.risk === 'high' ? (ongoingCase ? 3 : 2) : decision.risk === 'moderate' ? 1 : 0)
   273→   273→  const count = Math.min(Math.max(0, configured), ranked.length)
   274→   274→  if (count === 0) return 0
   275→   275→  // Snippet sufficiency: if every page we WOULD read already has a
   276→   276→  // substantial snippet, skip page retrieval entirely.
   277→   277→  const SUBSTANTIAL_SNIPPET = 260
   278→   278→  const allSubstantial = ranked.slice(0, count).every((s) => s.snippet.length >= SUBSTANTIAL_SNIPPET)
   279→   279→  return allSubstantial ? 0 : count
   280→   280→}

   282→   282→/**
   283→   283→ * Backoff before the single bounded search retry. Measured evidence
   284→   284→ * (worklog): the provider intermittently rejects a query it serves seconds
   285→   285→ * later (HTTP 422 "No search results available" on a healthy index), so one
   286→   286→ * short-delay retry recovers grounding without unbounded calls.
   287→   287→ */
   288→   288→export const SEARCH_RETRY_BACKOFF_MS = 1_200
   289→   289→
   290→   290→/**
   291→   291→ * Wall-time budget for ONE full-page fetch. Measured evidence (baseline
   292→   292→ * perf run): the provider occasionally takes ~20s for a slow page while the
   293→   293→ * p50 is well under 4s — an unbounded tail dominated total user-perceived
   294→   294→ * latency for multi-source research (pageReadMs 19.9s of a 23.6s turn).
   295→   295→ * A read that exceeds the budget is discarded HONESTLY: it surfaces as a
   296→   296→ * degraded note and the answer is grounded on the snippets we already have.
   297→   297→ */
   298→   298→export const PAGE_READ_BUDGET_MS = 10_000
   299→   299→
   300→   300→/**
   301→   301→ * Page-read content validation. Measured failure (work order evidence): the
   302→   302→ * page reader returned Chrome's ERR_BLOCKED_BY_CLIENT interstitial for a
   303→   303→ * news site and a literal "404 Not Found" body for a YouTube playlist — both
   304→   304→ * were recorded as successful reads and their garbage text entered the
   305→   305→ * grounding context as "verified article content". A read whose extracted
   306→   306→ * text is an error/wall page (or too short to be an article) is treated as a
   307→   307→ * FAILED read: the source falls back to its search snippet and the failure
   308→   308→ * degrades honestly. No fabricated content verification.
   309→   309→ */
   310→   310→const PAGE_ERROR_SIGNATURES: RegExp[] = [
   311→   311→  /err[_-]blocked[_-]by[_-]client/i,
   312→   312→  /this page has been blocked/i,
   313→   313→  /\b404\b\s*(not found)?\b/i,
   314→   314→  /page not found/i,
   315→   315→  /access (denied|forbidden)/i,
   316→   316→  /enable javascript/i,
   317→   317→  /sign in to confirm/i,
   318→   318→  /just a moment\.{3,}/i,
   319→   319→  /checking your browser/i,
   320→   320→  /attention required/i,
   321→   321→  /unusual traffic from your computer/i,
   322→   322→  /are you a robot/i,
   323→   323→  /consent\.youtube\.com/i,
   324→   324→]
   325→   325→/** Minimum extracted text for a read to count as real article content. */
   326→   326→export const PAGE_MIN_USEFUL_CHARS = 200
   327→   327→
   328→   328→export function isUsablePageText(text: string): boolean {
   329→   329→  const trimmed = text.trim()
   330→   330→  if (trimmed.length < PAGE_MIN_USEFUL_CHARS) return false
   331→   331→  // A single error signature near the START of the text means the body we
   332→   332→  // captured IS the error page, not an article that merely mentions one.
   333→   333→  const head = trimmed.slice(0, 400)
   334→   334→  return !PAGE_ERROR_SIGNATURES.some((re) => re.test(head))
   335→   335→}

   337→   337→/**
   338→   338→ * In-flight dedup: identical concurrent retrieval work (same normalized
   339→   339→ * query / same URL) shares ONE provider call instead of racing duplicates.
   340→   340→ * Bounded naturally — entries live only for the duration of the call.
   341→   341→ */
   342→   342→const inFlightSearches = new Map<string, Promise<WebSearchResult[]>>()
   343→   343→const inFlightPages = new Map<string, Promise<PageContent>>()
   344→   344→
   345→   345→function searchInFlightKey(q: string, opts: { num?: number; recencyDays?: number }): string {
   346→   346→  return `${q.toLowerCase().replace(/\s+/g, ' ').trim()}|${opts.recencyDays ?? ''}|${opts.num ?? ''}`
   347→   347→}
   348→   348→
   349→   349→export function runSharedSearch(q: string, opts: { num?: number; recencyDays?: number } = {}): Promise<WebSearchResult[]> {
   350→   350→  const key = searchInFlightKey(q, opts)
   351→   351→  const existing = inFlightSearches.get(key)
   352→   352→  if (existing) return existing
   353→   353→  const p = webSearch(q, opts).finally(() => inFlightSearches.delete(key))
   354→   354→  inFlightSearches.set(key, p)
   355→   355→  return p
   356→   356→}
   357→   357→
   358→   358→export function runSharedPageRead(url: string): Promise<PageContent> {
   359→   359→  const key = url
   360→   360→  const existing = inFlightPages.get(key)
   361→   361→  if (existing) return existing
   362→   362→  const p = readPage(url, PAGE_READ_BUDGET_MS).finally(() => inFlightPages.delete(key))
   363→   363→  inFlightPages.set(key, p)
   364→   364→  return p
   365→   365→}

   367→   367→/**
   368→   368→ * One search attempt with EXACTLY ONE bounded retry: a provider failure or a
   369→   369→ * zero-result response triggers a single same-query retry after a short
   370→   370→ * backoff. Never loops, never fabricates, NEVER throws — the outcome carries
   371→   371→ * the honest attempt count (even when both attempts fail) and the real
   372→   372→ * provider error for the pack's degraded notes.
   373→   373→ */
   374→   374→export interface BoundedSearchOutcome {
   375→   375→  results: WebSearchResult[]
   376→   376→  /** Real provider calls made (0 for cache hits, 1-2 otherwise). */
   377→   377→  attempts: number
   378→   378→  retried: boolean
   379→   379→  /** Most recent provider error message when the search ultimately failed. */
   380→   380→  error: string | null
   381→   381→}
   382→   382→
   383→   383→export async function searchWithBoundedRetry(
   384→   384→  q: string,
   385→   385→  opts: { num?: number; recencyDays?: number },
   386→   386→): Promise<BoundedSearchOutcome> {
   387→   387→  try {
   388→   388→    const first = await runSharedSearch(q, opts)
   389→   389→    if (first.length > 0) return { results: first, attempts: 1, retried: false, error: null }
   390→   390→  } catch {
   391→   391→    // First attempt failed (transient provider error) — fall through to retry.
   392→   392→  }
   393→   393→  await new Promise((r) => setTimeout(r, SEARCH_RETRY_BACKOFF_MS))
   394→   394→  try {
   395→   395→    const second = await runSharedSearch(q, opts)
   396→   396→    return { results: second, attempts: 2, retried: true, error: null }
   397→   397→  } catch (e) {
   398→   398→    return {
   399→   399→      results: [],
   400→   400→      attempts: 2,
   401→   401→      retried: true,
   402→   402→      error: e instanceof Error ? e.message : 'unavailable',
   403→   403→    }
   404→   404→  }
   405→   405→}

   407→   407→/**
   408→   408→ * Execute the full research pipeline for one question.
   409→   409→ * Never throws: all tool failures are collected into `degraded`.

   411→   411→export async function runResearch(
   412→   412→  message: string,
   413→   413→  decision: SearchDecision,
   414→   414→  opts: RunResearchOptions = {},
   415→   415→): Promise<EvidencePack> {
   416→   416→  const t0 = Date.now()
   417→   417→  const degraded: string[] = []
   418→   418→  const maxSources = opts.maxSources ?? 6
   419→   419→
   420→   420→  let rawQueries: string[]
   421→   421→  if (isStraightforwardQuestion(message)) {
   422→   422→    // Fast path: no model call — the question (minus directive scaffolding)
   423→   423→    // is the strong query. When the topic names a place whose broader region
   424→   424→    // is missing from the wording, one region-anchored variant is added so
   425→   425→    // local reporting is actually retrievable (measured: "Gingee girl
   426→   426→    // missing" alone surfaced a stale, unrelated foreign article).
   427→   427→    rawQueries = fastPathQueryVariants(toSearchQuery(message))
   428→   428→  } else {
   429→   429→    const generated = await generateQueries(message, decision)
   430→   430→    if (generated.usedFallback) degraded.push('query refinement unavailable; searched the raw question')
   431→   431→    rawQueries = generated.queries
   432→   432→    // Same region guarantee for the refined path (bounded to 3 queries).
   433→   433→    const region = regionContextFor(message)
   434→   434→    if (region && !rawQueries.some((q) => q.toLowerCase().includes(region.toLowerCase()))) {
   435→   435→      rawQueries = [...rawQueries.slice(0, 2), `${rawQueries[0]} ${region}`]
   436→   436→    }
   437→   437→  }
   438→   438→  // Ongoing incident/case: deterministically target the NEWEST reporting and
   439→   439→  // official police/government statements (work-order priorities), bounded.
   440→   440→  const sensitiveSet = buildSensitiveQuerySet(message, rawQueries)
   441→   441→  rawQueries = sensitiveSet.queries
   442→   442→  if (sensitiveSet.addedFreshnessQuery) {
   443→   443→    degraded.push(`incident detected: added newest-coverage query "${sensitiveSet.addedFreshnessQuery.slice(0, 60)}"`)
   444→   444→  }
   445→   445→  if (sensitiveSet.addedOfficialQuery) {
   446→   446→    degraded.push(`incident detected: added official-statement query "${sensitiveSet.addedOfficialQuery.slice(0, 60)}"`)
   447→   447→  }
   448→   448→  const queries = dedupeQueries(rawQueries)
   449→   449→
   450→   450→  // --- retrieval (parallel, bounded, cache-aware) ---------------------------
   451→   451→  let searchCalls = 0
   452→   452→  let searchRetries = 0
   453→   453→  let cacheHits = 0
   454→   454→  let oldestFetchAt: number | null = null
   455→   455→  const tSearchStart = Date.now()
   456→   456→  const settled = await Promise.allSettled(
   457→   457→    queries.map(async (q): Promise<BoundedSearchOutcome> => {
   458→   458→      const cached = getSearchCache<WebSearchResult>(q, decision.recencyDays, decision.risk === 'none' ? 'moderate' : decision.risk)
   459→   459→      if (cached.hit) {
   460→   460→        cacheHits += 1
   461→   461→        oldestFetchAt = oldestFetchAt === null
   462→   462→          ? Date.parse(cached.fetchedAt)
   463→   463→          : Math.min(oldestFetchAt, Date.parse(cached.fetchedAt))
   464→   464→        return { results: cached.value, attempts: 0, retried: false, error: null }
   465→   465→      }
   466→   466→      const outcome = await searchWithBoundedRetry(q, {
   467→   467→        num: 6,
   468→   468→        ...(decision.recencyDays ? { recencyDays: decision.recencyDays } : {}),
   469→   469→      })
   470→   470→      if (outcome.error === null) {
   471→   471→        cached.put(outcome.results)
   472→   472→        oldestFetchAt = oldestFetchAt === null ? Date.now() : Math.min(oldestFetchAt, Date.now())
   473→   473→      }
   474→   474→      return outcome
   475→   475→    }),
   476→   476→  )
   477→   477→  const rawResults: WebSearchResult[] = []
   478→   478→  settled.forEach((s, i) => {
   479→   479→    if (s.status === 'fulfilled') {
   480→   480→      searchCalls += s.value.attempts
   481→   481→      if (s.value.retried) searchRetries += 1
   482→   482→      if (s.value.error !== null) {
   483→   483→        degraded.push(`search "${queries[i].slice(0, 60)}" failed: ${s.value.error}`)
   484→   484→      } else {
   485→   485→        rawResults.push(...s.value.results)
   486→   486→      }
   487→   487→    } else {
   488→   488→      degraded.push(
   489→   489→        `search "${queries[i].slice(0, 60)}" failed: ${
   490→   490→          s.reason instanceof ResearchToolError ? s.reason.message : 'unavailable'
   491→   491→        }`,
   492→   492→      )
   493→   493→    }
   494→   494→  })
   495→   495→  void searchRetries
   496→   496→  const searchMs = Date.now() - tSearchStart
   497→   497→
   498→   498→  if (rawResults.length === 0) {
   499→   499→    return {
   500→   500→      sources: [],
   501→   501→      queries,
   502→   502→      degraded: degraded.length > 0 ? degraded : ['web search returned no results'],
   503→   503→      staleness: null,
   504→   504→      geoWarning: null,
   505→   505→      durationMs: Date.now() - t0,
   506→   506→      searchMs,
   507→   507→      pageReadMs: 0,
   508→   508→      searchCalls,
   509→   509→      pageReadCalls: 0,
   510→   510→      cacheHits,
   511→   511→      fetchedAt: new Date(oldestFetchAt ?? Date.now()).toISOString(),
   512→   512→    }
   513→   513→  }
   514→   514→
   515→   515→  let ranked = rankSources(message, rawResults, { freshnessMatters: decision.risk === 'high', limit: maxSources })
   516→   516→
   517→   517→  // --- one bounded follow-up for newer reporting ----------------------------
   518→   518→  // Two triggers, ONE bounded follow-up (never loops; failures degrade
   519→   519→  // honestly):
   520→   520→  //  1. STALE EVIDENCE — every selected source is stale or undated for a
   521→   521→  //     freshness-sensitive question (existing trigger); a 30-day window
   522→   522→  //     hunts for newer reporting.
   523→   523→  //  2. ONGOING CASE — the question names an evolving case (missing person,
   524→   524→  //     rescue, accident investigation...) and the newest dated source is
   525→   525→  //     more than ~3 days old (work order: "search for newer relevant
   526→   526→  //     reporting before summarizing an ongoing case"). A case summary
   527→   527→  //     built on week-old reporting can silently miss a resolution; a
   528→   528→  //     10-day-window follow-up checks once, and older reporting still
   529→   529→  //     ranks only as background.
   530→   530→  if (decision.risk === 'high' && ranked.length > 0) {
   531→   531→    const newestAge = newestSourceAgeDays(ranked)
   532→   532→    const ongoing = isOngoingCaseQuery(queries[0])
   533→   533→    const staleEvidence = needsFreshnessFollowUp(ranked)
   534→   534→    const ongoingStale = ongoing && newestAge !== null && newestAge > 3
   535→   535→    const ongoingUndated = ongoing && newestAge === null
   536→   536→    if (staleEvidence || ongoingStale || ongoingUndated) {
   537→   537→      const followUpQuery = buildFollowUpQuery(queries[0])
   538→   538→      if (!queries.includes(followUpQuery)) {
   539→   539→        const recency = ongoing ? 10 : 30
   540→   540→        const outcome = await searchWithBoundedRetry(followUpQuery, { num: 6, recencyDays: recency })
   541→   541→        if (outcome.error !== null) {
   542→   542→          degraded.push(`freshness follow-up search failed: ${outcome.error}`)
   543→   543→        } else if (outcome.results.length > 0) {
   544→   544→          searchCalls += outcome.attempts
   545→   545→          rawResults.push(...outcome.results)
   546→   546→          ranked = rankSources(message, rawResults, { freshnessMatters: decision.risk === 'high', limit: maxSources })
   547→   547→          queries.push(followUpQuery)
   548→   548→        }
   549→   549→      }
   550→   550→    }
   551→   551→  }
   552→   552→
   553→   553→  // --- full-page reading, only where snippets are insufficient --------------
   554→   554→  const sources: EvidenceSource[] = ranked.map((r) => ({ ...r, pageRead: false, excerpt: null }))
   555→   555→  const ongoingCase = isOngoingCaseQuery(queries[0])
   556→   556→  const readCount = decidePageReads(ranked, decision, opts, ongoingCase)
   557→   557→  let pageReadCalls = 0
   558→   558→  let pageReadMs = 0
   559→   559→  if (readCount > 0) {
   560→   560→    const tPageStart = Date.now()
   561→   561→    const toRead = sources.slice(0, readCount)
   562→   562→    const readResults = await Promise.allSettled(
   563→   563→      toRead.map((s) => {
   564→   564→        const cached = getPageCache<{ url: string; text: string; publishedTime: string | null }>(s.url)
   565→   565→        if (cached.hit) {
   566→   566→          oldestFetchAt = oldestFetchAt === null
   567→   567→            ? Date.parse(cached.fetchedAt)
   568→   568→            : Math.min(oldestFetchAt, Date.parse(cached.fetchedAt))
   569→   569→          return Promise.resolve(cached.value)
   570→   570→        }
   571→   571→        pageReadCalls += 1
   572→   572→        return runSharedPageRead(s.url).then((page) => {
   573→   573→          const value = { url: page.url, text: page.text.slice(0, 6_000), publishedTime: page.publishedTime }
   574→   574→          cached.put(value)
   575→   575→          oldestFetchAt = oldestFetchAt === null ? Date.now() : Math.min(oldestFetchAt, Date.now())
   576→   576→          return value
   577→   577→        })
   578→   578→      }),
   579→   579→    )
   580→   580→    readResults.forEach((res) => {
   581→   581→      if (res.status === 'fulfilled') {
   582→   582→        const match = sources.find((s) => samePage(s.url, res.value.url))
   583→   583→        if (match) {
   584→   584→          if (isUsablePageText(res.value.text)) {
   585→   585→            match.pageRead = true
   586→   586→            match.excerpt = res.value.text
   587→   587→            // Publication-date verification: the page's own structured
   588→   588→            // metadata (JSON-LD/meta) is the authoritative publication date —
   589→   589→            // it OVERRIDES the search-engine-supplied date (often a crawl or
   590→   590→            // index date, not the article date).
   591→   591→            if (res.value.publishedTime) match.date = res.value.publishedTime
   592→   592→            // Last-resort date extraction from the article TEXT (dateline
   593→   593→            // "Published: 22 Oct 2026") when neither the provider nor the
   594→   594→            // page metadata supplied one. Explicit dateline only — never a
   595→   595→            // bare year mention in body copy.
   596→   596→            else if (!match.date) match.date = extractDatelineFromText(res.value.text)
   597→   597→          } else {
   598→   598→            // Content-verification failure: what came back is an error or
   599→   599→            // blocked page, NOT the article. Honest fallback to the snippet.
   600→   600→            degraded.push(
   601→   601→              `page read for ${match.hostName.replace(/^www\./, '')} returned an error or blocked page; grounded on the search snippet`,
   602→   602→            )
   603→   603→          }
   604→   604→        }
   605→   605→      } else {
   606→   606→        degraded.push(`page read failed: ${res.reason instanceof ResearchToolError ? res.reason.message : 'unavailable'}`)
   607→   607→      }
   608→   608→    })
   609→   609→    pageReadMs = Date.now() - tPageStart
   610→   610→  }
   611→   611→
   612→   612→  // re-parse freshness in case page reading supplied a publishedTime
   613→   613→  // Geographic relevance audit: when the topic named a place and NO selected
   614→   614→  // source mentions it, the retrieved reports are probably about a different
   615→   615→  // case — the answer must say so instead of presenting them as the asked case.
   616→   616→  const geoTerms = geoTermsOf(message)
   617→   617→  const anyGeoMatch = geoTerms.length === 0 || sources.some((s) => {
   618→   618→    const hay = `${s.title} ${s.snippet} ${s.excerpt ?? ''}`.toLowerCase()
   619→   619→    return geoTerms.some((t) => hay.includes(t))
   620→   620→  })
   621→   621→  const geoWarning =
   622→   622→    geoTerms.length > 0 && !anyGeoMatch
   623→   623→      ? `none of the retrieved reports mention ${geoTerms.slice(0, 3).join(', ')}; the specific case or place may not be identified`
   624→   624→      : null
   625→   625→
   626→   626→  return {
   627→   627→    sources,
   628→   628→    queries,
   629→   629→    degraded,
   630→   630→    staleness: stalenessWarning(sources, decision.risk === 'high'),
   631→   631→    geoWarning,
   632→   632→    durationMs: Date.now() - t0,
   633→   633→    searchMs,
   634→   634→    pageReadMs,
   635→   635→    searchCalls,
   636→   636→    pageReadCalls,
   637→   637→    cacheHits,
   638→   638→    fetchedAt: new Date(oldestFetchAt ?? Date.now()).toISOString(),
   639→   639→  }
   640→   640→}
   641→   641→
   642→   642→function samePage(a: string, b: string): boolean {
   643→   643→  try {
   644→   644→    const ua = new URL(a)
   645→   645→    const ub = new URL(b)
   646→   646→    return ua.hostname.replace(/^www\./, '') === ub.hostname.replace(/^www\./, '') && ua.pathname === ub.pathname
   647→   647→  } catch {
   648→   648→    return a === b
   649→   649→  }
   650→   650→}

   652→   652→/** Compact, numbered evidence block for the solver prompt. */
   653→   653→export function formatEvidenceBlock(pack: EvidencePack): string | null {
   654→   654→  if (pack.sources.length === 0) return null
   655→   655→  const lines = pack.sources.map((s) => {
   656→   656→    const date = s.date ? ` (dated ${s.date})` : ' (no date supplied)'
   657→   657→    const body = s.excerpt ? s.excerpt.slice(0, 1_200) : s.snippet.slice(0, 600)
   658→   658→    return `[${s.id}] ${s.title}${date}\nURL: ${s.url}\nEXCERPT: ${body}`
   659→   659→  })
   660→   660→  return lines.join('\n\n')
   661→   661→}
