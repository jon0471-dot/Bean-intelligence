    26→const encoder = new TextEncoder()

    28→/** Build a Response whose body replays the given STRING frames. */
    29→function sseResponse(frames: string[], opts: { delayMs?: number } = {}): Response {
    30→  const stream = new ReadableStream<Uint8Array>({
    31→    async start(c) {
    32→      for (const f of frames) {
    33→        c.enqueue(encoder.encode(f))
    34→        if (opts.delayMs) await new Promise((r) => setTimeout(r, opts.delayMs))
    35→      }
    36→      c.close()
    37→    },
    38→  })
    39→  return new Response(stream, { status: 200, headers: { 'content-type': 'text/event-stream' } })
    40→}
    41→
    42→/** Build a Response from RAW byte chunks (for multibyte split tests). */
    43→function byteResponse(chunks: Uint8Array[]): Response {
    44→  const stream = new ReadableStream<Uint8Array>({
    45→    start(c) {
    46→      for (const b of chunks) c.enqueue(b)
    47→      c.close()
    48→    },
    49→  })
    50→  return new Response(stream, { status: 200, headers: { 'content-type': 'text/event-stream' } })
    51→}

    53→/** Callback recorder for consumeRiisheStream. */
    54→function recorder() {
    55→  const events: RiisheStreamEvent[] = []
    56→  const failures: StreamFailureInfo[] = []
    57→  let completed: RiisheStreamEvent | null = null
    58→  return {
    59→    callbacks: {
    60→      onEvent: (e: RiisheStreamEvent) => events.push(e),
    61→      onComplete: (e: RiisheStreamEvent) => {
    62→        completed = e
    63→      },
    64→      onFailure: (f: StreamFailureInfo) => failures.push(f),
    65→    },
    66→    events,
    67→    failures,
    68→    get completed() {
    69→      return completed
    70→    },
    71→  }
    72→}
