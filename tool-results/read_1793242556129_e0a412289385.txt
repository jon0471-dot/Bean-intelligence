    95→function useCopy(): [boolean, (text: string) => void] {
    96→  const [copied, setCopied] = useState(false)
    97→  const copy = useCallback((text: string) => {
    98→    const done = () => {
    99→      setCopied(true)
   100→      setTimeout(() => setCopied(false), 1600)
   101→    }
   102→    if (navigator.clipboard?.writeText) {
   103→      navigator.clipboard.writeText(text).then(done).catch(done)
   104→    } else {
   105→      const ta = document.createElement('textarea')
   106→      ta.value = text
   107→      document.body.appendChild(ta)
   108→      ta.select()
   109→      document.execCommand('copy')
   110→      document.body.removeChild(ta)
   111→      done()
   112→    }
   113→  }, [])
   114→  return [copied, copy]
   115→}

   145→function formatBytes(n: number): string {
   146→  if (n < 1024) return `${n} B`
   147→  if (n < 1024 * 1024) return `${(n / 1024).toFixed(1)} KB`
   148→  return `${(n / (1024 * 1024)).toFixed(1)} MB`
   149→}

   240→  async function uploadOne(file: File) {
   241→    const form = new FormData()
   242→    form.append('file', file)
   243→    const res = await fetch('/api/riishe/documents', { method: 'POST', body: form })
   244→    const data = await res.json().catch(() => ({}))
   245→    if (res.status === 401) window.dispatchEvent(new Event('bean:unauthorized'))
   246→    if (!res.ok && res.status !== 415) {
   247→      throw new Error(data?.error?.message ?? `Upload failed (${res.status})`)
   248→    }
   249→    if (res.status === 415 && !data?.document) {
   250→      // Unsafe or rejected outright — surface the server's actionable reason.
   251→      throw new Error(data?.error?.message ?? 'This file cannot be processed.')
   252→    }
   253→    setFiles((prev) => [...prev, data.document as FileMeta].slice(0, MAX_FILES))
   254→    if (data.document?.status === 'unsupported') {
   255→      setError(data.document.note ?? 'This file type cannot be analyzed yet — it stays attached but adds nothing to the answer.')
   256→    }
   257→  }

   259→  async function uploadFiles(list: FileList | File[]) {
   260→    const incoming = Array.from(list)
   261→    if (incoming.length === 0 || uploadingCount > 0) return
   262→    const room = MAX_FILES - files.length
   263→    if (room <= 0) {
   264→      setError(`Up to ${MAX_FILES} attachments per message.`)
   265→      return
   266→    }
   267→    setError(null)
   268→    setUploadingCount(incoming.length)
   269→    const batch = incoming.slice(0, room)
   270→    let failures = 0
   271→    for (const f of batch) {
   272→      try {
   273→        await uploadOne(f)
   274→      } catch (e) {
   275→        failures += 1
   276→        setError((e as Error).message)
   277→      } finally {
   278→        setUploadingCount((c) => Math.max(0, c - 1))
   279→      }
   280→    }
   281→    if (failures > 0 && batch.length > failures) {
   282→      setError('Some files were rejected — the message explains each failure.')
   283→    }
   284→  }

   286→  function stopGeneration() {

   290→  /** Regenerate the response to a SPECIFIC user turn (consistent actions on
   291→   *  every assistant message). For the trailing exchange this is the classic
   292→   *  retry; for earlier messages the server removes that exchange and
   293→   *  everything after it, then answers the same user turn again. */
   294→  function regenerateFrom(asstMsg: Msg) {
   295→    if (streaming) return
   296→    const idx = messages.findIndex((m) => m.id === asstMsg.id)
   297→    if (idx < 0) return
   298→    let userIdx = -1
   299→    for (let i = idx - 1; i >= 0; i -= 1) {
   300→      if (messages[i].role === 'user') { userIdx = i; break }
   301→    }
   302→    if (userIdx < 0) return
   303→    const userContent = messages[userIdx].content
   304→    const isTrailing = idx === messages.length - 1
   305→    // Without a persisted id a mid-conversation regenerate cannot truncate
   306→    // server-side history — refuse rather than silently duplicating the turn.
   307→    if (!isTrailing && !asstMsg.serverId) return
   308→    // Drop everything from the user turn onward (locally; the server mirrors
   309→    // this for earlier messages so history stays an honest sequence).
   310→    setMessages(messages.slice(0, userIdx))
   311→    setTimeout(() => void send(userContent, { retry: isTrailing && !asstMsg.serverId, retryFromId: isTrailing ? undefined : asstMsg.serverId }), 0)
   312→  }

   314→  async function send(overrideMessage?: string, opts?: { retry?: boolean; retryFromId?: string }) {
   315→    const message = (overrideMessage ?? input).trim()
   316→    if (!message || streaming) return
   317→    if (overrideMessage === undefined) setInput('')
   318→    setError(null)
   319→    setStreaming(true)
   320→    setResearching(false)
   321→    setRunningTool(null)
   322→    nearBottomRef.current = true // a sent message always scrolls into view
   323→    streamedRef.current = ''
   324→
   325→    const userMsg: Msg = { id: `u-${Date.now()}`, role: 'user', content: message }
   326→    const asstId = `a-${Date.now()}`
   327→    setMessages((prev) => [...prev, userMsg, { id: asstId, role: 'assistant', content: '', pending: true }])
   328→
   329→    const controller = new AbortController()
   330→    abortRef.current = controller
   331→    const attachable = files.filter((f) => f.status === 'ready')
   332→
   333→    try {
   334→      const res = await fetch(`/api/riishe/chat?stream=1${activeId ? `&conversationId=${activeId}` : ''}`, {
   335→        method: 'POST',
   336→        headers: { 'content-type': 'application/json' },
   337→        body: JSON.stringify({
   338→          message,
   339→          conversationId: activeId ?? undefined,
   340→          documentIds: attachable.length > 0 ? attachable.map((d) => d.id) : undefined,
   341→          retry: opts?.retry ?? false,
   342→          ...(opts?.retryFromId ? { retryFromMessageId: opts.retryFromId } : {}),
   343→        }),
   344→        signal: controller.signal,
   345→      })
   346→      if (!res.ok || !res.body) {
   347→        // Keep the global identity in sync: a 401 here means the session is
   348→        // gone server-side; drop the signed-in shell immediately.
   349→        if (res.status === 401) window.dispatchEvent(new Event('bean:unauthorized'))
   350→        const data = await res.json().catch(() => ({}))
   351→        throw new Error(data?.error?.message ?? `Request failed (${res.status})`)
   352→      }
   353→      const reader = res.body.getReader()
   354→      const decoder = new TextDecoder()
   355→      let buffer = ''
   356→      let convId: string | null = activeId
   357→      let failed = false
   358→
   359→      while (true) {
   360→        const { done, value } = await reader.read()
   361→        if (done) break
   362→        buffer += decoder.decode(value, { stream: true })
   363→        const lines = buffer.split('\n')
   364→        buffer = lines.pop() ?? ''
   365→        for (const line of lines) {
   366→          if (!line.startsWith('data: ')) continue
   367→          let evt: {
   368→            delta?: string
   369→            reset?: boolean
   370→            done?: boolean
   371→            error?: string
   372→            conversationId?: string
   373→            messageId?: string
   374→            partial?: boolean
   375→            stage?: 'researching' | 'tools' | 'solving' | 'verifying'
   376→            tool?: { name: string; status: 'started' | 'done' | 'failed' }
   377→            meta?: MsgMeta
   378→          }
   379→          try { evt = JSON.parse(line.slice(6)) } catch { continue }
   380→          if (evt.conversationId) convId = evt.conversationId
   381→          if (evt.stage === 'researching') setResearching(true)
   382→          if (evt.stage === 'tools') { setResearching(false); setRunningTool('working') }
   383→          if (evt.stage === 'solving' || evt.stage === 'verifying') { setResearching(false); setRunningTool(null) }
   384→          if (evt.tool) {
   385→            if (evt.tool.status === 'started') setRunningTool(evt.tool.name)
   386→            else setRunningTool(null)
   387→          }
   388→          if (evt.reset) {
   389→            // Reset protocol: a verification rewrite replaced the streamed
   390→            // draft — clear it so the final answer REPLACES, never appends.
   391→            streamedRef.current = ''
   392→            setMessages((prev) =>
   393→              prev.map((m) => (m.id === asstId ? { ...m, content: '', pending: true } : m)),
   394→            )
   395→          }
   396→          if (evt.error) { failed = true; setError(evt.error) }
   397→          if (evt.delta) {
   398→            streamedRef.current += evt.delta
   399→            setMessages((prev) =>
   400→              prev.map((m) => (m.id === asstId ? { ...m, content: streamedRef.current, pending: false } : m)),
   401→            )
   402→          }
   403→          if (evt.done) {
   404→            setResearching(false)
   405→            setRunningTool(null)
   406→            if (convId && !activeId) setActiveId(convId)
   407→            if (evt.messageId) {
   408→              setMessages((prev) => prev.map((m) => (m.id === asstId ? { ...m, serverId: evt.messageId } : m)))
   409→            }
   410→            if (evt.meta) {
   411→              setMessages((prev) => prev.map((m) => (m.id === asstId ? { ...m, meta: evt.meta } : m)))
   412→            }
   413→          }
   414→        }
   415→      }
   416→      // If nothing streamed (failed, or the server reported an empty partial),
   417→      // drop the empty assistant bubble — the error banner already explains
   418→      // what happened, and a perpetual "thinking" indicator would be a lie.
   419→      if (streamedRef.current === '') {
   420→        setMessages((prev) => prev.filter((m) => m.id !== asstId))
   421→      } else if (failed) {
   422→        setMessages((prev) => prev.map((m) => (m.id === asstId ? { ...m, partial: true, pending: false } : m)))
   423→      }
   424→      void loadConversations()
   425→    } catch (e) {
   426→      if ((e as Error).name === 'AbortError') {
   427→        // User pressed stop: keep what arrived and label it honestly.
   428→        if (streamedRef.current !== '') {
   429→          setMessages((prev) => prev.map((m) => (m.id === asstId ? { ...m, partial: true, pending: false } : m)))
   430→        } else {
   431→          setMessages((prev) => prev.filter((m) => m.id !== asstId))
   432→        }
   433→      } else {
   434→        setError((e as Error).message)
   435→        setMessages((prev) => prev.filter((m) => m.id !== asstId || m.content.length > 0))
   436→      }
   437→    } finally {
   438→      setStreaming(false)
   439→      setResearching(false)
   440→      setRunningTool(null)
   441→      abortRef.current = null
   442→      setMessages((prev) => prev.map((m) => ({ ...m, pending: false })))
   443→    }
   444→  }

   679→/**
   680→ * One assistant response: clean prose (citation markers render as source-name
   681→ * links; unresolved ones are stripped), the honest verification note when one
   682→ * exists, and CONSISTENT actions — Copy response, Regenerate, and View
   683→ * sources (only when sources exist) — on EVERY response, not just the last.
   684→ */
   685→function AssistantBubble({
   686→  msg,
   687→  onRegenerate,
   688→  streaming,
   689→  renderers,
   690→  researching,
   691→  runningTool,
   692→}: {
   693→  msg: Msg
   694→  onRegenerate: () => void
   695→  streaming: boolean
   696→  renderers: Parameters<typeof Markdown>[0]['components']
   697→  researching: boolean
   698→  runningTool: string | null
   699→}) {
   700→  const [sourcesOpen, setSourcesOpen] = useState(false)
   701→  const hasSources = (msg.meta?.sources?.length ?? 0) > 0
   702→  if (!msg.content) {
   703→    // Empty draft: the live status line (search / tools / thinking).
   704→    return researching ? (
   705→      <span className="inline-flex items-center gap-2 text-xs text-white/50">
   706→        <Globe className="h-3.5 w-3.5 animate-pulse text-[#d4af37]" /> Searching the web…
   707→      </span>
   708→    ) : runningTool ? (
   709→      <span className="inline-flex items-center gap-2 text-xs text-white/50">
   710→        <Terminal className="h-3.5 w-3.5 animate-pulse text-[#d4af37]" />
   711→        {runningTool === 'working' ? 'Working…' : 'Running tools…'}
   712→      </span>
   713→    ) : (
   714→      <span className="inline-flex items-center gap-2 text-xs text-white/40">
   715→        <Spinner /> Thinking…
   716→      </span>
   717→    )
   718→  }
   719→  const claims = mapSourceClaims(msg.content, msg.meta?.sources ?? [])
   720→  return (
   721→    <>
   722→      <Markdown
   723→        content={linkifyCitations(msg.content, msg.meta?.sources)}
   724→        components={renderers}
   725→      />
   726→      {msg.pending ? <Spinner className="ml-1 mt-1" /> : null}
   727→      {msg.partial ? (
   728→        <div className="mt-2 inline-flex items-center gap-1.5 rounded border border-amber-500/30 bg-amber-500/10 px-2 py-0.5 text-[10px] text-amber-300">
   729→          <AlertTriangle className="h-3 w-3" /> Incomplete — the stream was stopped or ended early
   730→        </div>
   731→      ) : null}
   732→      <AnswerMeta meta={msg.meta} forceOpen={sourcesOpen} claims={claims} />
   733→      {!msg.pending ? (
   734→        <div className="mt-2 flex flex-wrap items-center gap-2 border-t border-[rgba(212,175,55,0.12)] pt-2">
   735→          <CopyMessageButton content={msg.content} />
   736→          <button
   737→            onClick={onRegenerate}
   738→            disabled={streaming}
   739→            className="inline-flex items-center gap-1 rounded px-1.5 py-0.5 text-[10px] text-white/45 transition-colors hover:bg-white/10 hover:text-white/80 disabled:opacity-40"
   740→            aria-label="Regenerate response"
   741→          >
   742→            <RefreshCw className="h-3 w-3" /> Regenerate
   743→          </button>
   744→          {hasSources ? (
   745→            <button
   746→              onClick={() => setSourcesOpen(true)}
   747→              className="inline-flex items-center gap-1 rounded px-1.5 py-0.5 text-[10px] text-white/45 transition-colors hover:bg-white/10 hover:text-white/80"
   748→              aria-label="View sources"
   749→              aria-expanded={sourcesOpen}
   750→            >
   751→              <ExternalLink className="h-3 w-3" /> View sources
   752→            </button>
   753→          ) : null}
   754→        </div>
   755→      ) : null}
   756→    </>
   757→  )
   758→}

   774→/**
   775→ * Answer-first evidence: NO state badges, NO verification summary, NO tool
   776→ * chips, NO "last checked" telemetry in the reading flow. What remains:
   777→ *  - a short honest verification note ONLY when something material could not
   778→ *    be confirmed (uncovered claims, stale sources, wrong-case retrieval,
   779→ *    failed live verification);
   780→ *  - a subtle collapsed "Sources" section listing what was really retrieved,
   781→ *    each entry with its publisher, date, and the claim it supports.
   782→ */
   783→function AnswerMeta({
   784→  meta,
   785→  forceOpen,
   786→  claims,
   787→}: {
   788→  meta?: MsgMeta
   789→  forceOpen?: boolean
   790→  claims: Map<number, string>
   791→}) {
   792→  const detailsRef = useRef<HTMLDetailsElement>(null)
   793→  useEffect(() => {
   794→    if (forceOpen && detailsRef.current) detailsRef.current.open = true
   795→  }, [forceOpen])
   796→  const sources = meta?.sources ?? []
   797→  const note = meta?.verificationNote
   798→  const researchFailed = meta?.states?.includes('TOOL_UNAVAILABLE') ?? false
   799→  if (!researchFailed && !note && sources.length === 0) return null
   800→  return (
   801→    <div className="mt-2 border-t border-[rgba(212,175,55,0.12)] pt-2">
   802→      {researchFailed ? (
   803→        <div className="text-[11px] text-amber-300/90">
   804→          Live sources could not be checked for this answer — treat current details as unverified.
   805→        </div>
   806→      ) : note ? (
   807→        <div className="text-[11px] leading-snug text-white/40">{note}</div>
   808→      ) : null}
   809→
   810→      {sources.length > 0 ? (
   811→        <details ref={detailsRef} className="group mt-1">
   812→          <summary className="flex w-fit cursor-pointer list-none items-center gap-1 text-[11px] text-white/40 hover:text-white/65">
   813→            <ChevronDown className="h-3 w-3 transition-transform group-open:rotate-180" />
   814→            Sources ({sources.length})
   815→          </summary>
   816→          <ul className="mt-1.5 space-y-2 pl-4">
   817→            {sources.map((s) => (
   818→              <li key={s.id} id={`bean-src-${s.id}`} className="scroll-mt-6 break-words text-[11px] leading-snug">
   819→                <a
   820→                  href={s.url}
   821→                  target="_blank"
   822→                  rel="noopener noreferrer"
   823→                  className="break-words text-[#f0c75e] hover:underline"
   824→                >
   825→                  {s.title.slice(0, 90)}
   826→                  <ExternalLink className="ml-1 inline h-2.5 w-2.5" />
   827→                </a>
   828→                <span className="text-white/35"> — {s.hostName.replace(/^www\./, '')}</span>

   830→                  <span className="ml-1 text-white/30" title="Publication date stated by the source — distinct from the event date">
   831→                    · published {s.date}
   832→                  </span>
   833→                ) : (
   834→                  <span className="ml-1 text-white/25" title="No publication date could be verified for this source">
   835→                    · no date verified
   836→                  </span>
   837→                )}
   838→                {s.social ? <span className="ml-1 text-amber-300/50">· social-media post (unverified provenance)</span> : null}
   839→                <SourceClaimLine content={claims.get(s.id)} />
   840→              </li>
   841→            ))}
   842→          </ul>
   843→        </details>
   844→      ) : null}
   845→    </div>
   846→  )
   847→}
