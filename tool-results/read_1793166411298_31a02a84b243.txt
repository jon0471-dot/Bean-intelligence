    96→    96→function useCopy(): [boolean, (text: string) => void] {
    97→    97→  const [copied, setCopied] = useState(false)
    98→    98→  const copy = useCallback((text: string) => {
    99→    99→    const done = () => {
   100→   100→      setCopied(true)
   101→   101→      setTimeout(() => setCopied(false), 1600)
   102→   102→    }
   103→   103→    if (navigator.clipboard?.writeText) {
   104→   104→      navigator.clipboard.writeText(text).then(done).catch(done)
   105→   105→    } else {
   106→   106→      const ta = document.createElement('textarea')
   107→   107→      ta.value = text
   108→   108→      document.body.appendChild(ta)
   109→   109→      ta.select()
   110→   110→      document.execCommand('copy')
   111→   111→      document.body.removeChild(ta)
   112→   112→      done()
   113→   113→    }
   114→   114→  }, [])
   115→   115→  return [copied, copy]
   116→   116→}

   273→   273→  /** Regenerate the response to a SPECIFIC user turn (consistent actions on
   274→   274→   *  every assistant message). For the trailing exchange this is the classic
   275→   275→   *  retry; for earlier messages the server removes that exchange and
   276→   276→   *  everything after it, then answers the same user turn again. */
   277→   277→  function regenerateFrom(asstMsg: Msg) {
   278→   278→    if (streaming) return
   279→   279→    const idx = messages.findIndex((m) => m.id === asstMsg.id)
   280→   280→    if (idx < 0) return
   281→   281→    let userIdx = -1
   282→   282→    for (let i = idx - 1; i >= 0; i -= 1) {
   283→   283→      if (messages[i].role === 'user') { userIdx = i; break }
   284→   284→    }
   285→   285→    if (userIdx < 0) return
   286→   286→    const userContent = messages[userIdx].content
   287→   287→    const isTrailing = idx === messages.length - 1
   288→   288→    // Without a persisted id a mid-conversation regenerate cannot truncate
   289→   289→    // server-side history — refuse rather than silently duplicating the turn.
   290→   290→    if (!isTrailing && !asstMsg.serverId) return
   291→   291→    // Drop everything from the user turn onward (locally; the server mirrors
   292→   292→    // this for earlier messages so history stays an honest sequence).
   293→   293→    setMessages(messages.slice(0, userIdx))
   294→   294→    setTimeout(() => void send(userContent, { retry: isTrailing && !asstMsg.serverId, retryFromId: isTrailing ? undefined : asstMsg.serverId }), 0)
   295→   295→  }

   297→   297→  async function send(overrideMessage?: string, opts?: { retry?: boolean; retryFromId?: string }) {
   298→   298→    const message = (overrideMessage ?? input).trim()
   299→   299→    if (!message || streaming) return
   300→   300→    if (overrideMessage === undefined) setInput('')
   301→   301→    setError(null)
   302→   302→    setStreaming(true)
   303→   303→    setResearching(false)
   304→   304→    setRunningTool(null)
   305→   305→    streamedRef.current = ''
   306→   306→
   307→   307→    const userMsg: Msg = { id: `u-${Date.now()}`, role: 'user', content: message }
   308→   308→    const asstId = `a-${Date.now()}`
   309→   309→    setMessages((prev) => [...prev, userMsg, { id: asstId, role: 'assistant', content: '', pending: true }])
   310→   310→
   311→   311→    const controller = new AbortController()
   312→   312→    abortRef.current = controller
   313→   313→    const attachable = files.filter((f) => f.status === 'ready')
   314→   314→
   315→   315→    try {
   316→   316→      const res = await fetch(`/api/riishe/chat?stream=1${activeId ? `&conversationId=${activeId}` : ''}`, {
   317→   317→        method: 'POST',
   318→   318→        headers: { 'content-type': 'application/json' },
   319→   319→        body: JSON.stringify({
   320→   320→          message,
   321→   321→          conversationId: activeId ?? undefined,
   322→   322→          documentIds: attachable.length > 0 ? attachable.map((d) => d.id) : undefined,
   323→   323→          retry: opts?.retry ?? false,
   324→   324→          ...(opts?.retryFromId ? { retryFromMessageId: opts.retryFromId } : {}),
   325→   325→        }),
   326→   326→        signal: controller.signal,
   327→   327→      })
   328→   328→      if (!res.ok || !res.body) {
   329→   329→        // Keep the global identity in sync: a 401 here means the session is
   330→   330→        // gone server-side; drop the signed-in shell immediately.
   331→   331→        if (res.status === 401) window.dispatchEvent(new Event('bean:unauthorized'))
   332→   332→        const data = await res.json().catch(() => ({}))
   333→   333→        throw new Error(data?.error?.message ?? `Request failed (${res.status})`)
   334→   334→      }
   335→   335→      const reader = res.body.getReader()
   336→   336→      const decoder = new TextDecoder()
   337→   337→      let buffer = ''
   338→   338→      let convId: string | null = activeId
   339→   339→      let failed = false
   340→   340→
   341→   341→      while (true) {
   342→   342→        const { done, value } = await reader.read()
   343→   343→        if (done) break
   344→   344→        buffer += decoder.decode(value, { stream: true })
   345→   345→        const lines = buffer.split('\n')
   346→   346→        buffer = lines.pop() ?? ''
   347→   347→        for (const line of lines) {
   348→   348→          if (!line.startsWith('data: ')) continue
   349→   349→          let evt: {
   350→   350→            delta?: string
   351→   351→            reset?: boolean
   352→   352→            done?: boolean
   353→   353→            error?: string
   354→   354→            conversationId?: string
   355→   355→            messageId?: string
   356→   356→            partial?: boolean
   357→   357→            stage?: 'researching' | 'tools' | 'solving' | 'verifying'
   358→   358→            tool?: { name: string; status: 'started' | 'done' | 'failed' }
   359→   359→            meta?: MsgMeta
   360→   360→          }
   361→   361→          try { evt = JSON.parse(line.slice(6)) } catch { continue }
   362→   362→          if (evt.conversationId) convId = evt.conversationId
   363→   363→          if (evt.stage === 'researching') setResearching(true)
   364→   364→          if (evt.stage === 'tools') { setResearching(false); setRunningTool('working') }
   365→   365→          if (evt.stage === 'solving' || evt.stage === 'verifying') { setResearching(false); setRunningTool(null) }
   366→   366→          if (evt.tool) {
   367→   367→            if (evt.tool.status === 'started') setRunningTool(evt.tool.name)
   368→   368→            else setRunningTool(null)
   369→   369→          }
   370→   370→          if (evt.reset) {
   371→   371→            // Reset protocol: a verification rewrite replaced the streamed
   372→   372→            // draft — clear it so the final answer REPLACES, never appends.
   373→   373→            streamedRef.current = ''
   374→   374→            setMessages((prev) =>
   375→   375→              prev.map((m) => (m.id === asstId ? { ...m, content: '', pending: true } : m)),
   376→   376→            )
   377→   377→          }
   378→   378→          if (evt.error) { failed = true; setError(evt.error) }
   379→   379→          if (evt.delta) {
   380→   380→            streamedRef.current += evt.delta
   381→   381→            setMessages((prev) =>
   382→   382→              prev.map((m) => (m.id === asstId ? { ...m, content: streamedRef.current, pending: false } : m)),
   383→   383→            )
   384→   384→          }
   385→   385→          if (evt.done) {
   386→   386→            setResearching(false)
   387→   387→            setRunningTool(null)
   388→   388→            if (convId && !activeId) setActiveId(convId)
   389→   389→            if (evt.messageId) {
   390→   390→              setMessages((prev) => prev.map((m) => (m.id === asstId ? { ...m, serverId: evt.messageId } : m)))
   391→   391→            }
   392→   392→            if (evt.meta) {
   393→   393→              setMessages((prev) => prev.map((m) => (m.id === asstId ? { ...m, meta: evt.meta } : m)))
   394→   394→            }
   395→   395→          }
   396→   396→        }
   397→   397→      }
   398→   398→      // If nothing streamed (failed, or the server reported an empty partial),
   399→   399→      // drop the empty assistant bubble — the error banner already explains
   400→   400→      // what happened, and a perpetual "thinking" indicator would be a lie.
   401→   401→      if (streamedRef.current === '') {
   402→   402→        setMessages((prev) => prev.filter((m) => m.id !== asstId))
   403→   403→      } else if (failed) {
   404→   404→        setMessages((prev) => prev.map((m) => (m.id === asstId ? { ...m, partial: true, pending: false } : m)))
   405→   405→      }
   406→   406→      void loadConversations()
   407→   407→    } catch (e) {
   408→   408→      if ((e as Error).name === 'AbortError') {
   409→   409→        // User pressed stop: keep what arrived and label it honestly.
   410→   410→        if (streamedRef.current !== '') {
   411→   411→          setMessages((prev) => prev.map((m) => (m.id === asstId ? { ...m, partial: true, pending: false } : m)))
   412→   412→        } else {
   413→   413→          setMessages((prev) => prev.filter((m) => m.id !== asstId))
   414→   414→        }
   415→   415→      } else {
   416→   416→        setError((e as Error).message)
   417→   417→        setMessages((prev) => prev.filter((m) => m.id !== asstId || m.content.length > 0))
   418→   418→      }
   419→   419→    } finally {
   420→   420→      setStreaming(false)
   421→   421→      setResearching(false)
   422→   422→      setRunningTool(null)
   423→   423→      abortRef.current = null
   424→   424→      setMessages((prev) => prev.map((m) => ({ ...m, pending: false })))
   425→   425→    }
   426→   426→  }
